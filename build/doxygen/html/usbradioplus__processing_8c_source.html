291{

293 const char *application = ast_channel_appl(chan);
294 const char *data = ast_channel_data(chan);
295

761PROCESSING_PRIVATE int validate_profile(const struct txagc_profile *value)
762{
763 int source;
764 if (ast_strlen_zero(value->channel)) {
765 return -1;
766 }
767 if ((value->hardware.input_gain_configured &&
768 (value->hardware.input_gain_db < -30.0 || value->hardware.input_gain_db > 30.0)) ||
769 (value->hardware.output_a_gain_configured &&
770 (value->hardware.output_a_gain_db < -30.0 ||
771 value->hardware.output_a_gain_db > 30.0)) ||
772 (value->hardware.output_b_gain_configured &&
773 (value->hardware.output_b_gain_db < -30.0 ||
774 value->hardware.output_b_gain_db > 30.0))) {
775 ast_log(LOG_ERROR, "RadioPlus [hardware]: gain must be between -30 and 30 dB\n");
776 return -1;
777 }
778 for (source = 0; source < TXAGC_SOURCE_COUNT; ++source) {
779 if (validate_chain(&value->chains[source])) {
780 ast_log(LOG_ERROR, "RadioPlus [%s]: one or more values are out of range\n",
781 source_names[source]);
782 return -1;
783 }
784 if (source != TXAGC_LOCAL && value->chains[source].rnnoise_enabled) {
785 ast_log(LOG_ERROR, "RadioPlus [%s]: RNNoise is local-receiver-only\n",
786 source_names[source]);
787 return -1;
788 }
789 if (source != TXAGC_LOCAL &&
790 value->chains[source].agc.ctcss_filter_mode != TXAGC_CTCSS_FILTER_DISABLED) {
791 ast_log(LOG_ERROR,
792 "RadioPlus [%s]: CTCSS receive filtering is local-receiver-only\n",
793 source_names[source]);
794 return -1;
795 }
796 if (source != TXAGC_LOCAL && value->chains[source].agc.receive_bandpass_enabled) {
797 ast_log(LOG_ERROR,
798 "RadioPlus [%s]: receive band-pass is local-receiver-only\n",
799 source_names[source]);
800 return -1;
801 }
802 if (source != TXAGC_VOICE_TELEMETRY &&
803 (value->chains[source].agc.lookahead_limiter_enabled ||
804 value->chains[source].agc.post_limiter_lowpass_enabled)) {
805 ast_log(LOG_ERROR,
806 "RadioPlus [%s]: transmitter-tail stages are valid only in "
807 "[voice_telemetry]\n",
808 source_names[source]);
809 return -1;
810 }
811 }
812 return 0;
813}

1292PROCESSING_PRIVATE int add_override(struct txagc_profile *updated, struct ast_config *cfg,
1293 const char *section, const char *name)
1294{
1295 const char *value = ast_variable_retrieve(cfg, section, name);
1296 struct section_override *entry;
1297 char *end;
1298 long integer_limit;
1299 if (!value)
1300 return 0;
1301 if (strstr(name, "_enabled") || strstr(name, "_inverted")) {
1302 if (!ast_true(value) && !ast_false(value))
1303 goto invalid;
1304 } else if (!strcasecmp(name, "asterisk_jitter_buffer_implementation")) {
1305 if (strcasecmp(value, "fixed") && strcasecmp(value, "adaptive"))
1306 goto invalid;
1307 } else if (clean_slate_integer_limit(name, &integer_limit)) {
1308 if (!valid_nonnegative_integer(value, integer_limit))
1309 goto invalid;
1310 } else if (!strcasecmp(name, "hardware_deemphasis_corner_hz") ||
1311 !strcasecmp(name, "hardware_preemphasis_corner_hz")) {
1312 double frequency = strtod(value, &end);
1313 if (end == value || *end || !isfinite(frequency) || frequency <= 0.0 ||
1314 frequency > 500.0)
1315 goto invalid;
1316 } else if (!strncasecmp(name, "hardware_gpio_", 14)) {
1317 if (strcasecmp(value, "in") && strcasecmp(value, "out0") &&
1318 strcasecmp(value, "out1"))
1319 goto invalid;
1320 } else if (!strncasecmp(name, "hardware_parallel_pin_", 22)) {
1321 int input_pin = strstr(name, "_10_") || strstr(name, "_12_") ||
1322 strstr(name, "_13_") || strstr(name, "_15_");

1324 if (strcasecmp(value, "in") && strcasecmp(value, "cor") &&
1325 strcasecmp(value, "ctcss"))
1326 goto invalid;
1327 } else if (strcasecmp(value, "out0") && strcasecmp(value, "out1") &&
1328 strcasecmp(value, "ptt"))
1329 goto invalid;
1330 } else if (!strcasecmp(name, "hardware_parallel_port_device")) {
1331 if (ast_strlen_zero(value))
1332 goto invalid;
1333 } else if (!strcasecmp(name, "hardware_parallel_port_base_address")) {
1334 unsigned long address = strtoul(value, &end, 0);
1335 if (end == value || *end || address > UINT32_MAX)
1336 goto invalid;
1337 } else if (!strcasecmp(name, "audio_source")) {
1338 if (strcasecmp(value, "no") && strcasecmp(value, "speaker") &&
1339 strcasecmp(value, "flat"))
1340 goto invalid;
1341 } else if (!strcasecmp(name, "receive_source")) {
1342 if (strcasecmp(value, "no") && strcasecmp(value, "usb") &&
1343 strcasecmp(value, "usbinvert") && strcasecmp(value, "dsp") &&
1344 strcasecmp(value, "pp") && strcasecmp(value, "ppinvert"))
1345 goto invalid;
1346 } else if (!strcasecmp(name, "receive_frequencies") ||
1347 !strcasecmp(name, "transmit_frequencies")) {
1348 if (!valid_frequency_list(value))
1349 goto invalid;
1350 } else if (!strcasecmp(name, "transmit_default_hz")) {
1351 double number = strtod(value, &end);
1352 /* A zero value is not a disabled setting: the radio engine would
1353 * silently replace it with a compatibility default. Reject it here
1354 * so the configuration describes the requested signaling exactly. */
1355 if (end == value || *end || !isfinite(number) || number <= 0.0 ||
1356 !urp_ctcss_frequency_supported((float)number))
1357 goto invalid;
1358 } else if (!strcasecmp(name, "phase_shift_degrees") ||
1359 !strcasecmp(name, "tail_frequency_hz")) {
1360 double number = strtod(value, &end);
1361 if (end == value || *end || !isfinite(number) || number <= 0.0)
1362 goto invalid;
1363 } else if (!strcasecmp(name, "tail_duration_ms")) {
1364 long duration = strtol(value, &end, 0);
1365 /* Phase and replacement-tone tails reserve two 20 ms state frames. */
1366 if (end == value || *end || duration < 2L * MS_PER_FRAME || duration > INT16_MAX)
1367 goto invalid;
1368 } else if (!strcasecmp(name, "turnoff_mode")) {
1369 if (strcasecmp(value, "no") && strcasecmp(value, "ctcss_phase_shift") &&
1370 strcasecmp(value, "ctcss_tone_remove") && strcasecmp(value, "ctcss_tail_tone"))
1371 goto invalid;
1372 } else if (!strcasecmp(name, "receive_code") || !strcasecmp(name, "transmit_code")) {
1373 if (!valid_dcs_code(value))
1374 goto invalid;
1375 } else if (!strcasecmp(name, "turnoff_duration_ms")) {
1376 long duration = strtol(value, &end, 0);
1377 if (end == value || *end || duration < 150 || duration > 200)
1378 goto invalid;
1379 } else if (!strcasecmp(name, "signaling_method")) {
1380 if (strcasecmp(value, "carrier") && strcasecmp(value, "ctcss") &&
1381 strcasecmp(value, "dcs"))
1382 goto invalid;
1383 } else if (!strcasecmp(name, "cos_assignment")) {
1384 if (strcasecmp(value, "no") && strcasecmp(value, "usb") &&
1385 strcasecmp(value, "usbinvert") && strcasecmp(value, "dsp") &&
1386 strcasecmp(value, "vox") && strcasecmp(value, "pp") &&
1387 strcasecmp(value, "ppinvert"))
1388 goto invalid;
1389 } else if (!strcasecmp(name, "receive_decoder_gain_db")) {
1390 double number = strtod(value, &end);
1391 if (end == value || *end || !isfinite(number) || number < -60.0 || number > 60.0)
1392 goto invalid;
1393 } else if (!strcasecmp(name, "transmit_peak_dbfs") || !strcasecmp(name, "peak_dbfs")) {
1394 double number = strtod(value, &end);
1395 if (end == value || *end || !isfinite(number) || number < -90.0 || number > 0.0)
1396 goto invalid;
1397 } else if (!strcasecmp(name, "duplex_local_repeat_mode")) {
1398 if (strcasecmp(value, "hardware") && strcasecmp(value, "software"))
1399 goto invalid;
1400 } else if (strcasecmp(name, "hardware_device_identifier") &&
1401 strcasecmp(name, "hardware_serial") && strcasecmp(name, "hardware_user_key")) {
1402 double number = strtod(value, &end);
1403 if (end == value || *end || !isfinite(number))
1404 goto invalid;
1405 if (number < 0.0)
1406 goto invalid;
1407 if (!strcasecmp(name, "duplex_local_repeat_level") && number > 999.0)
1408 goto invalid;
1409 if (!strcasecmp(name, "hardware_clip_led_gpio") && number > 8.0)
1410 goto invalid;
1411 if ((!strcasecmp(name, "hardware_interface_type") ||
1412 !strcasecmp(name, "duplex_radio_mode")) &&
1413 number > 1.0)
1414 goto invalid;
1415 }
1416 if (updated->override_count >= MAX_SECTION_OVERRIDES)
1417 return -1;
1418 entry = &updated->overrides[updated->override_count++];
1419 ast_copy_string(entry->section, !strcasecmp(name, "channel_enabled") ? "general" : section,
1420 sizeof(entry->section));
1421 if (strchr(entry->section, ' '))
1422 *strchr(entry->section, ' ') = '\0';
1423 ast_copy_string(entry->name, name, sizeof(entry->name));
1424 ast_copy_string(entry->value, value, sizeof(entry->value));
1425 return 0;
1426invalid:
1427 ast_log(LOG_ERROR, "RadioPlus [%s]: invalid %s value '%s'\n", section, name, value);
1428 return -1;
1429}

1827static int scoped_section(char *destination, size_t size, const char *kind, const char *name)
1828{
1829 size_t kind_length = strlen(kind);
1830 size_t name_length = strlen(name);
1831
1832 if (kind_length + name_length + 2 > size)
1833 return -1;
1834 memcpy(destination, kind, kind_length);
1835 destination[kind_length] = ' ';
1836 memcpy(destination + kind_length + 1, name, name_length + 1);
1837 return 0;
1838}

1880PROCESSING_PRIVATE int resolve_profile_section(struct ast_config *cfg, const char *radio,
1881 const char *kind, char *section, size_t section_size)
1882{

1907PROCESSING_PRIVATE int load_settings_candidate(struct txagc_audio_snapshot **snapshot)
1908{
1909 struct ast_flags flags = {0};
1910 struct ast_config *cfg;
1911 struct txagc_settings *defaults;
1912 struct txagc_settings *updated;
1913 struct txagc_audio_snapshot *candidate_snapshot;
1914 const char *category = NULL;
1915
1916 if (!snapshot)
1917 return -1;
1918 *snapshot = NULL;
1919
1920 defaults = ast_calloc(1, sizeof(*defaults));
1921 updated = ast_calloc(1, sizeof(*updated));
1922 if (!defaults || !updated) {
1923 ast_free(defaults);
1924 ast_free(updated);
1925 return -1;
1926 }
1927 settings_defaults(defaults);
1928 settings_parse_error = 0;
1929 cfg = ast_config_load2(CONFIG_FILE, "chan_usbradioplus", flags);
1930 if (cfg == CONFIG_STATUS_FILEMISSING || cfg == CONFIG_STATUS_FILEINVALID) {
1931 ast_log(LOG_ERROR, "Unable to load valid %s\n", CONFIG_FILE);
1932 ast_free(defaults);
1933 ast_free(updated);
1934 return -1;
1935 }
1936 if (validate_option_names(cfg))
1937 goto invalid;
1938 {
1939 struct txagc_profile *shared = &defaults->profiles[0];
1940 const char *enabled = ast_variable_retrieve(cfg, "general", "channel_enabled");
1941 if (enabled) {
1942 if (ast_true(enabled))
1943 shared->enabled = 1;
1944 else if (ast_false(enabled))
1945 shared->enabled = 0;
1946 else
1947 goto invalid;
1948 }
1949 if (read_hardware(cfg, "hardware", &shared->hardware) ||
1950 read_section_overrides(shared, cfg, "asterisk", "hardware", "receive",
1951 "transmit", "ctcss", "dcs", "duplex", "diagnostics") ||
1952 read_chain(cfg, "local", &shared->chains[TXAGC_LOCAL]) ||
1953 read_chain(cfg, "link", &shared->chains[TXAGC_LINK]) ||
1954 read_chain(cfg, "voice_telemetry", &shared->chains[TXAGC_VOICE_TELEMETRY]) ||
1955 settings_parse_error)
1956 goto invalid;
1957 }
1958
1959 while ((category = ast_category_browse(cfg, category))) {
1960 struct txagc_profile *profile;
1961 char asterisk_section[MAX_CONFIG_SECTION];
1962 char hardware_section[MAX_CONFIG_SECTION];
1963 char receive_section[MAX_CONFIG_SECTION];
1964 char transmit_section[MAX_CONFIG_SECTION];
1965 char ctcss_section[MAX_CONFIG_SECTION];
1966 char dcs_section[MAX_CONFIG_SECTION];
1967 char duplex_section[MAX_CONFIG_SECTION];
1968 char diagnostics_section[MAX_CONFIG_SECTION];
1969 char local_section[MAX_CONFIG_SECTION];
1970 char link_section[MAX_CONFIG_SECTION];
1971 char voice_section[MAX_CONFIG_SECTION];
1972 const char *enabled;
1973 if (is_profile_section(category) || is_flat_section(category))
1974 continue;
1975 if (updated->profile_count >= MAX_RADIO_PROFILES) {
1976 ast_log(LOG_ERROR, "RadioPlus: too many named radio sections\n");
1977 goto invalid;
1978 }
1979 profile = &updated->profiles[updated->profile_count];
1980 *profile = defaults->profiles[0];
1981 ast_copy_string(profile->name, category, sizeof(profile->name));
1982 snprintf(profile->channel, sizeof(profile->channel), "RadioPlus/%s", category);
1983 if (resolve_profile_section(cfg, category, "asterisk", asterisk_section,
1984 sizeof(asterisk_section)) ||
1985 resolve_profile_section(cfg, category, "hardware", hardware_section,
1986 sizeof(hardware_section)) ||
1987 resolve_profile_section(cfg, category, "receive", receive_section,
1988 sizeof(receive_section)) ||
1989 resolve_profile_section(cfg, category, "transmit", transmit_section,
1990 sizeof(transmit_section)) ||
1991 resolve_profile_section(cfg, category, "ctcss", ctcss_section,
1992 sizeof(ctcss_section)) ||
1993 resolve_profile_section(cfg, category, "dcs", dcs_section,
1994 sizeof(dcs_section)) ||
1995 resolve_profile_section(cfg, category, "duplex", duplex_section,
1996 sizeof(duplex_section)) ||
1997 resolve_profile_section(cfg, category, "diagnostics", diagnostics_section,
1998 sizeof(diagnostics_section)) ||
1999 resolve_profile_section(cfg, category, "local", local_section,
2000 sizeof(local_section)) ||
2001 resolve_profile_section(cfg, category, "link", link_section,
2002 sizeof(link_section)) ||
2003 resolve_profile_section(cfg, category, "voice_telemetry", voice_section,
2004 sizeof(voice_section)))
2005 goto invalid;
2006 enabled = ast_variable_retrieve(cfg, category, "channel_enabled");
2007 if (enabled) {
2008 if (!ast_true(enabled) && !ast_false(enabled))
2009 goto invalid;
2010 profile->enabled = ast_true(enabled);
2011 }
2012 if (read_hardware(cfg, hardware_section, &profile->hardware) ||
2013 read_profile_overrides(profile, cfg, category, asterisk_section,
2014 hardware_section, receive_section, transmit_section,
2015 ctcss_section, dcs_section, duplex_section,
2016 diagnostics_section) ||
2017 read_chain(cfg, local_section, &profile->chains[TXAGC_LOCAL]) ||
2018 read_chain(cfg, link_section, &profile->chains[TXAGC_LINK]) ||
2019 read_chain(cfg, voice_section, &profile->chains[TXAGC_VOICE_TELEMETRY]))
2020 goto invalid;
2021 profile->local_enabled = profile->chains[TXAGC_LOCAL].enabled;
2022 profile->link_enabled = profile->chains[TXAGC_LINK].enabled;
2023 profile->rnnoise_enabled = profile->chains[TXAGC_LOCAL].rnnoise_enabled;
2024 profile->agc = profile->chains[TXAGC_LOCAL].agc;
2025 if (settings_parse_error || validate_profile(profile))
2026 goto invalid;
2027 ++updated->profile_count;
2028 }
2029 if (!updated->profile_count) {
2030 ast_log(LOG_ERROR, "RadioPlus: %s contains no named radio sections\n", CONFIG_FILE);
2031 goto invalid;
2032 }
2033 candidate_snapshot = allocate_audio_snapshot();
2034 if (!candidate_snapshot)
2035 goto invalid;
2036 candidate_snapshot->settings = *updated;
2037 *snapshot = candidate_snapshot;
2038 ast_config_destroy(cfg);
2039 ast_free(defaults);
2040 ast_free(updated);
2041 return 0;
2042
2043invalid:
2044 ast_config_destroy(cfg);
2045 ast_log(LOG_ERROR, "Invalid configuration in %s; keeping existing configuration\n",
2046 CONFIG_FILE);
2047 ast_free(defaults);
2048 ast_free(updated);
2049 return -1;
2050}

2204PROCESSING_PRIVATE int txagc_callback(struct ast_audiohook *audiohook, struct ast_channel *chan,
2205 struct ast_frame *frame,
2206 enum ast_audiohook_direction direction)
2207{

2210 unsigned int sample_rate;
2211 int16_t *pcm;
2212 size_t index;
2213 int result;
2214
2215 if (audiohook->status == AST_AUDIOHOOK_STATUS_DONE || frame->frametype != AST_FRAME_VOICE ||
2216 !frame->data.ptr || frame->samples <= 0) {
2217 return 0;
2218 }
2219 /* audiohook is the first member of its datastore payload. Recovering that
2220 * owner avoids channel datastore lookup from Asterisk's audio callback. */
2221 hook = (struct txagc_hook *)audiohook;
2222 (void)chan;
2223 if (direction != AST_AUDIOHOOK_DIRECTION_READ ||
2224 !atomic_load_explicit(&hook->link_enabled, memory_order_acquire)) {
2225 return 0;
2226 }
2227

2229 if (!sample_rate) {
2230 sample_rate = 8000;
2231 }

2233 * change is intentionally bypassed rather than rebuilding from this audio
2234 * callback. Synchronous processing has no queued output to invalidate. */

2236 return 0;
2237 }

2239 for (index = 0; index < (size_t)frame->samples; ++index)
2240 hook->samples[index] = pcm[index];

2242 if (!filter) {
2243 ++hook->statistics.processing_errors;
2244 link_callback_publish_statistics(hook);
2245 return 0;
2246 }
2247 result = txagc_avfilter_process_prepared(filter, hook->samples, (size_t)frame->samples);
2248 link_callback_copy_filter_statistics(&hook->statistics, filter);
2249 txagc_avfilter_slot_release(&hook->avfilter);
2250 if (result < 0) {
2251 /* Leave the Asterisk frame intact when a prepared graph rejects a block. */
2252 ++hook->statistics.processing_errors;
2253 link_callback_publish_statistics(hook);
2254 return 0;
2255 }
2256 for (index = 0; index < (size_t)frame->samples; ++index) {
2257 double value = hook->samples[index];
2258
2259 if (value > 32767.0)
2260 value = 32767.0;
2261 else if (value < -32768.0)
2262 value = -32768.0;
2263 pcm[index] = (int16_t)lrint(value);
2264 }
2265 link_callback_publish_statistics(hook);

2419PROCESSING_PRIVATE int stage_active_link_hooks(const struct txagc_settings *candidate,
2420 struct link_graph_transaction **result)
2421{
2422 struct ast_channel_iterator *iterator;
2423 struct ast_channel *channel;
2424 struct link_graph_transaction *transaction;
2425 size_t capacity = 0;
2426
2427 if (!candidate || !result)
2428 return -1;
2429 *result = NULL;
2430 iterator = ast_channel_iterator_all_new();
2431 /* Before the scanner starts, no link hook can exist. A startup/prime reload
2432 * may therefore commit an empty link transaction without an iterator. Once
2433 * scanning is active, failure to enumerate could hide a live hook, so reject
2434 * the candidate rather than publishing a partial generation. */
2435 if (!iterator) {
2436 if (scan_thread != AST_PTHREADT_NULL)
2437 return -1;
2438 transaction = ast_calloc(1, sizeof(*transaction));
2439 if (!transaction)
2440 return -1;
2441 *result = transaction;
2442 return 0;
2443 }
2444 transaction = ast_calloc(1, sizeof(*transaction));
2445 if (!transaction) {
2446 ast_channel_iterator_destroy(iterator);
2447 return -1;
2448 }
2449 while ((channel = ast_channel_iterator_next(iterator))) {
2450 const struct ast_datastore *datastore;
2451 struct link_graph_plan *plan;
2452 const struct txagc_profile *profile;
2453
2454 ast_channel_lock(channel);
2455 datastore = ast_channel_datastore_find(channel, &txagc_datastore, NULL);
2456 if (!datastore || !datastore->data) {
2457 ast_channel_unlock(channel);
2458 ast_channel_unref(channel);
2459 continue;
2460 }
2461 if (transaction->count == capacity) {
2462 size_t new_capacity = capacity ? capacity * 2U : 8U;
2463 struct link_graph_plan *expanded =
2464 ast_realloc(transaction->plans, new_capacity * sizeof(*expanded));
2465
2466 if (!expanded) {
2467 ast_channel_unlock(channel);
2468 ast_channel_unref(channel);
2469 ast_channel_iterator_destroy(iterator);
2470 discard_link_graph_transaction(transaction);
2471 return -1;
2472 }
2473 transaction->plans = expanded;
2474 capacity = new_capacity;
2475 }
2476 plan = &transaction->plans[transaction->count];
2477 memset(plan, 0, sizeof(*plan));
2478 plan->channel = channel;
2479 plan->hook = datastore->data;
2480 ast_audiohook_lock(&plan->hook->audiohook);
2481 ast_channel_unlock(channel);
2482 profile = find_profile_const(candidate, plan->hook->profile);
2483 plan->enabled = profile && profile->enabled && profile->chains[TXAGC_LINK].enabled;
2484 /* Count the retained plan before a fallible prepare so the common
2485 * discard path releases this hook and channel on every failure edge. */
2486 ++transaction->count;
2487 if (plan->enabled && txagc_avfilter_slot_candidate_prepare(
2488 &plan->candidate, &profile->chains[TXAGC_LINK].agc,
2489 plan->hook->sample_rate) < 0) {
2490 ast_log(LOG_ERROR, "RadioPlus: keeping previous link graph on %s\n",
2491 plan->hook->channel);
2492 ast_channel_iterator_destroy(iterator);
2493 discard_link_graph_transaction(transaction);
2494 return -1;
2495 }
2496 /* Retain both locks through peer preparation. A later failure therefore
2497 * leaves every graph and admission flag on its prior generation. */
2498 }
2499 ast_channel_iterator_destroy(iterator);
2500 *result = transaction;
2501 return 0;
2502}

2511PROCESSING_PRIVATE void publish_link_graph_transaction(struct link_graph_transaction *transaction)
2512{
2513 size_t index;
2514
2515 if (!transaction)
2516 return;
2517 for (index = 0; index < transaction->count; ++index) {
2518 struct link_graph_plan *plan = &transaction->plans[index];
2519
2520 if (plan->enabled) {
2521 if (txagc_avfilter_slot_publish_candidate(&plan->hook->avfilter,
2522 &plan->candidate) < 0) {
2523 /* A validated candidate is always publishable. This path protects
2524 * against an internal slot invariant failure without touching the
2525 * callback; retain the old admission state in that impossible case.
2526 */
2527 ast_log(LOG_ERROR,
2528 "RadioPlus: unable to publish link graph on %s\n",
2529 plan->hook->channel);
2530 continue;
2531 }
2532 }
2533 /* Synchronous processing has no queued output. Publishing the complete
2534 * graph before changing admission therefore makes each callback use either
2535 * the old graph, the new graph, or a clean bypass. */
2536 atomic_store_explicit(&plan->hook->link_enabled, plan->enabled,
2537 memory_order_release);
2538 }
2539 discard_link_graph_transaction(transaction);
2540}

2544{
