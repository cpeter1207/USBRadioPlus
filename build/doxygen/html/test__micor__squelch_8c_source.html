66{

169{
170 for (unsigned int scale = 1; scale <= 4; ++scale) {

172 uint32_t random = 1;
173 for (unsigned int i = 0; i < 2000 * SAMPLES_PER_MS; ++i) {

176 random = random * 1664525U + 1013904223U;
