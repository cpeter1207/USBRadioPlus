105 /* The hardware backend reserves its configured device using its app_rpt

123 void (*configure)(struct ast_channel *channel))
124{
125 native_format = ast_format_cache_get_slin_by_rate(native_rate);
126 if (ast_format_get_sample_rate(native_format) != native_rate)
127 return -1;
128 hardware = backend;
129 configure_native = configure;
130 advanced.capabilities = ast_format_cap_alloc(AST_FORMAT_CAP_FLAG_DEFAULT);
131 if (!advanced.capabilities)
132 return -1;
133 if (ast_format_cap_append(advanced.capabilities, native_format, 0) ||
134 ast_channel_register(&advanced)) {
135 ao2_cleanup(advanced.capabilities);
136 advanced.capabilities = NULL;
137 return -1;
