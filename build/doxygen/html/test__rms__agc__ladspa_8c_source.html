63static void initialize(struct fixture *fixture, unsigned long rate)
64{
65 static const float defaults[CONTROLS] = {-24, 200, 2, 6, 6, 6, -50, 3, 500, 1};
66 memset(fixture, 0, sizeof(*fixture));
67 fixture->descriptor = ladspa_descriptor(0);
68 fixture->handle = fixture->descriptor->instantiate(fixture->descriptor, rate);
69 assert(fixture->handle);
70 fixture->rate = rate;
71 memcpy(fixture->control, defaults, sizeof(defaults));
72 fixture->descriptor->connect_port(fixture->handle, USBRADIOPLUS_AGC_INPUT,
73 fixture->program);
74 fixture->descriptor->connect_port(fixture->handle, USBRADIOPLUS_AGC_DETECTOR,
75 fixture->detector);
76 fixture->descriptor->connect_port(fixture->handle, USBRADIOPLUS_AGC_OUTPUT,
77 fixture->output);
78 for (unsigned int index = 0; index < CONTROLS; ++index)
79 fixture->descriptor->connect_port(fixture->handle,
80 index + USBRADIOPLUS_AGC_TARGET_DBFS,
81 fixture->control + index);
82}

101static double drive(struct fixture *fixture, unsigned long samples, float detector, float program)
102{
103 double result = 0.0;
104 unsigned int before = allocations;
105 while (samples) {
106 unsigned long count = samples < BLOCK ? samples : BLOCK;
107 for (unsigned long index = 0; index < count; ++index) {
108 fixture->program[index] = program;
109 fixture->detector[index] = index % 2 ? detector : -detector;
110 }
111 fixture->descriptor->run(fixture->handle, count);
112 result = (double)fixture->output[count - 1] / program;
113 samples -= count;
114 }
115 assert(allocations == before);
116 return result;
117}

121{

123 LADSPA_Handle handle;

139 descriptor->connect_port(handle, USBRADIOPLUS_AGC_DETECTOR, input);
140 descriptor->run(handle, 1);
141 descriptor->connect_port(handle, USBRADIOPLUS_AGC_OUTPUT, output);
142 descriptor->connect_port(handle, USBRADIOPLUS_AGC_PORT_COUNT, input);
143 descriptor->run(handle, 0);
144 descriptor->run(handle, 2);

147 descriptor->activate(handle);

156static void test_gain_policy(unsigned long rate)
157{
158 struct fixture fixture;
159 double gain;
160 double frozen;
161 initialize(&fixture, rate);
162 /* An unreachable target still reaches the gain cap with a one-dB deadband. */
163 control(&fixture, USBRADIOPLUS_AGC_DEADBAND_DB, 1);
164 gain = drive(&fixture, rate / 4, 0.01F, 0.1F);
165 assert(fabs(gain - 1.0) < 1e-6);
166 gain = drive(&fixture, rate / 2, 0.01F, 0.1F);
167 assert(20.0 * log10(gain) > 0.4 && 20.0 * log10(gain) < 0.6);
168 gain = drive(&fixture, 4 * rate, 0.01F, 0.1F);
169 assert(fabs(20.0 * log10(gain) - 6.0) < 0.001);
170 /* Inactivity holds the last gain; it does not lift receiver noise between words. */
171 frozen = drive(&fixture, rate, 0.0001F, 0.1F);
172 gain = drive(&fixture, 2 * rate, 0.0001F, 0.1F);
173 assert(gain == frozen);
174 /* Strong activity can reduce gain immediately; it does not wait for the hold timer. */
175 gain = drive(&fixture, rate / 2, 0.5F, 0.1F);
176 assert(20.0 * log10(gain) < 3.2 && 20.0 * log10(gain) > 2.8);
177 gain = drive(&fixture, 3 * rate, 0.5F, 0.1F);
178 assert(fabs(20.0 * log10(gain) + 6.0) < 0.001);
179 /* A tightened configured gain cap is honored, without clipping the program. */
180 control(&fixture, USBRADIOPLUS_AGC_MAX_ATTENUATION_DB, 0);
181 gain = drive(&fixture, 1, 0.5F, 2.0F);
182 assert(gain == 1.0 && fixture.output[0] == 2.0F);
183 fixture.descriptor->cleanup(fixture.handle);
184}

187static void test_activity_and_detector(void)
188{
189 struct fixture fixture;
190 double gain;
191 double previous;
192 initialize(&fixture, 48000);
193 control(&fixture, USBRADIOPLUS_AGC_HOLD_MS, 0);
194 control(&fixture, USBRADIOPLUS_AGC_MAX_BOOST_DB, 30);
195 /* Loud program with quiet detector does not affect gain or invoke a limiter. */
196 assert(drive(&fixture, 48000, 0.001F, 4.0F) == 1.0);
197 previous = drive(&fixture, 24000, 0.006F, 0.1F);
198 /* -51 dBFS remains active after opening at -44 dBFS with three dB hysteresis. */
199 gain = drive(&fixture, 24000, 0.0028F, 0.1F);
200 assert(gain > previous);
201 previous = drive(&fixture, 48000, 0.001F, 0.1F);
202 gain = drive(&fixture, 48000, 0.001F, 0.1F);
203 assert(gain == previous);
204 /* Reacquired loud audio must not request boost from the previous silent average. */
205 gain = drive(&fixture, 4800, 0.5F, 0.1F);
206 assert(gain < previous);
207 fixture.descriptor->activate(fixture.handle);
208 control(&fixture, USBRADIOPLUS_AGC_DEADBAND_DB, 1);
209 gain = drive(&fixture, 3 * 48000, 0.063095734F, 0.1F);
210 assert(fabs(20.0 * log10(gain)) < 0.1);
211 fixture.descriptor->cleanup(fixture.handle);
212}

215static void test_hold_resets(void)
216{
217 struct fixture fixture;
218 double before;
219 double after;
220 initialize(&fixture, 8000);
221 for (unsigned int iteration = 0; iteration < 4; ++iteration) {
222 assert(fabs(drive(&fixture, 2000, 0.01F, 0.1F) - 1.0) < 1e-6);

253 fixture.descriptor->cleanup(fixture.handle);

259 fixture.detector[0] = 0.01F;
260 fixture.descriptor->run(fixture.handle, 1);
261 assert(fixture.output[0] == 0);
262 assert(isfinite(drive(&fixture, 8000, FLT_MAX, 0.1F)));

270static void test_block_independence(void)
271{
272 static const unsigned long rates[] = {8000, 16000, 44100, 48000};
273 for (unsigned int rate_index = 0; rate_index < sizeof(rates) / sizeof(rates[0]);
274 ++rate_index) {
275 struct fixture bulk;
276 struct fixture single;
277 initialize(&bulk, rates[rate_index]);
278 initialize(&single, rates[rate_index]);
279 /* LADSPA allows input/output aliasing unless the plugin advertises otherwise. */
280 single.descriptor->connect_port(single.handle, USBRADIOPLUS_AGC_OUTPUT,
281 single.program);
282 control(&bulk, USBRADIOPLUS_AGC_HOLD_MS, 0);
283 control(&single, USBRADIOPLUS_AGC_HOLD_MS, 0);
284 for (unsigned long block = 0; block < 400; ++block) {
285 for (unsigned long index = 0; index < BLOCK; ++index) {
286 bulk.program[index] =
287 (float)(0.2 * sin((double)(block * BLOCK + index) * 0.07));
288 bulk.detector[index] =
289 bulk.program[index] * (block < 100 ? 0.1F : 1.0F);
290 }
291 bulk.descriptor->run(bulk.handle, BLOCK);
292 for (unsigned long index = 0; index < BLOCK; ++index) {
293 single.program[0] = bulk.program[index];
294 single.detector[0] = bulk.detector[index];
295 single.descriptor->run(single.handle, 1);
296 assert(bulk.output[index] == single.program[0]);
297 }
298 }
299 bulk.descriptor->cleanup(bulk.handle);
300 single.descriptor->cleanup(single.handle);
301 }
302}
