44static int urp_loader_valid(const struct urp_asterisk_loader_descriptor *loader)
45{
46 return loader && loader->struct_size >= sizeof(*loader) &&
47 loader->abi_version == URP_AST_LOADER_ABI_VERSION && loader->capability &&
48 !strcmp(loader->capability, URP_AST_LOADER_CAPABILITY) && loader->load &&
49 loader->reload && loader->unload;
50}

56static int load_module(void)
57{
58 const struct urp_ast_provider_manifest providers = {
59 .struct_size = sizeof(providers),
60 .abi_version = URP_AST_LOADER_ABI_VERSION,
61 .ffmpeg = rptadv_ffmpeg_adapter_descriptor(),
62 .rnnoise = rptadv_rnnoise_adapter_descriptor(),
63 .ring = rpcr2_descriptor(),
64 .radio = rptadv_radio_descriptor(),
65 .samplerate = rptadv_samplerate_adapter_descriptor(),
66 .audio = rptadv_portaudio_alsa_adapter_descriptor(),
67 .gpio = rptadv_gpio_adapter_descriptor(),
68 };
69 int result;
70
71 rust_loader = usbradioplus_asterisk_loader_descriptor();
72 if (!urp_loader_valid(rust_loader)) {
73 ast_log(LOG_ERROR, "USBRadioPlus Rust adapter ABI is unavailable\n");
74 rust_loader = NULL;
75 return AST_MODULE_LOAD_DECLINE;
76 }
77 result = rust_loader->load(&providers, ast_module_info->self);
78 if (result == URP_AST_LOADER_OK)
79 return AST_MODULE_LOAD_SUCCESS;
80 rust_loader = NULL;
81 return result == URP_AST_LOADER_DECLINE ? AST_MODULE_LOAD_DECLINE : AST_MODULE_LOAD_FAILURE;
82}

88static int unload_module(void)
89{
90 int result;
91
92 if (!rust_loader)
93 return 0;
94 result = rust_loader->unload();
95 if (!result)
96 rust_loader = NULL;
97 return result;
98}

104static int reload_module(void)

106 return rust_loader ? rust_loader->reload() : -1;
