46static void process(const struct txagc_config *cfg, double *samples, size_t count,
47 unsigned int rate, size_t block)
48{
49 struct txagc_avfilter graph;
50 txagc_avfilter_init(&graph);
51 for (size_t offset = 0; offset < count; offset += block) {
52 size_t size = count - offset < block ? count - offset : block;
53 assert(txagc_avfilter_process(&graph, cfg, samples + offset, size, rate) == 0);
54 }
55 /* An AGC-only graph must return every sample immediately, including startup. */
56 assert(graph.underrun_samples == 0);
57 txagc_avfilter_destroy(&graph);
58}

80static double rms(const double *samples, size_t count)
81{
82 double sum = 0.0;
83 for (size_t index = 0; index < count; ++index)
84 sum += samples[index] * samples[index];
85 return 10.0 * log10(sum / (double)count / (32768.0 * 32768.0));

90static void test_rates_and_limits(void)
91{
92 static const unsigned int rates[] = {8000, 16000, 48000};
93 for (size_t index = 0; index < sizeof(rates) / sizeof(rates[0]); ++index) {
94 unsigned int rate = rates[index];
95 size_t count = rate * 8U;
96 double *samples = malloc(count * sizeof(*samples));
97 struct txagc_config cfg = configuration();
98 assert(samples);
99 cfg.sidechain_highpass_hz = cfg.sidechain_lowpass_hz = 0.0;
100 for (int loud = 0; loud < 2; ++loud) {
101 double input = loud ? -12.0 : -36.0;
102 double expected = loud ? -18.0 : -30.0;
103 tone(samples, count, rate, 1000.0, input);
104 process(&cfg, samples, count, rate, rate / 50U);
105 double measured = rms(samples + count - rate, rate);
106 printf("AGC %u Hz bounded %+.3f -> %+.3f dBFS RMS\n", rate, input,
107 measured);
108 assert(fabs(measured - expected) < 0.02);
109 }
110 /* A reachable target settles inside its deadband, not at a peak ceiling. */
111 tone(samples, count, rate, 1000.0, -21.0);
112 process(&cfg, samples, count, rate, 37);
113 assert(fabs(rms(samples + count - rate, rate) - cfg.target_dbfs) <= 1.05);
114 free(samples);
115 }
