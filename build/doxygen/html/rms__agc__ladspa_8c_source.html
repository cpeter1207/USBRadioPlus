93static double read_control(const struct rms_agc *state, unsigned int index)
94{
95 const LADSPA_Data *port = state->ports[index + USBRADIOPLUS_AGC_TARGET_DBFS];
96 double value = port ? (double)*port : specs[index].fallback;
97 if (!isfinite(value))
98 value = specs[index].fallback;
99 return bounded(value, specs[index].minimum, specs[index].maximum);
100}

105static void configure(struct rms_agc *state)
106{
107 double values[CONTROL_COUNT];
108 int changed = !state->configured;
109 for (unsigned int index = 0; index < CONTROL_COUNT; ++index)
110 values[index] = read_control(state, index);
111 /* An invalid threshold must not leave the activity detector permanently shut. */
112 if (values[6] >= values[0])
113 values[6] = specs[6].fallback;
114 for (unsigned int index = 0; index < CONTROL_COUNT; ++index)
115 changed |= values[index] != state->controls[index];
116 if (!changed)
117 return;
118 memcpy(state->controls, values, sizeof(values));
119 state->level_coefficient = -expm1(-1000.0 / (state->rate * values[1]));
120 state->opening_power = pow(10.0, values[6] / 10.0);
121 state->closing_power = pow(10.0, (values[6] - values[7]) / 10.0);
122 state->gain_db = bounded(20.0 * log10(state->gain), -values[5], values[4]);
123 state->gain = pow(10.0, state->gain_db / 20.0);
124 state->destination_gain = state->gain;
125 state->ramp_remaining = 0;
126 state->hold_samples = 0.0;
127 state->configured = 1;
128}

133static void update_gain(struct rms_agc *state)
134{
135 const double *control = state->controls;
136 double level = state->level_power / state->level_weight;
137 double wanted;
138 double difference;
139 double change;
140 if (!state->active) {
141 state->hold_samples = 0.0;
142 return;
143 }
144 /* On acquisition, the fast estimate prevents stale silence from commanding boost. */
145 if (state->acquisition_samples > 0.0)
146 level = fmax(level, state->fast_power);
147 wanted = control[0] - 10.0 * log10(fmax(level, DBL_MIN));
148 difference = wanted - state->gain_db;
149 /* Deadband describes distance from the RMS target, not distance from a gain cap. */
150 wanted = bounded(wanted, -control[5], control[4]);
151 change = 0.0;
152 if (difference < -control[9]) {
153 state->hold_samples = 0.0;
154 change = fmax(wanted - state->gain_db,
155 -control[3] * (double)state->elapsed / state->rate);
156 } else if (difference > control[9]) {
157 state->hold_samples = fmin(state->hold_samples + state->active_samples,
158 control[8] * state->rate / 1000.0);
159 if (state->hold_samples >= control[8] * state->rate / 1000.0)
160 change = fmin(wanted - state->gain_db,
161 control[2] * (double)state->elapsed / state->rate);
162 } else {
163 state->hold_samples = 0.0;
164 }
165 state->gain_db += change;
166 state->destination_gain = pow(10.0, state->gain_db / 20.0);
167 /* The following control interval may differ by one sample at unusual rates. */
168 state->ramp_remaining = (unsigned long)ceil((state->rate - (double)state->phase) / 1000.0);
169 state->gain_step = (state->destination_gain - state->gain) / (double)state->ramp_remaining;
170}

199static LADSPA_Handle instantiate(const LADSPA_Descriptor *descriptor, unsigned long sample_rate)
200{
201 struct rms_agc *state;
202 (void)descriptor;
203 if (sample_rate < 1000 || sample_rate > 384000)
204 return NULL;
205 state = calloc(1, sizeof(*state));
206 if (!state)
207 return NULL;
208 state->rate = (double)sample_rate;
209 state->fast_coefficient = -expm1(-100.0 / state->rate);
210 activate(state);
211 return state;
212}

231static void run(LADSPA_Handle handle, unsigned long count)
232{
233 struct rms_agc *state = handle;
234 const float *input = state->ports[USBRADIOPLUS_AGC_INPUT];
235 const float *detector = state->ports[USBRADIOPLUS_AGC_DETECTOR];
236 float *output = state->ports[USBRADIOPLUS_AGC_OUTPUT];
237 if (!input || !detector || !output)
238 return;
239 configure(state);
240 for (unsigned long index = 0; index < count; ++index) {
241 double sample = isfinite(detector[index]) ? (double)detector[index] : 0.0;
242 double power = sample * sample;
243 double result;
244 /* A floor far below every activity threshold prevents denormal CPU stalls in
245 * silence. */
246 state->fast_power =
247 fmax(1e-30, state->fast_power +
248 state->fast_coefficient * (power - state->fast_power));
249 state->level_power =
250 fmax(1e-30, state->level_power + state->level_coefficient *
251 (power - state->level_power));
252 state->level_weight += state->level_coefficient * (1.0 - state->level_weight);
253 if (state->active) {
254 if (state->fast_power < state->closing_power) {
255 state->active = 0;
256 state->hold_samples = 0.0;
257 state->active_samples = 0.0;
258 state->ramp_remaining = 0;
259 state->gain_db = 20.0 * log10(state->gain);
260 }
261 } else if (state->fast_power >= state->opening_power) {
262 state->active = 1;
263 /* Normalize the new activity's EMA instead of treating earlier silence as
264 * speech. */
265 state->level_power = state->level_coefficient * power;
266 state->level_weight = state->level_coefficient;
267 state->acquisition_samples = state->controls[1] * state->rate / 1000.0;
268 }
269 if (state->active)
270 state->active_samples += 1.0;
271 state->acquisition_samples = fmax(0.0, state->acquisition_samples - 1.0);
272 state->phase += 1000;
273 state->elapsed++;
274 if ((double)state->phase >= state->rate) {
275 state->phase -= (unsigned long)state->rate;
276 update_gain(state);
277 state->elapsed = 0;
278 state->active_samples = 0.0;
279 }
280 if (state->ramp_remaining) {
281 state->gain += state->gain_step;
282 if (!--state->ramp_remaining)
283 state->gain = state->destination_gain;
284 }
285 /* Corrupt input is silenced, not allowed to poison detector or gain state. */
286 result = (double)input[index] * state->gain;
287 output[index] = isfinite(result) && fabs(result) <= FLT_MAX ? (float)result : 0.0F;
288 }
289}
