364static void read_hardware_snapshot(const struct chan_usbradio_pvt *channel,
365 enum radio_tx_mix *txmixa, enum radio_tx_mix *txmixb)
366{
367 unsigned int attempt;
368
369 for (attempt = 0; attempt < 2U; ++attempt) {
370 unsigned int before;
371 unsigned int after;
372
373 before = atomic_load_explicit(&channel->plus_hardware_generation,
374 memory_order_acquire);
375 if (before & 1U)
376 continue;
377 *txmixa = (enum radio_tx_mix)atomic_load_explicit(&channel->plus_applied_txmixa,
378 memory_order_relaxed);
379 *txmixb = (enum radio_tx_mix)atomic_load_explicit(&channel->plus_applied_txmixb,
380 memory_order_relaxed);
381 after = atomic_load_explicit(&channel->plus_hardware_generation,
382 memory_order_acquire);
383#ifdef URP_PROCESSING_TESTING
384 after += atomic_exchange_explicit(&native_renderer_test_hardware_snapshot_delta, 0U,
385 memory_order_acq_rel);
386#endif
387 if (before == after)
388 return;
389 }
390 /* Never wait for a control-plane writer in an audio callback. */
391 *txmixa = (enum radio_tx_mix)atomic_load_explicit(&channel->plus_applied_txmixa,
392 memory_order_relaxed);
393 *txmixb = (enum radio_tx_mix)atomic_load_explicit(&channel->plus_applied_txmixb,
394 memory_order_relaxed);

547 input->tx_ctcss_enabled && !input->tx_ctcss_off);

554 renderer->dcs_inverted != input->dcs_transmit_inverted) {
555 urp_dcs_configure(&renderer->dcs, -1, 0, input->dcs_transmit_code,

558 renderer->dcs_inverted = input->dcs_transmit_inverted;
559 }
560 dcs_normal_active =
561 input->dcs_enabled && input->tx_ptt_out && input->tx_state == CHAN_TXSTATE_ACTIVE;
562 urp_dcs_generate(&renderer->dcs, renderer->dcs_program, URP_NATIVE_SAMPLES, URP_RATE_NATIVE,

564 input->dcs_turnoff_active);
565 if (txagc_avfilter_process_prepared(input->dcs_turnoff_active ? &graphs->dcs_turnoff
566 : &graphs->dcs,
567 renderer->dcs_program, URP_NATIVE_SAMPLES) < 0)
568 memset(renderer->dcs_program, 0, sizeof(renderer->dcs_program));
569 if (!dcs_normal_active && !input->dcs_turnoff_active)
570 memset(renderer->dcs_program, 0, sizeof(renderer->dcs_program));

583static void native_renderer_render_transmit(struct usbradioplus_native_renderer *renderer,
584 const struct native_renderer_input *input,
585 struct usbradioplus_native_graph_set *graphs,
586 short *stereo)
587{
588 struct chan_usbradio_pvt *channel = renderer->channel;
589 double ctcss_peak_a = input->tx_ctcss_peak;
590 double ctcss_peak_b = ctcss_peak_a;
591 enum radio_tx_mix txmixa;
592 enum radio_tx_mix txmixb;
593 size_t index;
594 size_t used = 0U;
595 size_t made = 0U;
596
597 memset(stereo, 0, URP_NATIVE_STEREO_SAMPLES * sizeof(*stereo));
598 native_renderer_generate_signaling(renderer, input, graphs);
599 memset(renderer->link_native, 0, sizeof(renderer->link_native));
600 if (graphs->legacy_interface &&
601 atomic_load_explicit(&channel->echoing, memory_order_acquire)) {
602 int have_frame = 0;
603
604 if (urp_sample_queue_samples(&channel->echo_queue) >= graphs->app_rpt_samples) {
605 for (index = 0; index < graphs->app_rpt_samples; ++index)
606 (void)urp_sample_queue_pop_sample(&channel->echo_queue,
607 &renderer->link_app[index]);
608 have_frame = 1;
609 }
610 if (!have_frame) {
611 if (atomic_load_explicit(&channel->txkeyed, memory_order_acquire))
612 renderer->statistics.link_queue_underflows++;
613 urp_src_reset(renderer->echo_up);
614 } else if (graphs->app_rpt_rate == URP_RATE_NATIVE) {
615 memcpy(renderer->link_native, renderer->link_app,
616 sizeof(renderer->link_native));
617 } else if (urp_rate_convert_prepared(renderer->echo_up, renderer->link_app,
618 graphs->app_rpt_samples, graphs->app_rpt_rate,
619 renderer->link_native, URP_NATIVE_SAMPLES,
620 URP_RATE_NATIVE, &used, &made) ||
621 used != graphs->app_rpt_samples) {
622 renderer->statistics.src_errors++;
623 memset(renderer->link_native, 0, sizeof(renderer->link_native));
624 } else if (made < URP_NATIVE_SAMPLES) {
625 memset(renderer->link_native + made, 0,
626 (URP_NATIVE_SAMPLES - made) * sizeof(*renderer->link_native));
627 }
628 } else if (!read_native_program(channel, graphs, renderer->link_native) &&
629 atomic_load_explicit(&channel->txkeyed, memory_order_acquire)) {
630 renderer->statistics.link_queue_underflows++;
631 }
632 for (index = 0; index < URP_NATIVE_SAMPLES; ++index)
633 renderer->program[index] = renderer->link_native[index];
634 memset(renderer->local_program, 0, sizeof(renderer->local_program));
635 if (graphs->legacy_interface && renderer->parrot.playing) {
636 urp_parrot_play(&renderer->parrot, renderer->local_program, URP_NATIVE_SAMPLES);
637 renderer->statistics.parrot_playback_frames++;
638 if (!renderer->parrot.playing)
639 atomic_store_explicit(&channel->echoing, 0, memory_order_release);
640 } else if (graphs->legacy_interface && input->rxkeyed && graphs->software_repeat_enabled) {
641 urp_native_repeat_prepare(renderer->local_program, renderer->local_native,
642 URP_NATIVE_SAMPLES, 1.0,

988/* Process one physical callback block and write direct app/DAC outputs.
989 *
990 * Receive processing always runs so app_rpt sees every ADC frame. Program PCM
991 * and TX-only state advance only when the DAC will accept the matching frame;
992 * a full device therefore cannot consume the program ring and make a dropout.
993 */
994void usbradioplus_native_tick(struct chan_usbradio_pvt *channel, int transmit_ready)
995{
996 struct usbradioplus_native_renderer *renderer;
997 struct usbradioplus_native_graph_set *graphs;
998 struct native_renderer_input input;
999 short *app_pcm;
1000 const short *adc_pcm;
1001
1002 if (!channel)
1003 return;
1004 renderer = channel->plus_native_renderer;
1005 if (!renderer) {
1006 native_renderer_silence(channel);
1007 usbradioplus_publish_hardware_ptt(channel,
1008 channel->radio ? channel->radio->txPttOut : 0);
1009 return;
1010 }
1011 native_renderer_silence(channel);
1012 graphs = usbradioplus_native_graphs_acquire(channel);
1013 if (!graphs) {
1014 usbradioplus_publish_hardware_ptt(channel,
1015 channel->radio ? channel->radio->txPttOut : 0);
1016 return;
1017 }
1018 native_renderer_snapshot(&input, channel);
1019 native_renderer_apply_requests(renderer, graphs);
1020 if (renderer->app_rpt_rate != graphs->app_rpt_rate) {
1021 /* Converter history is callback-owned; reload never touches it. */
1022 urp_src_reset(renderer->echo_up);
1023 urp_src_reset(renderer->down);
1024 renderer->app_rpt_rate = graphs->app_rpt_rate;
1025 }
1026 adc_pcm = (const short *)(channel->usbradio_read_buf + AST_FRIENDLY_OFFSET);
1027 app_pcm = (short *)(channel->usbradio_read_buf_8k + AST_FRIENDLY_OFFSET);
1028 native_renderer_render_receive(renderer, &input, graphs, adc_pcm, app_pcm);
1029 if (transmit_ready)
1030 native_renderer_render_transmit(renderer, &input, graphs,
1031 (short *)channel->usbradio_write_buf);
1032 native_renderer_finish_block(renderer, graphs);
1033 usbradioplus_native_graphs_release(channel);
1034 /* PTT remains a signaling-engine decision, independent of audio availability. */
1035 usbradioplus_publish_hardware_ptt(channel, channel->radio ? channel->radio->txPttOut : 0);
1036}
