71RNNOISE_PRIVATE int configure_rate(struct txagc_rnnoise *state, unsigned int sample_rate)
72{
73 int error;
74
75 if (state->active && state->input_rate == sample_rate && state->denoise &&
76 state->upsampler && state->downsampler) {
77 return 0;
78 }
79 txagc_rnnoise_destroy(state);
80 state->denoise = rnnoise_create(NULL);
81 state->upsampler = src_new(SRC_SINC_FASTEST, 1, &error);
82 if (!state->upsampler) {
83 state->errors++;
84 return -1;
85 }
86 state->downsampler = src_new(SRC_SINC_FASTEST, 1, &error);
87 if (!state->denoise || !state->downsampler) {
88 state->errors++;
89 return -1;
90 }
91 state->input_rate = sample_rate;
92 state->active = 1;
93 return 0;
94}

137int txagc_rnnoise_process_double(struct txagc_rnnoise *state, double *samples, size_t count,
138 unsigned int sample_rate)
139{
140 float input[2048];
141 float converted[4096];
142 float clean[TXAGC_RNNOISE_FRAME];
143 float down[1024];
144 SRC_DATA data;
145 size_t i;
146
147 if (!sample_rate || count > sizeof(input) / sizeof(input[0]) ||
148 configure_rate(state, sample_rate)) {
149 return -1;
150 }
151 state->active = 1;
152 for (i = 0; i < count; ++i) {
153 input[i] = samples[i];
154 }
155 memset(&data, 0, sizeof(data));
156 data.data_in = input;
157 data.input_frames = count;
158 data.data_out = converted;
159 data.output_frames = sizeof(converted) / sizeof(converted[0]);
160 data.src_ratio = (double)TXAGC_RNNOISE_RATE / sample_rate;
161 if (src_process(state->upsampler, &data) || data.input_frames_used != (long)count ||
162 append(state->up_fifo, &state->up_count, converted, data.output_frames_gen)) {
163 state->errors++;
164 reset_stream(state);
165 return -1;
166 }
167
168 while (state->up_count >= TXAGC_RNNOISE_FRAME) {
169 long generated;
170 state->vad_probability =
171 rnnoise_process_frame(state->denoise, clean, state->up_fifo);
172 state->vad_sum += state->vad_probability;
173 state->rnnoise_frames++;
174 consume(state->up_fifo, &state->up_count, TXAGC_RNNOISE_FRAME);
175 if (!state->primed) {
176 state->primed = 1;
177 continue;
178 }
179
180 memset(&data, 0, sizeof(data));
181 data.data_in = clean;
182 data.input_frames = TXAGC_RNNOISE_FRAME;
183 data.data_out = down;
184 data.output_frames = sizeof(down) / sizeof(down[0]);
185 data.src_ratio = (double)sample_rate / TXAGC_RNNOISE_RATE;
186 if (src_process(state->downsampler, &data)) {
187 state->errors++;
188 reset_stream(state);
189 return -1;
190 }
191 generated = data.output_frames_gen;
192 if (append(state->down_fifo, &state->down_count, down, generated)) {
193 state->errors++;
194 reset_stream(state);
195 return -1;
196 }
197 }
198
199 {
200 size_t available = state->down_count;
201 if (available > count) {
202 available = count;
203 }
204 for (i = 0; i < available; ++i) {
205 samples[i] = state->down_fifo[i];
206 state->output_samples++;
207 }
208 consume(state->down_fifo, &state->down_count, available);
209 for (; i < count; ++i) {
210 samples[i] = 0;
211 state->startup_samples++;
212 }
213 }
214 return 0;
215}

217int txagc_rnnoise_process(struct txagc_rnnoise *state, int16_t *samples, size_t count,
218 unsigned int sample_rate)
219{
220 double work[2048];
221 size_t i;
222 int result;
223
224 if (!samples || count > sizeof(work) / sizeof(work[0])) {
225 return -1;
226 }
227 for (i = 0; i < count; ++i) {
228 work[i] = samples[i];
229 }
230 result = txagc_rnnoise_process_double(state, work, count, sample_rate);
231 if (result) {
232 return result;
233 }
234 for (i = 0; i < count; ++i)
235 samples[i] = pcm_from_double(work[i]);
236 return 0;
237}
