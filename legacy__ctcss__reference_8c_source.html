48static int32_t fir_step(i16 *state, const i16 *coef, int taps, int32_t gain, int32_t sample,
49 int32_t output_gain)
50{
51 int64_t sum = 0;
52 int i;
53 for (i = taps - 1; i > 0; --i)
54 state[i] = state[i - 1];
55 state[0] = (i16)sample;
56 for (i = 0; i < taps; ++i)
57 sum += (int32_t)coef[i] * state[i];
58 return (int32_t)(((sum / gain) * output_gain) / 256);
59}

77static struct level_result characterize(float frequency, int wide, int lsd_output_gain,
78 int output_gain, int verbose)
79{
80 const i16 *lsd_coef = wide ? coef_fir_lpf_250_9_66 : coef_fir_lpf_215_9_88;
81 const int lsd_taps = wide ? taps_fir_lpf_250_9_66 : taps_fir_lpf_215_9_88;
82 const int32_t lsd_gain = wide ? gain_fir_lpf_250_9_66 : gain_fir_lpf_215_9_88;
83 i16 lsd_state[128] = {0}, out_state[128] = {0};
84 int32_t phase = 0, peak = 0, trough = 0;
85 const int32_t freq10 = (int32_t)(frequency * 10.0f);
86 const int32_t step = (256 * freq10 * 128) / 8000 / 10;
87 int n, k;
88
89 for (n = 0; n < CHARACTERIZE_SAMPLES; ++n) {
90 int32_t generated = ((int32_t)legacy_sine(phase / 128) * 128) / 256;
91 int32_t filtered = fir_step(lsd_state, lsd_coef, lsd_taps, lsd_gain, generated,
92 lsd_output_gain);
93 phase = (phase + step) % (256 * 128);
94 for (k = 0; k < 6; ++k) {
95 int32_t output = fir_step(out_state, fir_txlpf[0].coefs, fir_txlpf[0].taps,
96 fir_txlpf[0].gain, filtered, output_gain);
97 if (n >= CHARACTERIZE_SETTLING_SAMPLES) {
98 if (output > peak)
99 peak = output;
100 if (output < trough)
101 trough = output;
102 }
103 }
104 }
105 if (verbose)
106 printf("%5.1f %s peak=%d trough=%d p2p=%d\n", frequency, wide ? "250" : "215", peak,
107 trough, peak - trough);
108 return (struct level_result){peak, trough};
109}

115static int legacy_mult_calc(int value)
116{
117 int pot = (value / 4) * 4 + 2;
