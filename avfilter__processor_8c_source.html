99AVFILTER_PRIVATE int graph_append(char *graph, size_t size, const char *format, ...)
100{
101 va_list ap;
102 size_t used = strlen(graph);
103 int result;
104
105 if (used >= size) {
106 return AVERROR(ENOSPC);
107 }
108 va_start(ap, format);
109 result = vsnprintf(graph + used, size - used, format, ap);
110 va_end(ap);
111 if (result < 0 || (size_t)result >= size - used) {
112 return AVERROR(ENOSPC);
113 }
114 return 0;
115}

185 enum cleanup_meter meter)
186{
187 double peak;
188 double rms;
189 double *current;
190 double *maximum;
191
192 if (astats_value(frame, "lavfi.astats.Overall.RMS_level", &rms)) {
193 return;
194 }
195 if (meter == CLEANUP_PRE_FULL) {
196 if (astats_value(frame, "lavfi.astats.Overall.Peak_level", &peak)) {
197 return;
198 }
199 state->cleanup_pre_peak_dbfs = peak;
200 state->cleanup_pre_rms_dbfs = rms;
201 if (peak > state->cleanup_pre_max_peak_dbfs) {
202 state->cleanup_pre_max_peak_dbfs = peak;
203 }
204 if (rms > state->cleanup_pre_max_rms_dbfs) {
205 state->cleanup_pre_max_rms_dbfs = rms;
206 }
207 return;
208 }
209 switch (meter) {
210 case CLEANUP_PRE_5_8:
211 current = &state->cleanup_pre_5_8_rms_dbfs;
212 maximum = &state->cleanup_pre_5_8_max_rms_dbfs;
213 break;
214 case CLEANUP_PRE_8_PLUS:
215 current = &state->cleanup_pre_8_plus_rms_dbfs;
216 maximum = &state->cleanup_pre_8_plus_max_rms_dbfs;
217 break;
218 case CLEANUP_POST_5_8:
219 current = &state->cleanup_post_5_8_rms_dbfs;
220 maximum = &state->cleanup_post_5_8_max_rms_dbfs;
221 break;
222 default:
223 current = &state->cleanup_post_8_plus_rms_dbfs;
224 maximum = &state->cleanup_post_8_plus_max_rms_dbfs;
225 break;
226 }
227 *current = rms;
228 if (rms > *maximum) {
229 *maximum = rms;
230 }

299AVFILTER_PRIVATE int add_emphasis(char *graph, size_t size, const char *input, const char *output,
300 int production, double corner_hz, double reference_hz,
301 unsigned int sample_rate)
302{
303 double pole = exp(-2.0 * M_PI * corner_hz / sample_rate);
304 double omega = 2.0 * M_PI * reference_hz / sample_rate;
305 double inverse_at_reference =
306 sqrt(1.0 + pole * pole - 2.0 * pole * cos(omega)) / (1.0 - pole);
307 if (production) {
308 double scale = 1.0 / inverse_at_reference;
309 return graph_append(graph, size,
310 "[%s]biquad=b0=%.17g:b1=%.17g:b2=0:a0=1:a1=0:a2=0:"
311 "precision=f64[%s];",
312 input, scale / (1.0 - pole), -pole * scale / (1.0 - pole),
313 output);
314 }
315 return graph_append(graph, size,
316 "[%s]biquad=b0=%.17g:b1=0:b2=0:a0=1:a1=%.17g:a2=0:"
317 "precision=f64[%s];",
318 input, inverse_at_reference * (1.0 - pole), -pole, output);
319}

343static int add_three_band_dynamics(char *graph, size_t size, const char *input, const char *output,
344 const char *prefix, double low_crossover, double high_crossover,
345 const struct dynamics_band bands[3], const char *detection)
346{
347 static const char *const names[] = {"lo", "mid", "hi"};
348 int result;
349
350 /* Complementary crossover outputs retain unity summed response. Each band's
351 * detector follows its own program audio; no speech sidechain trims a band. */
352 result = graph_append(graph, size,
353 "[%s]acrossover=split=%.9g %.9g:order=4th[%slo][%smid][%shi];", input,
354 low_crossover, high_crossover, prefix, prefix, prefix);
355 if (result < 0)
356 return result;
357 for (unsigned int index = 0; index < 3; ++index) {
358 const struct dynamics_band *band = &bands[index];
359 char makeup[80] = "";
360 /* FFmpeg's compressor makeup cannot attenuate. A post-compression
361 * gain supports the configured negative gains without changing detection. */
362 if (band->makeup_gain_db != 0.0)
363 snprintf(makeup, sizeof(makeup), ",volume=%.12g:precision=double",
364 db_to_linear(band->makeup_gain_db));
365 result = graph_append(graph, size,
366 "[%s%s]acompressor=mode=downward:threshold=%.12g:ratio=%.9g:"
367 "knee=%.12g:attack=%.9g:release=%.9g:detection=%s%s[%s%sc];",
368 prefix, names[index], db_to_linear(band->threshold_dbfs),
369 clamp(band->ratio, 1.0, 20.0), db_to_linear(band->knee_db),
370 band->attack_ms, band->release_ms, detection, makeup, prefix,
371 names[index]);
372 if (result < 0)
373 return result;
374 }
375 return graph_append(graph, size, "[%sloc][%smidc][%shic]amix=inputs=3:normalize=0[%s];",
376 prefix, prefix, prefix, output);
377}

389AVFILTER_PRIVATE int add_dynamic_stage(char *graph, size_t size, const struct txagc_config *cfg,
390 enum txagc_stage kind, const char *current, const char *next,
391 unsigned int serial)
392{
393 char options[1024];
394 char prefix[32];
395
396 snprintf(prefix, sizeof(prefix), "d%u", serial);
397 switch (kind) {
398 case TXAGC_STAGE_DEESSER: {
399 double octave_ratio;
400 double q;
401 if (!cfg->deesser_enabled)
402 return 0;
403 octave_ratio = pow(2.0, cfg->deesser_width_octaves);
404 q = sqrt(octave_ratio) / (octave_ratio - 1.0);
405#if LIBAVFILTER_VERSION_MAJOR >= 9
406 return graph_append(
407 graph, size,
408 "[%s]adynamicequalizer=threshold=%.12g:dfrequency=%.9g:"
409 "dqfactor=%.12g:tfrequency=%.9g:tqfactor=%.12g:"
410 "attack=%.9g:release=%.9g:ratio=%.9g:range=%.12g:"
411 "mode=cutabove:dftype=bandpass:tftype=bell:precision=double[%s];",
412 current, db_to_linear(cfg->deesser_threshold_dbfs),
413 cfg->deesser_frequency_hz, q, cfg->deesser_frequency_hz, q,
414 cfg->deesser_attack_ms, cfg->deesser_release_ms, cfg->deesser_ratio,
415 cfg->deesser_max_reduction_db / 2.0, next);
416#else
417 double ratio = cfg->deesser_ratio;
418 double span = -cfg->deesser_threshold_dbfs;
419 double detector_reduction = cfg->deesser_max_reduction_db / 2.0;
420 /* FFmpeg 5.1 does not apply range to cuts. Cap its ratio so a
421 * full-scale detector input cannot exceed the requested reduction.
422 * Its bell response applies the detector gain twice at band center. */
423 if (span > detector_reduction) {
424 double limited_ratio = span / (span - detector_reduction);
425 if (ratio > limited_ratio)
426 ratio = limited_ratio;
427 }
428 return graph_append(graph, size,
429 "[%s]adynamicequalizer=threshold=%.12g:dfrequency=%.9g:"
430 "dqfactor=%.12g:tfrequency=%.9g:tqfactor=%.12g:"
431 "attack=%.9g:release=%.9g:ratio=%.9g:range=%.12g:"
432 "mode=cut:tftype=bell[%s];",
433 current, db_to_linear(cfg->deesser_threshold_dbfs),
434 cfg->deesser_frequency_hz, q, cfg->deesser_frequency_hz, q,
435 cfg->deesser_attack_ms, cfg->deesser_release_ms, ratio, 0.0,
436 next);
437#endif
438 }
439 case TXAGC_STAGE_EQUALIZER:
440 if (!cfg->equalizer_enabled)
441 return 0;
442 return graph_append(graph, size,
443 "[%s]bass=g=%.9g:f=%.9g:t=s:w=%.9g:r=f64,"
444 "equalizer=f=%.9g:t=o:w=%.9g:g=%.9g:r=f64,"
445 "treble=g=%.9g:f=%.9g:t=s:w=%.9g:r=f64[%s];",
446 current, cfg->equalizer_low_gain_db,
447 cfg->equalizer_low_frequency_hz, cfg->equalizer_low_slope,
448 cfg->equalizer_mid_frequency_hz,
449 cfg->equalizer_mid_width_octaves, cfg->equalizer_mid_gain_db,
450 cfg->equalizer_high_gain_db, cfg->equalizer_high_frequency_hz,
451 cfg->equalizer_high_slope, next);
452 case TXAGC_STAGE_AGC:
453 if (!cfg->agc_enabled)
454 return 0;
455 /* Pack program and detector into explicit channels for the LADSPA
456 * two-input/one-output effect. Only its detector sees the filters.
457 * FFmpeg 5 needs the effect's single output labeled explicitly as mono. */
458 snprintf(options, sizeof(options),
459 "inputs=2:channel_layout=stereo:map=0.0-FL|1.0-FR,"
460 "ladspa=file='%s':plugin=usbradioplus_agc:controls="
461 "c0=%.12g|c1=%.12g|c2=%.12g|c3=%.12g|c4=%.12g|c5=%.12g|"
462 "c6=%.12g|c7=%.12g|c8=%.12g|c9=%.12g,"
463 "aformat=channel_layouts=mono",
464 URP_AGC_PLUGIN_PATH, cfg->target_dbfs, cfg->agc_rms_averaging_ms,
465 cfg->agc_gain_increase_db_per_second, cfg->agc_gain_decrease_db_per_second,
466 cfg->max_gain_db, cfg->max_attenuation_db,
467 cfg->agc_activity_threshold_dbfs, cfg->agc_activity_hysteresis_db,
468 cfg->agc_hold_ms, cfg->agc_deadband_db);
469 return add_sidechain_stage(graph, size, current, next, prefix, "join",
470 cfg->sidechain_highpass_hz, cfg->sidechain_lowpass_hz,
471 options);
472 case TXAGC_STAGE_EXPANDER:
473 if (!cfg->expander_enabled)
474 return 0;
475 snprintf(options, sizeof(options),
476 "threshold=%.12g:ratio=%.9g:range=%.12g:attack=%.9g:"
477 "release=%.9g:knee=2.828427:detection=rms",
478 db_to_linear(cfg->expander_threshold_dbfs),
479 clamp(cfg->expander_ratio, 1.0, 20.0),
480 db_to_linear(-fabs(cfg->expander_max_attenuation_db)),
481 cfg->expander_attack_ms, cfg->expander_release_ms);
482 return add_sidechain_stage(graph, size, current, next, prefix, "sidechaingate",
483 cfg->expander_sidechain_highpass_hz,
484 cfg->expander_sidechain_lowpass_hz, options);
485 case TXAGC_STAGE_COMPRESSOR: {
486 const struct dynamics_band bands[] = {
487 {cfg->compressor_low_threshold_dbfs, cfg->compressor_low_ratio,
488 cfg->compressor_low_makeup_gain_db, cfg->compressor_low_knee_db,
489 cfg->compressor_low_attack_ms, cfg->compressor_low_release_ms},
490 {cfg->compressor_mid_threshold_dbfs, cfg->compressor_mid_ratio,
491 cfg->compressor_mid_makeup_gain_db, cfg->compressor_mid_knee_db,
492 cfg->compressor_mid_attack_ms, cfg->compressor_mid_release_ms},
493 {cfg->compressor_high_threshold_dbfs, cfg->compressor_high_ratio,
494 cfg->compressor_high_makeup_gain_db, cfg->compressor_high_knee_db,
495 cfg->compressor_high_attack_ms, cfg->compressor_high_release_ms},
496 };
497 if (!cfg->compressor_enabled)
498 return 0;
499 if (cfg->compressor_bands == 3)
500 return add_three_band_dynamics(graph, size, current, next, prefix,
501 cfg->compressor_low_crossover_hz,
502 cfg->compressor_high_crossover_hz, bands,
503 "rms");
504 snprintf(options, sizeof(options),
505 "mode=downward:threshold=%.12g:ratio=%.9g:attack=%.9g:"
506 "release=%.9g:knee=2.828427:detection=rms,volume=%.12g:precision=double",
507 db_to_linear(cfg->compressor_threshold_dbfs),
508 clamp(cfg->compressor_ratio, 1.0, 20.0), cfg->compressor_attack_ms,
509 cfg->compressor_release_ms, db_to_linear(cfg->compressor_makeup_gain_db));
510 return add_sidechain_stage(graph, size, current, next, prefix, "sidechaincompress",
511 cfg->compressor_sidechain_highpass_hz,
512 cfg->compressor_sidechain_lowpass_hz, options);
513 }
514 case TXAGC_STAGE_LIMITER: {
515 const struct dynamics_band bands[] = {
516 {cfg->low_limiter_threshold_dbfs, cfg->low_limiter_ratio, 0.0,
517 cfg->low_limiter_knee_db, cfg->low_limiter_attack_ms,
518 cfg->low_limiter_release_ms},
519 {cfg->mid_limiter_threshold_dbfs, cfg->mid_limiter_ratio, 0.0,
520 cfg->mid_limiter_knee_db, cfg->mid_limiter_attack_ms,
521 cfg->mid_limiter_release_ms},
522 {cfg->high_limiter_threshold_dbfs, cfg->high_limiter_ratio, 0.0,
523 cfg->high_limiter_knee_db, cfg->high_limiter_attack_ms,
524 cfg->high_limiter_release_ms},
525 };
526 if (!cfg->limiter_enabled)
527 return 0;
528 if (cfg->limiter_bands == 3)
529 return add_three_band_dynamics(
530 graph, size, current, next, prefix, cfg->limiter_low_crossover_hz,
531 cfg->limiter_high_crossover_hz, bands, "peak");
532 return graph_append(graph, size,
533 "[%s]acompressor=mode=downward:threshold=%.12g:ratio=%.9g:"
534 "knee=%.12g:attack=%.9g:release=%.9g:detection=peak[%s];",
535 current, db_to_linear(cfg->limiter_threshold_dbfs),
536 clamp(cfg->limiter_ratio, 1.0, 20.0),
537 db_to_linear(cfg->limiter_knee_db), cfg->limiter_attack_ms,
538 cfg->limiter_release_ms, next);
539 }
540 }
541 return AVERROR(EINVAL);
542}

551AVFILTER_PRIVATE int build_description(char *graph, size_t size, const struct txagc_config *cfg,
552 unsigned int sample_rate)
553{
554 char current[NAME_SIZE] = "s0";
555 const char *graph_input = "in";
556 char next[NAME_SIZE];
557 unsigned int stage = 1;
558 unsigned int order_index;
559
560 graph[0] = '\0';
561 /* A link can run at a lower sample rate than the native radio chain. Reject
562 * impossible band definitions instead of silently moving their crossovers. */
563 if (cfg->compressor_enabled && cfg->compressor_bands == 3 &&
564 cfg->compressor_high_crossover_hz >= sample_rate * 0.5) {
565 av_log(NULL, AV_LOG_ERROR,
566 "USBRadioPlus: compressor crossovers %.9g/%.9g Hz must be below "
567 "Nyquist %.9g Hz at %u Hz sample rate\n",
568 cfg->compressor_low_crossover_hz, cfg->compressor_high_crossover_hz,
569 sample_rate * 0.5, sample_rate);
570 return AVERROR(EINVAL);
571 }
572 if (cfg->limiter_enabled && cfg->limiter_bands == 3 &&
573 cfg->limiter_high_crossover_hz >= sample_rate * 0.5) {
574 av_log(NULL, AV_LOG_ERROR,
575 "USBRadioPlus: limiter crossovers %.9g/%.9g Hz must be below "
576 "Nyquist %.9g Hz at %u Hz sample rate\n",
577 cfg->limiter_low_crossover_hz, cfg->limiter_high_crossover_hz,
578 sample_rate * 0.5, sample_rate);
579 return AVERROR(EINVAL);
580 }
581 if (cfg->deemphasis_enabled) {
582 if (add_emphasis(graph, size, "in", "deemphasized", 0, cfg->emphasis_corner_hz,
583 cfg->emphasis_reference_hz, sample_rate) < 0) {
584 return AVERROR(ENOSPC);
585 }
586 graph_input = "deemphasized";
587 }
588 if (cfg->receive_bandpass_enabled) {
589 if (add_brickwall_bandpass(graph, size, graph_input, "rxbandpass", "rxbp",
590 cfg->receive_bandpass_highpass_hz,
591 cfg->receive_bandpass_lowpass_hz) < 0) {
592 return AVERROR(ENOSPC);
593 }
594 graph_input = "rxbandpass";
595 }
596 if (cfg->ctcss_filter_mode == TXAGC_CTCSS_FILTER_NOTCH) {
597 const char *cursor = cfg->ctcss_notch_frequencies;
598 char input_name[NAME_SIZE];
599 char output_name[NAME_SIZE];
600 unsigned int notch = 0;
601
602 snprintf(input_name, sizeof(input_name), "%s", graph_input);
603 while (*cursor) {
604 char *end;
605 double frequency = strtod(cursor, &end);
606 unsigned int section;
607 if (end == cursor) {
608 ++cursor;
609 continue;
610 }
611 cursor = end;
612 if (frequency < 50.0 || frequency > 300.0)
613 continue;
614 /* Higher order supplies 50 dB rejection at the TIA-603
615 * frequency-tolerance edges without widening into speech. */
616 for (section = 0; section < CTCSS_NOTCH_SECTIONS; ++section) {
617 snprintf(output_name, sizeof(output_name), "ctn%u_%u", notch,
618 section);
619 if (graph_append(graph, size,
620 "[%s]bandreject=f=%.9g:t=h:w=%.9g:r=f64[%s];",
621 input_name, frequency, cfg->ctcss_notch_width_hz,
622 output_name) < 0)
623 return AVERROR(ENOSPC);
624 snprintf(input_name, sizeof(input_name), "%s", output_name);
625 }
626 ++notch;
627 }
628 if (notch) {
629 snprintf(next, sizeof(next), "%s", input_name);
630 graph_input = next;
631 }
632 } else if (cfg->ctcss_filter_mode == TXAGC_CTCSS_FILTER_HIGHPASS &&
633 cfg->ctcss_highpass_hz > 0.0) {
634 if (graph_append(graph, size,
635 "[%s]acrossover=split=%.9g:order=20th[ctlow][cthigh];"
636 "[ctlow]anullsink;",
637 graph_input, cfg->ctcss_highpass_hz) < 0)
638 return AVERROR(ENOSPC);
639 graph_input = "cthigh";
640 }
641 if (graph_append(graph, size,
642 "[%s]asplit=2[programin][metertap];"
643 "[metertap]astats=metadata=1:reset=1:measure_perchannel=none:measure_"
644 "overall=Peak_level+RMS_level[in_meter];"
645 "[programin]volume=%.12g[%s];",
646 graph_input, db_to_linear(cfg->input_gain_db), current) < 0) {
647 return AVERROR(ENOSPC);
648 }
649
650 for (order_index = 0; order_index < cfg->stage_count; ++order_index) {
651 int enabled =
652 (cfg->stage_order[order_index] == TXAGC_STAGE_AGC && cfg->agc_enabled) ||
653 (cfg->stage_order[order_index] == TXAGC_STAGE_EXPANDER &&
654 cfg->expander_enabled) ||
655 (cfg->stage_order[order_index] == TXAGC_STAGE_COMPRESSOR &&
656 cfg->compressor_enabled) ||
657 (cfg->stage_order[order_index] == TXAGC_STAGE_LIMITER &&
658 cfg->limiter_enabled) ||
659 (cfg->stage_order[order_index] == TXAGC_STAGE_EQUALIZER &&
660 cfg->equalizer_enabled) ||
661 (cfg->stage_order[order_index] == TXAGC_STAGE_DEESSER &&
662 cfg->deesser_enabled);
663 if (!enabled)
664 continue;
665 snprintf(next, sizeof(next), "s%u", stage++);
666 if (add_dynamic_stage(graph, size, cfg, cfg->stage_order[order_index], current,
667 next, order_index) < 0)
668 return AVERROR(ENOSPC);
669 snprintf(current, sizeof(current), "%s", next);
670 }
671
672 if (cfg->preemphasis_enabled) {
673 snprintf(next, sizeof(next), "s%u", stage++);
674 if (add_emphasis(graph, size, current, next, 1, cfg->emphasis_corner_hz,
675 cfg->emphasis_reference_hz, sample_rate) < 0) {
676 return AVERROR(ENOSPC);
677 }
678 snprintf(current, sizeof(current), "%s", next);
679 }
680
681 if (cfg->dcs_spectral_shaping_enabled && cfg->dcs_spectral_lowpass_hz > 0.0) {
682 char rejected[NAME_SIZE];
683 snprintf(next, sizeof(next), "s%u", stage++);
684 snprintf(rejected, sizeof(rejected), "lp%urej", stage);
685 if (graph_append(graph, size,
686 "[%s]acrossover=split=%.9g:order=20th[%s][%s];[%s]anullsink;",
687 current, cfg->dcs_spectral_lowpass_hz, next, rejected,
688 rejected) < 0)
689 return AVERROR(ENOSPC);
690 snprintf(current, sizeof(current), "%s", next);
691 }
692
693 /* Output gain precedes the final limiter so it cannot defeat the configured
694 * ceiling. PCM conversion clamps only values that cannot be represented. */
695 if (cfg->output_gain_db != 0.0) {
696 snprintf(next, sizeof(next), "s%u", stage++);
697 if (graph_append(graph, size, "[%s]volume=%.12g[%s];", current,
698 db_to_linear(cfg->output_gain_db), next) < 0) {
699 return AVERROR(ENOSPC);
700 }
701 snprintf(current, sizeof(current), "%s", next);
702 }
703
704 if (cfg->lookahead_limiter_enabled) {
705 snprintf(next, sizeof(next), "s%u", stage++);
706 if (graph_append(graph, size,
707 "[%s]alimiter=limit=%.12g:attack=%.9g:release=%.9g:"
708 "level=0:latency=0[%s];",
709 current, db_to_linear(cfg->lookahead_limit_dbfs),
710 clamp(cfg->lookahead_ms, 0.1, 80.0), cfg->lookahead_release_ms,
711 next) < 0) {
712 return AVERROR(ENOSPC);
713 }
714 snprintf(current, sizeof(current), "%s", next);
715 }
716
717 if (cfg->post_limiter_lowpass_enabled) {
718 char rejected[NAME_SIZE];
719 snprintf(next, sizeof(next), "s%u", stage++);
720 snprintf(rejected, sizeof(rejected), "cln%urej", stage);
721 if (graph_append(graph, size,
722 "[%s]asplit=3[cleanupmain][cleanuppre][cleanupspec];"
723 "[cleanuppre]astats=metadata=1:reset=1:measure_perchannel=none:"
724 "measure_overall=Peak_level+RMS_level[cleanup_pre_meter];"
725 "[cleanupspec]acrossover=split=5000 8000:order=20th"
726 "[prebelow5][pre5to8][preabove8];"
727 "[prebelow5]anullsink;"
728 "[pre5to8]astats=metadata=1:reset=1:measure_perchannel=none:"
729 "measure_overall=RMS_level[cleanup_pre_5_8_meter];"
730 "[preabove8]astats=metadata=1:reset=1:measure_perchannel=none:"
731 "measure_overall=RMS_level[cleanup_pre_8_plus_meter];"
732 "[cleanupmain]acrossover=split=%.9g:order=20th[%s][%s];"
733 "[%s]anullsink;",
734 current, cfg->post_limiter_lowpass_hz, next, rejected,
735 rejected) < 0) {
736 return AVERROR(ENOSPC);
737 }
738 snprintf(current, sizeof(current), "%s", next);
739 }
740
741 if (cfg->post_limiter_lowpass_enabled) {
742 return graph_append(graph, size,
743 "[%s]asplit=2[postmain][postspec];"
744 "[postspec]acrossover=split=5000 8000:order=20th"
745 "[postbelow5][post5to8][postabove8];"
746 "[postbelow5]anullsink;"
747 "[post5to8]astats=metadata=1:reset=1:measure_perchannel=none:"
748 "measure_overall=RMS_level[cleanup_post_5_8_meter];"
749 "[postabove8]astats=metadata=1:reset=1:measure_perchannel=none:"
750 "measure_overall=RMS_level[cleanup_post_8_plus_meter];"
751 "[postmain]astats=metadata=1:reset=1:measure_perchannel=none:"
752 "measure_overall=Peak_level+RMS_level[out]",
753 current);
754 }
755 return graph_append(graph, size,
756 "[%s]astats=metadata=1:reset=1:measure_perchannel=none:measure_overall="
757 "Peak_level+RMS_level[out]",
758 current);
759}

802AVFILTER_PRIVATE unsigned int input_capacity_for_rate(unsigned int sample_rate)
803{
804 unsigned int capacity;
805
806 if (!sample_rate || sample_rate > INT_MAX)
807 return 0;
808 capacity = sample_rate / AVFILTER_INPUT_CAPACITY_DIVISOR;
809 return capacity < AVFILTER_INPUT_CAPACITY_MIN ? AVFILTER_INPUT_CAPACITY_MIN : capacity;
810}

817AVFILTER_PRIVATE int allocate_runtime_buffers(struct txagc_avfilter *state,
818 unsigned int sample_rate)
819{
820 unsigned int capacity = input_capacity_for_rate(sample_rate);
821
822 /* configure() has already rejected a rate that cannot produce a capacity.
823 * It also divides every accepted rate by ten, so multiplying the result by
824 * the fixed pool count cannot overflow an int. */
825 state->input_capacity = capacity;
826 state->fifo_capacity = capacity * TXAGC_AVFILTER_INPUT_FRAME_COUNT;
827 state->fifo = av_audio_fifo_alloc(AV_SAMPLE_FMT_DBL, 1, (int)state->fifo_capacity);
828 if (!state->fifo)
829 return AVERROR(ENOMEM);
830 for (size_t index = 0; index < TXAGC_AVFILTER_INPUT_FRAME_COUNT; ++index) {
831 AVFrame *frame = av_frame_alloc();
832 int result;
833
834 if (!frame)
835 return AVERROR(ENOMEM);
836 frame->format = AV_SAMPLE_FMT_DBL;
837 frame->sample_rate = (int)sample_rate;
838 frame->nb_samples = (int)capacity;
839 av_channel_layout_default(&frame->ch_layout, 1);
840 result = av_frame_get_buffer(frame, 0);
841 if (result < 0) {
842 av_frame_free(&frame);
843 return result;
844 }
845 state->input_frames[index] = frame;
846 }
847 state->output_frame = av_frame_alloc();
848 return state->output_frame ? 0 : AVERROR(ENOMEM);
849}

857AVFILTER_PRIVATE int configure(struct txagc_avfilter *state, const struct txagc_config *config,
858 unsigned int sample_rate)
859{
860 const AVFilter *source_filter;
861 const AVFilter *sink_filter;
862 AVFilterInOut *inputs = NULL;
863 AVFilterInOut *outputs = NULL;
864 AVFilterInOut *meter_input = NULL;
865 AVFilterInOut *meter_tail = NULL;
866 AVFilterInOut *cleanup_input;
867 AVFilterContext **cleanup_sink_slots[] = {
868 &state->cleanup_pre_sink, &state->cleanup_pre_5_8_sink,
869 &state->cleanup_pre_8_plus_sink, &state->cleanup_post_5_8_sink,
870 &state->cleanup_post_8_plus_sink,
871 };
872 const char *cleanup_filter_names[] = {
873 "cleanup_pre_sink", "cleanup_pre_5_8_sink", "cleanup_pre_8_plus_sink",
874 "cleanup_post_5_8_sink", "cleanup_post_8_plus_sink",
875 };
876 const char *cleanup_pad_names[] = {
877 "cleanup_pre_meter", "cleanup_pre_5_8_meter", "cleanup_pre_8_plus_meter",
878 "cleanup_post_5_8_meter", "cleanup_post_8_plus_meter",
879 };
880 char args[256];
881 char description[GRAPH_SIZE];
882 int result;
883
884 if (!state || !config || !input_capacity_for_rate(sample_rate))
885 return AVERROR(EINVAL);
886 free_graph(state);
887 result = build_description(description, sizeof(description), config, sample_rate);
888 if (result < 0) {
889 return result;
890 }
891 state->graph = avfilter_graph_alloc();
892 if (!state->graph) {
893 return AVERROR(ENOMEM);
894 }
895 source_filter = avfilter_get_by_name("abuffer");
896 sink_filter = avfilter_get_by_name("abuffersink");
897 snprintf(args, sizeof(args),
898 "time_base=1/%u:sample_rate=%u:sample_fmt=dbl:channel_layout=mono", sample_rate,
899 sample_rate);
900 result = avfilter_graph_create_filter(&state->source, source_filter, "source", args, NULL,
901 state->graph);
902 if (result < 0) {
903 goto fail;
904 }
905 if (config->post_limiter_lowpass_enabled) {
906 for (size_t index = 0; index < 5; ++index) {
907 AVFilterContext **sink_slot = cleanup_sink_slots[index];
908 result = avfilter_graph_create_filter(sink_slot, sink_filter,
909 cleanup_filter_names[index], NULL,
910 NULL, state->graph);
911 if (result < 0) {
912 goto fail;
913 }
914 set_double_sample_format(*sink_slot);
915 }
916 }
917 result = avfilter_graph_create_filter(&state->sink, sink_filter, "sink", NULL, NULL,
918 state->graph);
919 if (result < 0) {
920 goto fail;
921 }
922 result = avfilter_graph_create_filter(&state->meter_sink, sink_filter, "meter_sink", NULL,
923 NULL, state->graph);
924 if (result < 0) {
925 goto fail;
926 }
927 set_double_sample_format(state->sink);
928 set_double_sample_format(state->meter_sink);
929
930 outputs = avfilter_inout_alloc();
931 inputs = avfilter_inout_alloc();
932 meter_input = avfilter_inout_alloc();
933 if (!outputs || !inputs || !meter_input) {
934 result = AVERROR(ENOMEM);
935 goto fail;
936 }
937 outputs->name = av_strdup("in");
938 outputs->filter_ctx = state->source;
939 outputs->pad_idx = 0;
940 inputs->name = av_strdup("out");
941 inputs->filter_ctx = state->sink;
942 inputs->pad_idx = 0;
943 inputs->next = meter_input;
944 meter_tail = meter_input;
945 meter_input->name = av_strdup("in_meter");
946 meter_input->filter_ctx = state->meter_sink;
947 meter_input->pad_idx = 0;
948 if (config->post_limiter_lowpass_enabled) {
949 for (size_t index = 0; index < 5; ++index) {
950 cleanup_input = avfilter_inout_alloc();
951 if (!cleanup_input) {
952 result = AVERROR(ENOMEM);
953 goto fail;
954 }
955 meter_tail->next = cleanup_input;
956 meter_tail = cleanup_input;
957 cleanup_input->name = av_strdup(cleanup_pad_names[index]);
958 cleanup_input->filter_ctx = *cleanup_sink_slots[index];
959 cleanup_input->pad_idx = 0;
960 }
961 }
962 result = avfilter_graph_parse_ptr(state->graph, description, &inputs, &outputs, NULL);
963 if (result < 0) {
964 goto fail;
965 }
966 result = avfilter_graph_config(state->graph, NULL);
967 if (result < 0) {
968 goto fail;
969 }
970 result = allocate_runtime_buffers(state, sample_rate);
971 if (result < 0) {
972 goto fail;
973 }
974 memcpy(&state->config, config, sizeof(*config));
975 state->sample_rate = sample_rate;
976 state->configured = 1;
977 state->failed = 0;
978 state->latency_samples = 0;
979 state->buffered_samples = 0;
980 state->output_started = 0;
981 state->startup_fill_samples = 0;
982 avfilter_inout_free(&inputs);
983 avfilter_inout_free(&outputs);
984 return 0;
985
986fail:
987 avfilter_inout_free(&inputs);
988 avfilter_inout_free(&outputs);
989 free_graph(state);
990 state->failed = result;
991 return result;
992}

1265int txagc_avfilter_slot_publish_candidate(struct txagc_avfilter_slot *slot,
1266 struct txagc_avfilter_slot_candidate *candidate)
1267{
1268 struct txagc_avfilter_slot_node *old;
1269
1270 if (!slot || !candidate || !candidate->node)
1271 return AVERROR(EINVAL);
1272 slot_lock(slot);
1273 old = slot->owned;
1274 /* owned is assigned only after candidate preparation completes, so an
1275 * existing node always contains a configured graph. */
1276 if (old && old->filter.sample_rate == candidate->node->filter.sample_rate &&
1277 txagc_config_equal(&old->filter.config, &candidate->node->filter.config)) {
1278 slot_unlock(slot);
1279 txagc_avfilter_slot_candidate_destroy(candidate);
1280 return 0;
1281 }
1282 /* Publish only a complete graph. Readers acquire active before touching any
1283 * graph member, so an unpublished candidate cannot disturb active audio. */
1284 atomic_store_explicit(&slot->active, &candidate->node->filter, memory_order_seq_cst);
1285 slot_wait_for_readers(slot);
1286 slot->owned = candidate->node;
1287 candidate->node = NULL;
1288 if (old) {
1289 txagc_avfilter_destroy(&old->filter);
1290 av_free(old);
1291 }
1292 slot_unlock(slot);
1293 return 0;
1294}

1296int txagc_avfilter_slot_prepare(struct txagc_avfilter_slot *slot, const struct txagc_config *config,
1297 unsigned int sample_rate)
1298{
1299 struct txagc_avfilter_slot_candidate candidate = {0};
1300 const struct txagc_avfilter_slot_node *active;
1301 int result;
1302
1303 if (!slot || !config)
1304 return AVERROR(EINVAL);
1305 /* Preserve the no-op fast path: an unchanged control-plane reload must not
1306 * allocate an FFmpeg candidate simply to discard it moments later. */
1307 slot_lock(slot);
1308 active = slot->owned;
1309 /* An owned node is always a successfully prepared graph. */
1310 if (active && active->filter.sample_rate == sample_rate &&
1311 txagc_config_equal(&active->filter.config, config)) {
1312 slot_unlock(slot);
1313 return 0;
1314 }
1315 slot_unlock(slot);
1316 result = txagc_avfilter_slot_candidate_prepare(&candidate, config, sample_rate);
1317
1318 if (result < 0)
1319 return result;
1320 /* A locally initialized candidate and non-NULL slot satisfy the only
1321 * fallible preconditions of publication. It either consumes the candidate
1322 * or completes a semantic no-op, so no post-publication cleanup is needed. */
1323 return txagc_avfilter_slot_publish_candidate(slot, &candidate);
1324}

1348{
1349 struct txagc_avfilter *state;
1350
1351 if (!slot)
1352 return NULL;
1353 /* Claim a read-side reference before loading active. These operations pair
1354 * with replacement's sequentially-consistent active store and reader count
1355 * wait: either the reader pins the old graph before replacement observes its
1356 * count, or it loads the replacement after that zero-count observation. */
1357 atomic_fetch_add_explicit(&slot->readers, 1U, memory_order_seq_cst);
1358 state = atomic_load_explicit(&slot->active, memory_order_seq_cst);
1359 if (!state)
1360 atomic_fetch_sub_explicit(&slot->readers, 1U, memory_order_seq_cst);
1361 return state;
1362}

1420int txagc_avfilter_process_prepared(struct txagc_avfilter *state, double *samples, size_t count)
1421{
1422 AVFrame *input;
1423 AVFrame *output;
1424 int available;
1425 int result;
1426 int written;
1427 double *output_pointer;
1428 size_t copied = 0;
1429
1430 if (!state || !samples || !count || !state->configured || !state->fifo ||
1431 !state->output_frame || count > state->input_capacity)
1432 return AVERROR(EINVAL);
1433 input = next_input_frame(state);
1434 if (!input)
1435 return AVERROR(EAGAIN);
1436 output = state->output_frame;
1437 input->nb_samples = (int)count;
1438 /* Asterisk/usbradio uses floating point with 16-bit PCM units while
1439 * libavfilter's DBL format uses the normalized -1.0 .. +1.0 convention. */
1440 for (size_t index = 0; index < count; ++index) {
1441 ((double *)input->data[0])[index] = samples[index] / 32768.0;
1442 }
1443 result = av_buffersrc_add_frame_flags(state->source, input,
1444 AV_BUFFERSRC_FLAG_KEEP_REF | AV_BUFFERSRC_FLAG_PUSH);
1445 if (result < 0) {
1446 goto done;
1447 }
1448 state->input_samples += count;
1449 while ((result = av_buffersink_get_frame(state->meter_sink, output)) >= 0) {
1450 update_astats(state, output, 1);
1451 av_frame_unref(output);
1452 }
1453 if (result != AVERROR(EAGAIN) && result != AVERROR_EOF) {
1454 goto done;
1455 }
1456 if (state->config.post_limiter_lowpass_enabled) {
1457 result = drain_cleanup_meter(state, state->cleanup_pre_sink, output,
1458 CLEANUP_PRE_FULL);
1459 if (result < 0)
1460 goto done;
1461 result = drain_cleanup_meter(state, state->cleanup_pre_5_8_sink, output,
1462 CLEANUP_PRE_5_8);
1463 if (result < 0)
1464 goto done;
1465 result = drain_cleanup_meter(state, state->cleanup_pre_8_plus_sink, output,
1466 CLEANUP_PRE_8_PLUS);
1467 if (result < 0)
1468 goto done;
1469 result = drain_cleanup_meter(state, state->cleanup_post_5_8_sink, output,
1470 CLEANUP_POST_5_8);
1471 if (result < 0)
1472 goto done;
1473 result = drain_cleanup_meter(state, state->cleanup_post_8_plus_sink, output,
1474 CLEANUP_POST_8_PLUS);
1475 if (result < 0)
1476 goto done;
1477 }
1478
1479 while ((result = av_buffersink_get_frame(state->sink, output)) >= 0) {
1480 void *fifo_data[1];
1481 update_astats(state, output, 0);
1482 if (output->nb_samples > av_audio_fifo_space(state->fifo)) {
1483 result = AVERROR(ENOSPC);
1484 goto done;
1485 }
1486 output_pointer = (double *)output->data[0];
1487 fifo_data[0] = output_pointer;
1488 written = av_audio_fifo_write(state->fifo, fifo_data, output->nb_samples);
1489 if (written != output->nb_samples) {
1490 result = AVERROR(EIO);
1491 goto done;
1492 }
1493 av_frame_unref(output);
1494 }
1495 if (result != AVERROR(EAGAIN) && result != AVERROR_EOF) {
1496 goto done;
1497 }
1498 available = av_audio_fifo_size(state->fifo);
1499 copied = available < (int)count ? (size_t)available : count;
1500 if (!state->output_started) {
1501 size_t fill = count - copied;
1502 memset(samples, 0, fill * sizeof(*samples));
1503 state->startup_fill_samples += fill;
1504 state->underrun_samples += fill;
1505 if (copied) {
1506 output_pointer = samples + fill;
1507 av_audio_fifo_read(state->fifo, (void **)&output_pointer, (int)copied);
1508 for (size_t index = fill; index < count; ++index) {
1509 samples[index] *= 32768.0;
1510 }
1511 state->latency_samples = (unsigned int)state->startup_fill_samples;
1512 state->output_started = 1;
1513 }
1514 } else {
1515 if (copied) {
1516 output_pointer = samples;
1517 av_audio_fifo_read(state->fifo, (void **)&output_pointer, (int)copied);
1518 for (size_t index = 0; index < copied; ++index) {
1519 samples[index] *= 32768.0;
1520 }
1521 }
1522 if (copied < count) {
1523 memset(samples + copied, 0, (count - copied) * sizeof(*samples));
1524 state->underrun_samples += count - copied;
1525 state->runtime_underrun_samples += count - copied;
1526 }
1527 }
1528 state->buffered_samples = (unsigned int)av_audio_fifo_size(state->fifo);
1529 state->output_samples += copied;
1530 result = 0;
1531
1532done:
1533 av_frame_unref(output);
1534 return result;
1535}
