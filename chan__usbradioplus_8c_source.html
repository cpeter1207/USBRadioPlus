381URP_CHANNEL_LOCAL void *hidthread(void *arg)
382{
383 unsigned char buf[4], bufsave[4], keyed, ctcssed;
384 char *s, lasttxtmp;
385 register int i, j, k;
386 int res;
387 struct usb_device *usb_dev;
388 struct usb_dev_handle *usb_handle;
389 struct chan_usbradio_pvt *o = arg, *ao;
390 struct timeval then;
391 struct pollfd rfds[1];
392
393 usb_dev = NULL;
394 usb_handle = NULL;
395 /* enable gpio_set so that we will write GPIO information upon start up */
396 o->gpio_set = 1;
397
398#ifdef HAVE_SYS_IO
399 if (haspp == 2) {
400 ioperm(pbase, 2, 1);
401 }
402#endif
403 /* This is the main loop for this thread.
404 * It performs setup and initialization of the usb device.
405 * After setup is complete and the device can be accessed,
406 * it enters a processing loop responsible for interacting
407 * with the usb hid device
408 */
409 while (!o->stophid) {
410 char serial[sizeof(o->serial)] = {'\0'};
411
412 ast_radio_time(&o->lasthidtime);
413 ast_mutex_lock(&usb_dev_lock);
414 o->hasusb = 0;
415 o->usbass = 0;
416 o->devicenum = 0;
417 if (usb_handle) {
418 usb_close(usb_handle);
419 }
420 usb_handle = NULL;
421 usb_dev = NULL;
422 ast_radio_hid_device_mklist();
423
424 /* Check to see if our specified device string
425 * matches to a device that is attached to this system, or exists
426 * in our channel configuration.
427 *
428 * If no device string is specified, attempt to assign the first
429 * found device.
430 */
431 ast_radio_time(&o->lasthidtime);
432
433 /* If configuration has a serial number defined, find the device */
434 if (!ast_strlen_zero(o->serial)) {
435 int index;
436 char *index_devstr = NULL;
437
438 for (index = 0;; index++) {
439 index_devstr = ast_radio_usb_get_devstr(index);
440 if (ast_strlen_zero(index_devstr)) {
441 /* if no more devices */
442 break;
443 }
444
445 /* get the device serial number */
446 if (ast_radio_usb_get_serial(index_devstr, serial,
447 sizeof(serial)) == 0) {
448 /* if no serial number */
449 continue;
450 }
451
452 if (strcmp(o->serial, serial) == 0) {
453 /*
454 * We found a device with the matching serial number, set
455 * the devstr to the matching device.
456 */
457 ast_log(LOG_NOTICE, "Matched device serial %s to %s\n",
458 o->serial, o->name);
459 ast_copy_string(o->devstr, index_devstr, sizeof(o->devstr));
460 break;
461 }
462 }
463 }
464
465 /* Automatically assign a devstr if one was not specified in the configuration. */
466 if (ast_strlen_zero(o->devstr)) {
467 int index = 0;
468 char *index_devstr = NULL;
469
470 for (;;) {
471 index_devstr = ast_radio_usb_get_devstr(index);
472 if (ast_strlen_zero(index_devstr)) {
473 if (!o->device_error) {
474 ast_log(LOG_ERROR,
475 "Channel %s: No USB devices are available "
476 "for assignment.\n",
477 o->name);
478 o->device_error = 1;
479 }
480 ast_mutex_unlock(&usb_dev_lock);
481 usleep(500000);
482 break;
483 }
484 /* We found an available device - see if it already in use */
485 for (ao = usbradio_default.next; ao; ao = ao->next) {
486 if (ao->usbass && (!strcmp(ao->devstr, index_devstr))) {
487 break;
488 }
489 }
490 if (ao) {
491 index++;
492 continue;
493 }
494 /* We found an unused device assign it to our node */
495 ast_copy_string(o->devstr, index_devstr, sizeof(o->devstr));
496 ast_log(LOG_NOTICE,
497 "Channel %s: Automatically assigned USB device %s to "
498 "USBRadio channel\n",
499 o->name, o->devstr);
500 if (ast_radio_usb_get_serial(index_devstr, serial, sizeof(serial)) >
501 0) {
502 ast_copy_string(o->serial, serial, sizeof(o->serial));
503 }
504 break;
505 }
506 if (ast_strlen_zero(o->devstr)) {
507 continue;
508 }
509 }
510
511 if (!ast_radio_usb_list_check(o->devstr)) {
512 /* The device string did not match.
513 * Now look through the attached devices and see
514 * one of those is associated with one of our
515 * configured channels.
516 */
517 s = find_installed_usb_match();
518 if (ast_strlen_zero(s)) {
519 if (!o->device_error) {
520 ast_log(LOG_ERROR,
521 "Channel %s: Device string %s was not found.\n",
522 o->name, o->devstr);
523 o->device_error = 1;
524 }
525 ast_mutex_unlock(&usb_dev_lock);
526 usleep(500000);
527 continue;
528 }
529 i = ast_radio_usb_get_usbdev(s);
530 if (i < 0) {
531 ast_mutex_unlock(&usb_dev_lock);
532 usleep(500000);
533 continue;
534 }
535 /* See if this device is already assigned to another usb channel */
536 for (ao = usbradio_default.next; ao; ao = ao->next) {
537 if (ao->usbass && (!strcmp(ao->devstr, s))) {
538 break;
539 }
540 }
541 if (ao) {
542 ast_log(LOG_ERROR,
543 "Channel %s: Device string %s is already assigned to "
544 "channel %s",
545 o->name, s, ao->name);
546 ast_mutex_unlock(&usb_dev_lock);
547 usleep(500000);
548 continue;
549 }
550 ast_log(LOG_NOTICE,
551 "Channel %s: Assigned USB device %s to usbradio channel\n", o->name,
552 s);
553 ast_copy_string(o->devstr, s, sizeof(o->devstr));
554 }
555 /* Double check to see if the device string is assigned to another usb channel */
556 for (ao = usbradio_default.next; ao; ao = ao->next) {
557 if (ao->usbass && (!strcmp(ao->devstr, o->devstr))) {
558 break;
559 }
560 }
561 if (ao) {
562 ast_log(LOG_ERROR,
563 "Channel %s: Device string %s is already assigned to channel %s",
564 o->name, o->devstr, ao->name);
565 ast_mutex_unlock(&usb_dev_lock);
566 usleep(500000);
567 continue;
568 }
569 /* get the index to the device and assign it to our channel */
570 i = ast_radio_usb_get_usbdev(o->devstr);
571 if (i < 0) {
572 ast_mutex_unlock(&usb_dev_lock);
573 usleep(500000);
574 continue;
575 }
576 o->devicenum = i;
577 o->device_error = 0;
578 ast_radio_time(&o->lasthidtime);
579 o->usbass = 1;
580 ast_mutex_unlock(&usb_dev_lock);
581 /* set the audio mixer values */
582 o->micmax = ast_radio_amixer_max(o->devicenum, MIXER_PARAM_MIC_CAPTURE_VOL);
583 o->spkrmax = ast_radio_amixer_max(o->devicenum, MIXER_PARAM_SPKR_PLAYBACK_VOL);
584 if (o->spkrmax == -1) {
585 o->newname = 1;
586 o->spkrmax = ast_radio_amixer_max(o->devicenum,
587 MIXER_PARAM_SPKR_PLAYBACK_VOL_NEW);
588 }
589 /* initialize the usb device */
590 usb_dev = ast_radio_hid_device_init(o->devstr);
591 if (usb_dev == NULL) {
592 ast_log(LOG_ERROR, "Channel %s: Cannot initialize device %s\n", o->name,
593 o->devstr);
594 usleep(500000);
595 continue;
596 }
597 /* open the usb device device */
598 usb_handle = usb_open(usb_dev);
599 if (usb_handle == NULL) {
600 ast_log(LOG_ERROR, "Channel %s: Cannot open device %s\n", o->name,
601 o->devstr);
602 usleep(500000);
603 continue;
604 }
605 /* attempt to claim the usb hid interface and detach from the kernel */
606 if (usb_claim_interface(usb_handle, C108_HID_INTERFACE) < 0) {
607 if (usb_detach_kernel_driver_np(usb_handle, C108_HID_INTERFACE) < 0) {
608 ast_log(LOG_ERROR,
609 "Channel %s: Is not able to detach the USB device\n",
610 o->name);
611 usleep(500000);
612 continue;
613 }
614 if (usb_claim_interface(usb_handle, C108_HID_INTERFACE) < 0) {
615 ast_log(LOG_ERROR,
616 "Channel %s: Is not able to claim the USB device\n",
617 o->name);
618 usleep(500000);
619 continue;
620 }
621 }
622 /* write initial value to GPIO */
623 memset(buf, 0, sizeof(buf));
624 buf[o->hid_gpio_ctl_loc] = o->hid_gpio_ctl;
625 buf[o->hid_gpio_loc] = o->hid_gpio_val;
626 ast_radio_hid_set_outputs(usb_handle, buf);
627 memcpy(bufsave, buf, sizeof(buf));
628 /* setup the pttkick pipe
629 * this pipe is used for timing the main processing loop
630 * it also signaled when the ptt changes to exit the timer
631 */
632 if (o->pttkick[0] != -1) {
633 close(o->pttkick[0]);
634 o->pttkick[0] = -1;
635 }
636 if (o->pttkick[1] != -1) {
637 close(o->pttkick[1]);
638 o->pttkick[1] = -1;
639 }
640 if (pipe(o->pttkick) == -1) {
641 ast_log(LOG_ERROR, "Channel %s: Is not able to create a pipe\n", o->name);
642 pthread_exit(NULL);
643 }
644 if ((usb_dev->descriptor.idProduct & 0xfffc) == C108_PRODUCT_ID) {
645 o->devtype = C108_PRODUCT_ID;
646 } else {
647 o->devtype = usb_dev->descriptor.idProduct;
648 }
649 ast_debug(5, "Channel %s: Starting normally.\n", o->name);
650 ast_debug(5, "Channel %s: Attached to usb device %s.\n", o->name, o->devstr);
651 /* setup the xmpr subsystem */
652 if (o->radio == NULL) {
653 urp_radio_state tChan;
654
655 memset(&tChan, 0, sizeof(urp_radio_state));
656
657 tChan.pTxCodeDefault = o->txctcssdefault;
658 tChan.pRxCodeSrc = o->rxctcssfreqs;
659 tChan.pTxCodeSrc = o->txctcssfreqs;
660
661 tChan.rxDemod = o->rxdemod;
662 tChan.rxCdType = effective_rxcdtype(o);
663 tChan.voxHangTime = o->voxhangtime;
664 tChan.rxSqVoxAdj = o->rxsqvoxadj;
665
666 tChan.txMixA = effective_txmixa(o);
667 tChan.txMixB = effective_txmixb(o);
668
669 tChan.rxCpuSaver = o->rxcpusaver;
670 tChan.txCpuSaver = o->txcpusaver;
671
672 tChan.b.rxpolarity = o->rxpolarity;
673 tChan.b.txpolarity = o->txpolarity;
674
675 tChan.b.dcsrxpolarity = o->dcsrxpolarity;
676 tChan.b.dcstxpolarity = o->dcstxpolarity;
677
678 tChan.b.lsdrxpolarity = o->lsdrxpolarity;
679 tChan.b.lsdtxpolarity = o->lsdtxpolarity;
680
681 tChan.tracetype = o->tracetype;
682 tChan.tracelevel = o->tracelevel;
683
684 tChan.rptnum = o->rptnum;
685 tChan.idleinterval = o->idleinterval;
686 tChan.turnoffs = o->turnoffs;
687 tChan.area = o->area;
688 tChan.ukey = o->ukey;
689 tChan.name = o->name;
690 tChan.fever = o->fever;
691
692 o->radio = urp_radio_create(&tChan, FRAME_SIZE);
693 if (!o->radio) {
694 ast_log(LOG_ERROR,
695 "Channel %s: signaling engine initialization failed\n",
696 o->name);
697 usleep(500000);
698 continue;
699 }
700
701 o->radio->radioDuplex = o->radioduplex;
702 o->radio->b.loopback = 0;
703 o->radio->txsettletime = o->txsettletime;
704 o->radio->txrxblankingtime = o->txrxblankingtime;
705 o->radio->rxCpuSaver = o->rxcpusaver;
706 o->radio->txCpuSaver = o->txcpusaver;
707
708 *(o->radio->prxSquelchAdjust) =
709 ((999 - o->rxsquelchadj) * 32767) / AUDIO_ADJUSTMENT;
710 *(o->radio->prxVoiceAdjust) = effective_rx_decoder_gain(o) * M_Q8;
711 *(o->radio->prxCtcssAdjust) = o->rxctcssadj * M_Q8;
712 o->radio->rxCtcss->relax = o->rxctcssrelax;
713 o->radio->txTocType = o->txtoctype;
714
715 if (urp_tx_pair_has_tone((enum urp_tx_output_mode)o->txmixa,
716 (enum urp_tx_output_mode)o->txmixb)) {
717 set_txctcss_level(o);
718 }
719
720 if (!urp_tx_pair_has_voice((enum urp_tx_output_mode)o->txmixa,
721 (enum urp_tx_output_mode)o->txmixb)) {
722 ast_log(LOG_ERROR, "Channel %s: No txvoice output configured.\n",
723 o->name);
724 }
725
726 if (urp_tx_tone_route_missing(o->txctcssfreq,
727 (enum urp_tx_output_mode)o->txmixa,
728 (enum urp_tx_output_mode)o->txmixb)) {
729 ast_log(LOG_ERROR, "No txtone output configured.\n");
730 }
731
732 if (o->radioactive) {
733 struct chan_usbradio_pvt *active_channel;
734 for (active_channel = usbradio_default.next; active_channel;
735 active_channel = active_channel->next)
736 active_channel->radioactive = 0;
737 usbradio_active = o->name;
738 o->radioactive = 1;
739 ast_log(LOG_NOTICE, "radio active set to [%s]\n", o->name);
740 }
741 }
742 radio_config(o);
743 mixer_write(o);
744 mult_set(o);
745
746 if (apply_processing_config_overrides(o, o->name))
747 ast_log(LOG_WARNING, "Unable to apply current RadioPlus settings for %s\n",
748 o->name);
749
750 mixer_write(o);
751 mult_set(o);
752 set_txctcss_level(o);
753
754 ast_mutex_lock(&o->eepromlock);
755 if (o->wanteeprom) {
756 o->eepromctl = 1;
757 }
758 ast_mutex_unlock(&o->eepromlock);
759
760 setformat(o, O_RDWR);
761 o->hasusb = 1;
762 o->had_gpios_in = 0;
763
764 memset(&rfds, 0, sizeof(rfds));
765 rfds[0].fd = o->pttkick[1];
766 rfds[0].events = POLLIN;
767
768 ast_radio_time(&o->lasthidtime);
769 /* Main processing loop for GPIO
770 * This loop process every 50 milliseconds.
771 * The timer can be interrupted by writing to
772 * the pttkick pipe.
773 */
774 while ((!o->stophid) && o->hasusb) {
775 then = ast_radio_tvnow();
776 /* poll the pttkick pipe - timeout after 50 milliseconds */
777 res = ast_poll(rfds, 1, 50);
778 if (res < 0) {
779 ast_log(LOG_WARNING, "Channel %s: Poll failed: %s\n", o->name,
780 strerror(errno));
781 usleep(10000);
782 continue;
783 }
784 if (rfds[0].revents) {
785 char c;
786
787 int bytes = read(o->pttkick[0], &c, 1);
788 if (bytes <= 0) {
789 ast_log(LOG_ERROR, "Channel %s: pttkick read failed: %s\n",
790 o->name, strerror(errno));
791 }
792 }
793 /* see if we need to process an eeprom read or write */
794 if (o->wanteeprom) {
795 ast_mutex_lock(&o->eepromlock);
796 if (o->eepromctl == 1) { /* to read */
797 /* if CS okay */
798 if (!ast_radio_get_eeprom(usb_handle, o->eeprom)) {
799 if (o->eeprom[EEPROM_USER_MAGIC_ADDR] !=
800 EEPROM_MAGIC) {
801 ast_log(LOG_ERROR,
802 "Channel %s: EEPROM bad magic "
803 "number\n",
804 o->name);
805 } else {
806 o->rxmixerset =
807 o->eeprom[EEPROM_USER_RXMIXERSET];
808 o->txmixaset =
809 o->eeprom[EEPROM_USER_TXMIXASET];
810 o->txmixbset =
811 o->eeprom[EEPROM_USER_TXMIXBSET];
812 o->txctcssadj =
813 o->eeprom[EEPROM_USER_TXCTCSSADJ];
814 o->rxsquelchadj =
815 o->eeprom[EEPROM_USER_RXSQUELCHADJ];
816 ast_log(LOG_NOTICE,
817 "Channel %s: EEPROM Loaded\n",
818 o->name);
819 mixer_write(o);
820 mult_set(o);
821 set_txctcss_level(o);
822 }
823 } else {
824 ast_log(LOG_ERROR,
825 "Channel %s: USB adapter has no EEPROM "
826 "installed or Checksum is bad\n",
827 o->name);
828 }
829 ast_radio_hid_set_outputs(usb_handle, bufsave);
830 }
831 if (o->eepromctl == 2) { /* to write */
832 ast_radio_put_eeprom(usb_handle, o->eeprom);
833 ast_radio_hid_set_outputs(usb_handle, bufsave);
834 ast_log(LOG_NOTICE,
835 "Channel %s: USB parameters written to EEPROM\n",
836 o->name);
837 }
838 o->eepromctl = 0;
839 ast_mutex_unlock(&o->eepromlock);
840 }
841 ast_mutex_lock(&o->usblock);
842 buf[o->hid_gpio_ctl_loc] = o->hid_gpio_ctl;
843 ast_radio_hid_get_inputs(usb_handle, buf);
844 /* See if we are keyed */
845 keyed = !(buf[o->hid_io_cor_loc] & o->hid_io_cor);
846 if (keyed != o->rxhidsq) {
847 ast_debug(2, "Channel %s: Update rxhidsq = %d\n", o->name, keyed);
848 o->rxhidsq = keyed;
849 }
850 /* See if we are receiving ctcss */
851 ctcssed = !(buf[o->hid_io_ctcss_loc] & o->hid_io_ctcss);
852 if (ctcssed != o->rxhidctcss) {
853 ast_debug(2, "Channel %s: Update rxhidctcss = %d\n", o->name,
854 ctcssed);
855 o->rxhidctcss = ctcssed;
856 }
857 /* Get the GPIO information */
858 j = buf[o->hid_gpio_loc];
859 /* If this device is a CM108AH, map the "HOOK" bit (which used to
860 be GPIO2 in the CM108 into the GPIO position */
861 if (o->devtype == C108AH_PRODUCT_ID) {
862 j |= 2; /* set GPIO2 bit */
863 /* if HOOK is asserted, clear GPIO bit */
864 if (buf[o->hid_io_cor_loc] & 0x10) {
865 j &= ~2;
866 }
867 }
868 for (i = 0; i < GPIO_PINCOUNT; i++) {
869 /* if a valid input bit, dont clear it */
870 if ((o->gpios[i]) && (!strcasecmp(o->gpios[i], "in")) &&
871 (o->valid_gpios & (1 << i))) {
872 continue;
873 }
874 j &= ~(1 << i); /* clear the bit, since its not an input */
875 }
876 if ((!o->had_gpios_in) || (o->last_gpios_in != j)) {
877 char buf1[100];
878 struct ast_frame fr = {
879 .frametype = AST_FRAME_TEXT,
880 .src = __PRETTY_FUNCTION__,
881 };
882
883 for (i = 0; i < GPIO_PINCOUNT; i++) {
884 /* skip if not specified */
885 if (!o->gpios[i]) {
886 continue;
887 }
888 /* skip if not input */
889 if (strcasecmp(o->gpios[i], "in")) {
890 continue;
891 }
892 /* skip if not a valid GPIO */
893 if (!(o->valid_gpios & (1 << i))) {
894 continue;
895 }
896 /* if bit has changed, or never reported */
897 if ((!o->had_gpios_in) ||
898 ((o->last_gpios_in & (1 << i)) != (j & (1 << i)))) {
899 snprintf(buf1, sizeof(buf1), "GPIO%d %d\n", i + 1,
900 (j & (1 << i)) ? 1 : 0);
901 fr.data.ptr = buf1;
902 fr.datalen = strlen(buf1);
903 ast_queue_frame(o->owner, &fr);
904 }
905 }
906 o->had_gpios_in = 1;
907 o->last_gpios_in = j;
908 }
909 /* process the parallel port GPIO */
910 if (haspp) {
911 ast_mutex_lock(&pp_lock);
912 j = k = ast_radio_ppread(haspp, ppfd, pbase, pport) ^
913 0x80; /* get PP input */
914 ast_mutex_unlock(&pp_lock);
915 for (i = 10; i <= 15; i++) {
916 /* if a valid input bit, dont clear it */
917 if ((o->pps[i]) && (!strcasecmp(o->pps[i], "in"))) {
918 continue;
919 }
920 j &= ~(1 << ppinshift[i]); /* clear the bit, since its not
921 an input */
922 }
923 if ((!o->had_pp_in) || (o->last_pp_in != j)) {
924 char buf1[100];
925 struct ast_frame fr = {
926 .frametype = AST_FRAME_TEXT,
927 .src = __PRETTY_FUNCTION__,
928 };
929
930 for (i = 10; i <= 15; i++) {
931 /* skip if not specified */
932 if (!o->pps[i]) {
933 continue;
934 }
935 /* skip if not input */
936 if (strcasecmp(o->pps[i], "in")) {
937 continue;
938 }
939 /* if bit has changed, or never reported */
940 if ((!o->had_pp_in) ||
941 ((o->last_pp_in & (1 << ppinshift[i])) !=
942 (j & (1 << ppinshift[i])))) {
943 snprintf(buf1, sizeof(buf1), "PP%d %d\n", i,
944 (j & (1 << ppinshift[i])) ? 1 : 0);
945 fr.data.ptr = buf1;
946 fr.datalen = strlen(buf1);
947 ast_queue_frame(o->owner, &fr);
948 }
949 }
950 o->had_pp_in = 1;
951 o->last_pp_in = j;
952 }
953 o->rxppsq = o->rxppctcss = 0;
954 for (i = 10; i <= 15; i++) {
955 if ((o->pps[i]) && (!strcasecmp(o->pps[i], "cor"))) {
956 j = k &
957 (1
958 << ppinshift[i]); /* set the bit accordingly */
959 if (j != o->rxppsq) {
960 ast_debug(
961 2,
962 "Channel %s: update rxppsq = %d\n",
963 o->name, j);
964 o->rxppsq = j;
965 }
966 } else if ((o->pps[i]) &&
967 (!strcasecmp(o->pps[i], "ctcss"))) {
968 o->rxppctcss =
969 k & (1 << ppinshift[i]); /* set the bit
970 accordingly */
971 }
972 }
973 }
974 j = ast_tvdiff_ms(ast_radio_tvnow(), then);
975 /* make output inversion mask (for pulseage) */
976 o->hid_gpio_lastmask = o->hid_gpio_pulsemask;
977 o->hid_gpio_pulsemask = 0;
978 for (i = 0; i < GPIO_PINCOUNT; i++) {
979 k = o->hid_gpio_pulsetimer[i];
980 if (k) {
981 k -= j;
982 if (k < 0) {
983 k = 0;
984 }
985 o->hid_gpio_pulsetimer[i] = k;
986 }
987 if (k) {
988 o->hid_gpio_pulsemask |= 1 << i;
989 }
990 }
991 if (o->hid_gpio_pulsemask ||
992 o->hid_gpio_lastmask) { /* if anything inverted (temporarily) */
993 buf[o->hid_gpio_loc] = o->hid_gpio_val ^ o->hid_gpio_pulsemask;
994 buf[o->hid_gpio_ctl_loc] = o->hid_gpio_ctl;
995 ast_radio_hid_set_outputs(usb_handle, buf);
996 }
997 if (o->gpio_set) {
998 o->gpio_set = 0;
999 buf[o->hid_gpio_loc] = o->hid_gpio_val ^ o->hid_gpio_pulsemask;
1000 buf[o->hid_gpio_ctl_loc] = o->hid_gpio_ctl;
1001 ast_radio_hid_set_outputs(usb_handle, buf);
1002 }
1003 /* Apply transmitter state changes from the signaling engine. */
1004 lasttxtmp = o->radio->txPttOut;
1005
1006 k = 0;
1007 if (haspp) {
1008 for (i = 2; i <= 9; i++) {
1009 /* skip if this one not specified */
1010 if (!o->pps[i]) {
1011 continue;
1012 }
1013 /* skip if not ptt */
1014 if (strncasecmp(o->pps[i], "ptt", 3)) {
1015 continue;
1016 }
1017 k |= (1 << (i - 2)); /* make mask */
1018 }
1019 }
1020 if (o->lasttx != lasttxtmp) {
1021 o->radio->txPttHid = o->lasttx = lasttxtmp;
1022 ast_debug(2, "Channel %s: tx set to %d\n", o->name, o->lasttx);
1023 ast_mutex_lock(&pp_lock);
1024 urp_apply_ptt_outputs(lasttxtmp, o->invertptt, (uint8_t)k,
1025 o->hid_io_ptt, &o->hid_gpio_val, &pp_val);
1026 if (k) {
1027 ast_radio_ppwrite(haspp, ppfd, pbase, pport, pp_val);
1028 }
1029 ast_mutex_unlock(&pp_lock);
1030 buf[o->hid_gpio_loc] = o->hid_gpio_val ^ o->hid_gpio_pulsemask;
1031 buf[o->hid_gpio_ctl_loc] = o->hid_gpio_ctl;
1032 memcpy(bufsave, buf, sizeof(buf));
1033 ast_radio_hid_set_outputs(usb_handle, buf);
1034 }
1035 ast_radio_time(&o->lasthidtime);
1036 ast_mutex_unlock(&o->usblock);
1037 }
1038 o->radio->txPttOut = 0;
1039 o->lasttx = 0;
1040 ast_mutex_lock(&o->usblock);
1041 o->hid_gpio_val &= ~o->hid_io_ptt;
1042 if (o->invertptt) {
1043 o->hid_gpio_val |= o->hid_io_ptt;
1044 }
1045 buf[o->hid_gpio_loc] = o->hid_gpio_val ^ o->hid_gpio_pulsemask;
1046 buf[o->hid_gpio_ctl_loc] = o->hid_gpio_ctl;
1047 ast_radio_hid_set_outputs(usb_handle, buf);
1048 ast_mutex_unlock(&o->usblock);
1049 }
1050 /* clean up before exiting the thread */
1051 if (o->radio) {
1052 o->radio->txPttOut = 0;
1053 }
1054 o->lasttx = 0;
1055 if (usb_handle) {
1056 ast_mutex_lock(&o->usblock);
1057 o->hid_gpio_val &= ~o->hid_io_ptt;
1058 if (o->invertptt) {
1059 o->hid_gpio_val |= o->hid_io_ptt;
1060 }
1061 buf[o->hid_gpio_loc] = o->hid_gpio_val;
1062 buf[o->hid_gpio_ctl_loc] = o->hid_gpio_ctl;
1063 ast_radio_hid_set_outputs(usb_handle, buf);
1064 ast_mutex_unlock(&o->usblock);
1065 }
1066 pthread_exit(0);
1067}

1069URP_CHANNEL_LOCAL int used_blocks(struct chan_usbradio_pvt *o)
1070{
1071 struct audio_buf_info info;
1072
1073 if (ioctl(o->sounddev, SNDCTL_DSP_GETOSPACE, &info)) {
1074 if (!(o->warned & WARN_used_blocks)) {
1075 ast_log(LOG_WARNING, "Channel %s: Error reading output space.\n", o->name);
1076 o->warned |= WARN_used_blocks;
1077 }
1078 return 1;
1079 }
1080
1081 /* Cache capacity, not the free-space snapshot: idle playback remains active. */
1082 if (o->total_blocks == 0) {
1083 ast_debug(1, "Channel %s: fragment total %d, size %d, available %d, bytes %d\n",
1084 o->name, info.fragstotal, info.fragsize, info.fragments, info.bytes);
1085 o->total_blocks = info.fragstotal;
1086 /* Check the queue size, it cannot exceed the total fragments */
1087 if (o->queuesize >= (unsigned int)info.fragstotal) {
1088 o->queuesize = info.fragstotal - 1;
1089 if (o->queuesize < 2) {
1090 o->queuesize = QUEUE_SIZE;
1091 }
1092 ast_debug(1, "Channel %s: Queue size reset to %d\n", o->name, o->queuesize);
1093 }
1094 }
1095
1096 return o->total_blocks - info.fragments;
1097}

1099URP_CHANNEL_LOCAL int soundcard_writeframe(struct chan_usbradio_pvt *o, short *data)
1100{
1101 static const short silence[URP_NATIVE_SAMPLES * 2] = {0};
1102 const short *output = data;
1103 int res;
1104
1105 /* If the sound device is not open, setformat will open the device */
1106 if (o->sounddev < 0) {
1107 setformat(o, O_RDWR);
1108 }
1109 if (o->sounddev < 0) {
1110 return 0; /* not fatal */
1111 }
1112 /* Keep the DAC fed at the capture clock's cadence, including while idle.
1113 * PTT selects audio or silence; it never starts or stops device writes. */
1114 if (!o->radio->txPttIn && !o->radio->txPttOut) {
1115 output = silence;
1116 }
1117 /* Bound device latency without adding an extra silent block on an empty queue. */
1118 res = used_blocks(o);
1119 if ((unsigned int)res > o->queuesize) { /* no room to write a block */
1120 o->plus_sound_dropped_frames++;
1121 ast_log(LOG_WARNING,
1122 "Channel %s: Sound device write buffer overflow - used %d blocks\n",
1123 o->name, res);
1124 return 0;
1125 }
1126 res = write(o->sounddev, output, sizeof(silence));
1127 if (res < 0) {
1128 o->plus_sound_short_writes++;
1129 ast_log(LOG_ERROR, "Channel %s: Sound card write error %s\n", o->name,
1130 strerror(errno));
1131 } else if (res != (int)sizeof(silence)) {
1132 o->plus_sound_short_writes++;
1133 ast_log(LOG_ERROR, "Channel %s: Sound card wrote %d bytes of %zu\n", o->name, res,
1134 sizeof(silence));
1135 }
1136
1137 return res;
1138}

1140URP_CHANNEL_LOCAL int setformat(struct chan_usbradio_pvt *o, int mode)
1141{
1142 int fmt, desired, res, fd;
1143 char device[100];
1144
1145 /* A reopened device may have a different playback-buffer capacity. */
1146 o->total_blocks = 0;
1147 /* If the device is open, close it */
1148 if (o->sounddev >= 0) {
1149 ioctl(o->sounddev, SNDCTL_DSP_RESET, 0);
1150 close(o->sounddev);
1151 o->duplex = M_UNSET;
1152 o->sounddev = -1;
1153 }
1154 if (mode == O_CLOSE) { /* we are done */
1155 return 0;
1156 }
1157
1158 ast_copy_string(device, "/dev/dsp", sizeof(device));
1159 if (o->devicenum) {
1160 sprintf(device, "/dev/dsp%d", o->devicenum);
1161 }
1162 /* open the device */
1163 fd = o->sounddev = open(device, mode | O_NONBLOCK);
1164 if (fd < 0) {
1165 ast_log(LOG_ERROR, "Channel %s: Unable to open DSP device %d: %s.\n", o->name,
1166 o->devicenum, strerror(errno));
1167 return -1;
1168 }
1169 if (o->owner) {
1170 ast_channel_internal_fd_set(o->owner, 0, fd);
1171 }
1172
1173#if __BYTE_ORDER == __LITTLE_ENDIAN
1174 fmt = AFMT_S16_LE;
1175#else
1176 fmt = AFMT_S16_BE;
1177#endif
1178 res = ioctl(fd, SNDCTL_DSP_SETFMT, &fmt);
1179 if (res < 0) {
1180 ast_log(LOG_WARNING, "Channel %s: Unable to set format to 16-bit signed\n",
1181 o->name);
1182 return -1;
1183 }
1184 /* set our duplex mode based on the way we opened the device. */
1185 switch (mode) {
1186 case O_RDWR:
1187 (void)ioctl(fd, SNDCTL_DSP_SETDUPLEX, 0);
1188 /* Check to see if duplex set (FreeBSD Bug) */
1189 res = ioctl(fd, SNDCTL_DSP_GETCAPS, &fmt);
1190 if (res == 0 && (fmt & DSP_CAP_DUPLEX)) {
1191 o->duplex = M_FULL;
1192 }
1193 break;
1194 case O_WRONLY:
1195 o->duplex = M_WRITE;
1196 break;
1197 case O_RDONLY:
1198 o->duplex = M_READ;
1199 break;
1200 default:
1201 break;
1202 }
1203
1204 fmt = 1;
1205 res = ioctl(fd, SNDCTL_DSP_STEREO, &fmt);
1206 if (res < 0) {
1207 ast_log(LOG_WARNING, "Channel %s: Failed to set audio device to stereo\n", o->name);
1208 return -1;
1209 }
1210 fmt = desired = 48000; /* 48000 Hz desired */
1211 res = ioctl(fd, SNDCTL_DSP_SPEED, &fmt);
1212 if (res < 0) {
1213 ast_log(LOG_WARNING, "Channel %s: Failed to set audio device sample rate.\n",
1214 o->name);
1215 return -1;
1216 }
1217 if (fmt != desired) {
1218 if (!(o->warned & WARN_speed)) {
1219 ast_log(LOG_WARNING,
1220 "Channel %s: Requested %d Hz, got %d Hz -- sound may be choppy.\n",
1221 o->name, desired, fmt);
1222 o->warned |= WARN_speed;
1223 }
1224 }
1225 /*
1226 * on Freebsd, SETFRAGMENT does not work very well on some cards.
1227 * Default to use 256 bytes, let the user override
1228 */
1229 if (o->frags) {
1230 fmt = o->frags;
1231 res = ioctl(fd, SNDCTL_DSP_SETFRAGMENT, &fmt);
1232 if (res < 0) {
1233 if (!(o->warned & WARN_frag)) {
1234 ast_log(LOG_WARNING,
1235 "Channel %s: Unable to set fragment size -- sound may be "
1236 "choppy.\n",
1237 o->name);
1238 o->warned |= WARN_frag;
1239 }
1240 }
1241 }
1242 /* on some cards, we need SNDCTL_DSP_SETTRIGGER to start outputting */
1243 res = PCM_ENABLE_INPUT | PCM_ENABLE_OUTPUT;
1244 res = ioctl(fd, SNDCTL_DSP_SETTRIGGER, &res);
1245 /* it may fail if we are in half duplex, never mind */
1246 return 0;
1247}

1249URP_CHANNEL_LOCAL int usbradio_text(struct ast_channel *c, const char *text)
1250{
1251 struct chan_usbradio_pvt *o = ast_channel_tech_pvt(c);
1252 char cmd[16], pwr;
1253 int cnt, i, j;
1254 double tx, rx;
1255
1256#define STR_SZ 15 /* Size of text strings */
1257 char rxs[STR_SZ + 1], txs[STR_SZ + 1], txpl[STR_SZ + 1], rxpl[STR_SZ + 1];
1258 if (!o) {
1259 return -1;
1260 }
1261
1262#ifdef HAVE_SYS_IO
1263 if (haspp == 2) {
1264 ioperm(pbase, 2, 1);
1265 }
1266#endif
1267
1268 cmd[0] = rxs[0] = txs[0] = rxpl[0] = txpl[0] = pwr = '\0';
1269
1270 /* print received messages */
1271 ast_debug(3, "Channel %s: Console Received usbradio text %s >>\n", o->name, text);
1272
1273 cnt = sscanf(text, "%15s " S_FMT(STR_SZ) S_FMT(STR_SZ) S_FMT(STR_SZ) S_FMT(STR_SZ) "%c",
1274 cmd, rxs, txs, rxpl, txpl, &pwr);
1275
1276 /* set channel on parallel port */
1277 if (strcmp(cmd, "SETCHAN") == 0) {
1278 u8 chan;
1279 chan = strtod(rxs, NULL);
1280 usbradioplus_set_channel(chan);
1281 ast_debug(3, "Channel %s: SETCHAN cmd: %s chan: %i\n", o->name, text, chan);
1282 return 0;
1283 }
1284
1285 /* set receive CTCSS */
1286 if (strcmp(cmd, "RXCTCSS") == 0) {
1287 u8 x;
1288 x = strtod(rxs, NULL);
1289 o->rxctcssoverride = !x;
1290 ast_debug(3, "Channel %s: RXCTCSS cmd: %s\n", o->name, text);
1291 return 0;
1292 }
1293
1294 /* set transmit CTCSS (app_rpt itxctcss -> TXCTCSS 0/1) */
1295 if (strcmp(cmd, "TXCTCSS") == 0) {
1296 if (cnt < 2 || (strcmp(rxs, "0") && strcmp(rxs, "1"))) {
1297 ast_log(LOG_WARNING, "Channel %s: Invalid TXCTCSS command: %s\n", o->name,
1298 text);
1299 return 0;
1300 }
1301 if (o->radio)
1302 o->radio->b.txCtcssOff = rxs[0] != '1';
1303 ast_debug(3, "Channel %s: TXCTCSS cmd: %s\n", o->name, text);
1304 return 0;
1305 }
1306
1307 /* GPIO command */
1308 if (!strncmp(text, "GPIO", 4)) {
1309 cnt = sscanf(text, "%15s " N_FMT(d) " " N_FMT(d), cmd, &i, &j);
1310 if (cnt < 3) {
1311 return 0;
1312 }
1313 if ((i < 1) || (i > GPIO_PINCOUNT)) {
1314 return 0;
1315 }
1316 i--;
1317 /* skip if not valid */
1318 if (!(o->valid_gpios & (1 << i))) {
1319 return 0;
1320 }
1321 ast_mutex_lock(&o->usblock);
1322 if (j > 1) { /* if to request pulse-age */
1323 o->hid_gpio_pulsetimer[i] = j - 1;
1324 } else {
1325 /* clear pulsetimer, if in the middle of running */
1326 o->hid_gpio_pulsetimer[i] = 0;
1327 o->hid_gpio_val &= ~(1 << i);
1328 if (j) {
1329 o->hid_gpio_val |= 1 << i;
1330 }
1331 o->gpio_set = 1;
1332 }
1333 ast_mutex_unlock(&o->usblock);
1334 kickptt(o);
1335 return 0;
1336 }
1337
1338 /* Parallel port command */
1339 if (!strncmp(text, "PP", 2)) {
1340 cnt = sscanf(text, "%15s " N_FMT(d) " " N_FMT(d), cmd, &i, &j);
1341 if (cnt < 3) {
1342 return 0;
1343 }
1344 if ((i < 2) || (i > 9)) {
1345 return 0;
1346 }
1347 ast_mutex_lock(&pp_lock);
1348 if (j > 1) { /* if to request pulse-age */
1349 pp_pulsetimer[i] = j - 1;
1350 } else {
1351 /* clear pulsetimer, if in the middle of running */
1352 pp_pulsetimer[i] = 0;
1353 pp_val &= ~(1 << (i - 2));
1354 if (j) {
1355 pp_val |= 1 << (i - 2);
1356 }
1357 ast_radio_ppwrite(haspp, ppfd, pbase, pport, pp_val);
1358 }
1359 ast_mutex_unlock(&pp_lock);
1360 return 0;
1361 }
1362
1363 if (cnt < 6) {
1364 ast_log(LOG_ERROR, "Channel %s: Cannot parse usbradio text: %s\n", o->name, text);
1365 return 0;
1366 } else {
1367 ast_debug(3, "Channel %s: << %s %s %s %s %s %c >> \n", o->name, cmd, rxs, txs, rxpl,
1368 txpl, pwr);
1369 }
1370
1371 /* set frequency command */
1372 if (strcmp(cmd, "SETFREQ") == 0) {
1373 ast_debug(3, "Channel %s: SETFREQ cmd: %s\n", o->name, text);
1374 tx = strtod(txs, NULL);
1375 rx = strtod(rxs, NULL);
1376 o->set_txfreq = round(tx * (double)1000000);
1377 o->set_rxfreq = round(rx * (double)1000000);
1378 o->set_txpower = (pwr == 'H');
1379 ast_copy_string(o->set_rxctcssfreqs, rxpl, sizeof(o->set_rxctcssfreqs));
1380 ast_copy_string(o->set_txctcssfreqs, txpl, sizeof(o->set_txctcssfreqs));
1381
1382 o->remoted = 1;
1383 radio_config(o);
1384 return 0;
1385 }
1386 ast_log(LOG_ERROR, "Channel %s: Cannot parse usbradio cmd: %s\n", o->name, text);
1387 return 0;
1388}

1419URP_CHANNEL_LOCAL int usbradio_write(struct ast_channel *c, struct ast_frame *f)
1420{
1421 struct chan_usbradio_pvt *o = ast_channel_tech_pvt(c);
1422
1423 if (!o->hasusb) {
1424 return 0;
1425 }
1426 if (o->sounddev < 0) {
1427 setformat(o, O_RDWR);
1428 }
1429 if (o->sounddev < 0) {
1430 return 0; /* not fatal */
1431 }
1432 /*
1433 * we could receive a block which is not a multiple of our
1434 * FRAME_SIZE, so buffer it locally and write to the device
1435 * in FRAME_SIZE chunks.
1436 * Keep the residue stored for future use.
1437 */
1438
1439#if DEBUG_CAPTURES == 1
1440 /* Write input data to a file.
1441 * Left channel has the audio, right channel shows txkeyed
1442 */
1443 if (ftxcapraw && o->txcapraw) {
1444 short i, tbuff[f->datalen];
1445 memset(tbuff, 0, sizeof(tbuff));
1446 for (i = 0; i < f->datalen; i += 2) {
1447 tbuff[i] = ((short *)(f->data.ptr))[i / 2];
1448 tbuff[i + 1] = o->txkeyed * M_Q13;
1449 }
1450 fwrite(tbuff, 2, f->datalen, ftxcapraw);
1451 }
1452#endif
1453
1454 /* Preserve app_rpt's continuous stream, including idle silence, so PTT
1455 * transitions do not interrupt clock recovery. Echo owns the input while playing. */
1456 if (!o->echoing) {
1457 usbradioplus_queue_program(o, f->data.ptr, f->datalen / sizeof(short));
1458 }
1459
1460 return 0;
1461}

1463URP_CHANNEL_LOCAL struct ast_frame *usbradio_read(struct ast_channel *c)
1464{
1465 int res, oldpttout;
1466 int cd, sd;
1467 int was_rxkeyed;
1468 struct chan_usbradio_pvt *o = ast_channel_tech_pvt(c);
1469 struct ast_frame *f = &o->read_f, *f1;
1470 time_t now;
1471
1472 /* check to the if the hid thread is still processing */
1473 if (o->lasthidtime) {
1474 ast_radio_time(&now);
1475 if ((now - o->lasthidtime) > 3) {
1476 ast_log(LOG_ERROR,
1477 "Channel %s: HID process has died or is not responding.\n",
1478 o->name);
1479 return NULL;
1480 }
1481 }
1482 /* Set frame defaults */
1483 memset(f, 0, sizeof(struct ast_frame));
1484 f->frametype = AST_FRAME_NULL;
1485 f->src = __PRETTY_FUNCTION__;
1486
1487 /* if USB device not ready, just return NULL frame */
1488 if (!o->hasusb) {
1489 if (o->rxkeyed) {
1490 struct ast_frame wf = {
1491 .frametype = AST_FRAME_CONTROL,
1492 .subclass.integer = AST_CONTROL_RADIO_UNKEY,
1493 .src = __PRETTY_FUNCTION__,
1494 };
1495
1496 o->lastrx = 0;
1497 o->rxkeyed = 0;
1498 ast_queue_frame(o->owner, &wf);
1499 if (o->duplex3 && o->duplex3mode == DUPLEX3_MODE_HARDWARE) {
1500 ast_radio_setamixer(o->devicenum, MIXER_PARAM_MIC_PLAYBACK_SW, 0,
1501 0);
1502 }
1503 }
1504 return &ast_null_frame;
1505 }
1506
1507 /* If we have stopped echoing, clear the echo queue */
1508 if (!o->echomode) {
1509 ast_mutex_lock(&o->echolock);
1510 o->echoing = 0;
1511 while (o->echoq.q_forw != &o->echoq) {
1512 struct qelem *q = o->echoq.q_forw;
1513 remque(q);
1514 ast_free(q);
1515 }
1516 ast_mutex_unlock(&o->echolock);
1517 }
1518
1519 /* If we are in echomode and we have stopped receiving audio
1520 * queue up the packets we have stored in the echo queue
1521 * for playback.
1522 */
1523 if (o->echomode && !usbradioplus_native_echo(o) && (!o->rxkeyed)) {
1524 ast_mutex_lock(&o->echolock);
1525 /* if there is something in the queue */
1526 if (o->echoq.q_forw != &o->echoq) {
1527 struct usbecho *u = (struct usbecho *)o->echoq.q_forw;
1528 remque((struct qelem *)u);
1529 usbradioplus_queue_program(o, u->data, FRAME_SIZE);
1530 ast_free(u);
1531 o->echoing = 1;
1532 } else {
1533 o->echoing = 0;
1534 }
1535 ast_mutex_unlock(&o->echolock);
1536 }
1537
1538 /* Read audio data from the USB sound device.
1539 * Sound data will arrive at 48000 samples per second
1540 * in stereo format.
1541 */
1542 res = read(o->sounddev, o->usbradio_read_buf + o->readpos,
1543 sizeof(o->usbradio_read_buf) - o->readpos);
1544 if (res < 0) { /* Audio data not ready, return a NULL frame */
1545 if (errno != EAGAIN) {
1546 o->readerrs = 0;
1547 o->hasusb = 0;
1548 return &ast_null_frame;
1549 }
1550 if (o->readerrs++ > READERR_THRESHOLD) {
1551 ast_log(LOG_ERROR, "Stuck USB read channel [%s], un-sticking it!\n",
1552 o->name);
1553 o->readerrs = 0;
1554 o->hasusb = 0;
1555 return &ast_null_frame;
1556 }
1557 if (o->readerrs == 1) {
1558 ast_log(LOG_WARNING, "Possibly stuck USB read channel. [%s]\n", o->name);
1559 }
1560 return &ast_null_frame;
1561 }
1562
1563#if DEBUG_CAPTURES == 1
1564 if (o->rxcapraw && frxcapraw) {
1565 fwrite(o->usbradio_read_buf + o->readpos, 1, res, frxcapraw);
1566 }
1567#endif
1568
1569 if (o->readerrs) {
1570 ast_log(LOG_WARNING, "USB read channel [%s] was not stuck.\n", o->name);
1571 }
1572
1573 o->readerrs = 0;
1574 o->readpos += res;
1575 if ((size_t)o->readpos < sizeof(o->usbradio_read_buf)) { /* not enough samples */
1576 return &ast_null_frame;
1577 }
1578
1579 /* Check for ADC clipping and input audio statistics before any filtering is done.
1580 * FRAME_SIZE define refers to 8Ksps mono which is 160 samples per 20mS USB frame.
1581 * ast_radio_check_audio() takes the read buffer as received (48K stereo),
1582 * extracts the mono 48K channel, checks amplitude and distortion characteristics,
1583 * and returns true if clipping was detected.
1584 */
1585 if (ast_radio_check_audio((short *)o->usbradio_read_buf, &o->rxaudiostats,
1586 12 * FRAME_SIZE)) {
1587 if (o->clipledgpio) {
1588 /* Set Clip LED GPIO pulsetimer if not already set */
1589 if (!o->hid_gpio_pulsetimer[o->clipledgpio - 1]) {
1590 o->hid_gpio_pulsetimer[o->clipledgpio - 1] = CLIP_LED_HOLD_TIME_MS;
1591 }
1592 }
1593 }
1594
1595 was_rxkeyed = o->rxkeyed;
1596 /* Only app_rpt and tuning own PTT. */
1597 if (o->txkeyed || o->txtestkey) {
1598 if (!o->radio->txPttIn) {
1599 o->radio->txPttIn = 1;
1600 ast_debug(3, "Channel %s: txPttIn = %i.\n", o->name, o->radio->txPttIn);
1601 }
1602 } else if (o->radio->txPttIn) {
1603 o->radio->txPttIn = 0;
1604 ast_debug(3, "Channel %s: txPttIn = %i.\n", o->name, o->radio->txPttIn);
1605 }
1606 oldpttout = o->radio->txPttOut;
1607
1608 usbradioplus_prepare_squelch_audio(o);
1609 urp_radio_process(o->radio, (i16 *)o->plus_squelch_native,
1610 (i16 *)(o->usbradio_read_buf_8k + AST_FRIENDLY_OFFSET),
1611 (i16 *)(o->usbradio_write_buf));
1612 usbradioplus_refresh_ctcss_decode(o);
1613 usbradioplus_native_tick(o);
1614
1615 if (oldpttout != o->radio->txPttOut) {
1616 usbradioplus_program_radio(o);
1617 ast_debug(3, "Channel %s: txPttOut = %i.\n", o->name, o->radio->txPttOut);
1618 kickptt(o);
1619 }
1620
1621#if DEBUG_CAPTURES == 1 && URP_RADIO_DEBUG == 1
1622 if (o->txcap2 && ftxcaptrace) {
1623 fwrite((o->radio->ptxDebug), 1, FRAME_SIZE * 2 * 16, ftxcaptrace);
1624 }
1625#endif
1626
1627 /* Write the received audio to the sound card */
1628 soundcard_writeframe(o, (short *)o->usbradio_write_buf);
1629
1630#if DEBUG_CAPTURES == 1 && URP_RADIO_DEBUG == 1
1631 if (frxcaptrace && o->rxcap2 && o->radioactive) {
1632 fwrite((o->radio->prxDebug), 1, FRAME_SIZE * 2 * 16, frxcaptrace);
1633 }
1634#endif
1635
1636 /* Check for carrier detect - COR active */
1637 {
1638 enum radio_carrier_detect rxcdtype = effective_rxcdtype(o);
1639 cd = 0;
1640 if (rxcdtype == CD_HID && (o->radio->rxExtCarrierDetect != o->rxhidsq)) {
1641 o->radio->rxExtCarrierDetect = o->rxhidsq;
1642 }
1643
1644 if (rxcdtype == CD_HID_INVERT && (o->radio->rxExtCarrierDetect == o->rxhidsq)) {
1645 o->radio->rxExtCarrierDetect = !o->rxhidsq;
1646 }
1647
1648 if (usbradioplus_carrier_detected(o, rxcdtype)) {
1649 if (!o->radio->txPttOut || o->radioduplex) {
1650 cd = 1;
1651 }
1652 } else {
1653 cd = 0;
1654 }
1655
1656 if (cd != o->rxcarrierdetect) {
1657 o->rxcarrierdetect = cd;
1658 ast_debug(3, "Channel %s: rxcarrierdetect = %i.\n", o->name, cd);
1659 }
1660 o->rx_cos_active = cd;
1661 }
1662
1663 /* Check for SD - CTCSS active. */
1664 if (usbradioplus_ctcss_detected(o)) {
1665 sd = 1;
1666 } else {
1667 sd = 0;
1668 }
1669 if (o->rxsdtype == SD_HID) {
1670 sd = o->rxhidctcss;
1671 } else if (o->rxsdtype == SD_HID_INVERT) {
1672 sd = !o->rxhidctcss;
1673 } else if (o->rxsdtype == SD_PP) {
1674 sd = o->rxppctcss;
1675 } else if (o->rxsdtype == SD_PP_INVERT) {
1676 sd = !o->rxppctcss;
1677 }
1678 /* See if we are overriding CTCSS to active */
1679 if (o->rxctcssoverride) {
1680 sd = 1;
1681 }
1682 o->rx_ctcss_active = sd;
1683
1684 /* Special case where cd and sd have been configured for no */
1685 if (effective_rxcdtype(o) == CD_IGNORE && o->rxsdtype == SD_IGNORE) {
1686 cd = 0;
1687 sd = 0;
1688 }
1689
1690 /* Timer for how long TX has been unkeyed - used with txoffdelay */
1691 if (o->txoffdelay) {
1692 if (o->txkeyed == 1) {
1693 o->txoffcnt = 0; /* If keyed, set this to zero. */
1694 } else {
1695 o->txoffcnt++;
1696 if (o->txoffcnt > MS_TO_FRAMES(TX_OFF_DELAY_MAX)) {
1697 o->txoffcnt = MS_TO_FRAMES(TX_OFF_DELAY_MAX); /* Limit the count */
1698 }
1699 }
1700 }
1701
1702 /* Check conditions and set receiver active */
1703 if (cd && sd) {
1704 if (!o->rxkeyed) {
1705 ast_debug(3, "Channel %s: o->rxkeyed = 1.\n", o->name);
1706 }
1707 if (o->rxkeyed ||
1708 ((o->txoffcnt >= o->txoffdelay) && (o->rxoncnt >= o->rxondelay))) {
1709 o->rxkeyed = 1;
1710 } else {
1711 o->rxoncnt++;
1712 }
1713 } else {
1714 if (o->rxkeyed) {
1715 ast_debug(3, "Channel %s: o->rxkeyed = 0.\n", o->name);
1716 }
1717 o->rxkeyed = 0;
1718 o->rxoncnt = 0;
1719 }
1720 if (o->echomode && usbradioplus_native_echo(o)) {
1721 usbradioplus_parrot_rx_transition(o, was_rxkeyed);
1722 }
1723
1724 /* If we are in echomode and receiving audio, store
1725 * it in the echo queue for later playback.
1726 */
1727 if (o->echomode && !usbradioplus_native_echo(o) && o->rxkeyed && (!o->echoing)) {
1728 register int x;
1729 struct usbecho *u;
1730
1731 ast_mutex_lock(&o->echolock);
1732 x = 0;
1733 /* get count of frames */
1734 for (u = (struct usbecho *)o->echoq.q_forw; u != (struct usbecho *)&o->echoq;
1735 u = (struct usbecho *)u->q_forw)
1736 x++;
1737 if (x < o->echomax) {
1738 u = ast_calloc(1, sizeof(struct usbecho));
1739 if (u) {
1740 memcpy(u->data, (o->usbradio_read_buf_8k + AST_FRIENDLY_OFFSET),
1741 FRAME_SIZE * 2);
1742 insque((struct qelem *)u, o->echoq.q_back);
1743 }
1744 }
1745 ast_mutex_unlock(&o->echolock);
1746 }
1747
1748 /* Send a message to indicate rx signal detect conditions */
1749 if (o->lastrx && (!o->rxkeyed)) {
1750 struct ast_frame wf = {
1751 .frametype = AST_FRAME_CONTROL,
1752 .subclass.integer = AST_CONTROL_RADIO_UNKEY,
1753 .src = __PRETTY_FUNCTION__,
1754 };
1755
1756 o->lastrx = 0;
1757 ast_queue_frame(o->owner, &wf);
1758 if (o->duplex3 && o->duplex3mode == DUPLEX3_MODE_HARDWARE) {
1759 ast_radio_setamixer(o->devicenum, MIXER_PARAM_MIC_PLAYBACK_SW, 0, 0);
1760 }
1761 } else if ((!o->lastrx) && (o->rxkeyed)) {
1762 struct ast_frame wf = {
1763 .frametype = AST_FRAME_CONTROL,
1764 .subclass.integer = AST_CONTROL_RADIO_KEY,
1765 .src = __PRETTY_FUNCTION__,
1766 };
1767
1768 o->lastrx = 1;
1769 if (o->rxctcssdecode) {
1770 wf.data.ptr = o->rxctcssfreq;
1771 wf.datalen = strlen(o->rxctcssfreq) + 1;
1772 ast_debug(7, "Radio Key - CTCSS frequency=%s.\n", o->rxctcssfreq);
1773 }
1774 ast_queue_frame(o->owner, &wf);
1775 o->count_rssi_update = 1;
1776 if (o->duplex3 && o->duplex3mode == DUPLEX3_MODE_HARDWARE) {
1777 ast_radio_setamixer(o->devicenum, MIXER_PARAM_MIC_PLAYBACK_SW, 1, 0);
1778 }
1779 }
1780
1781 /* reset read pointer for next frame */
1782 o->readpos = AST_FRIENDLY_OFFSET;
1783 /* Do not return the frame if the channel is not up */
1784 if (ast_channel_state(c) != AST_STATE_UP) {
1785 return &ast_null_frame;
1786 }
1787 /* ok we can build and deliver the frame to the caller */
1788 f->frametype = AST_FRAME_VOICE;
1789 f->subclass.format = ast_format_slin;
1790 f->offset = AST_FRIENDLY_OFFSET;
1791 f->samples = o->plus_app_rpt_samples;
1792 f->datalen = f->samples * sizeof(short);
1793 f->data.ptr = o->usbradio_read_buf_8k + AST_FRIENDLY_OFFSET;
1794 f->src = __PRETTY_FUNCTION__;
1795 if (!o->rxkeyed) {
1796 memset(f->data.ptr, 0, f->datalen);
1797 }
1798 /* Process the audio to see if contains DTMF */
1799 if (o->usedtmf && o->dsp) {
1800 f1 = ast_dsp_process(c, o->dsp, f);
1801 if ((f1->frametype == AST_FRAME_DTMF_END) ||
1802 (f1->frametype == AST_FRAME_DTMF_BEGIN)) {
1803 if ((f1->subclass.integer == 'm') || (f1->subclass.integer == 'u')) {
1804 f1->frametype = AST_FRAME_NULL;
1805 f1->subclass.integer = 0;
1806 return f1;
1807 }
1808 if (f1->frametype == AST_FRAME_DTMF_END) {
1809 f1->len = ast_tvdiff_ms(ast_radio_tvnow(), o->tonetime);
1810 if (option_verbose) {
1811 ast_log(LOG_NOTICE,
1812 "Channel %s: Got DTMF char %c duration %ld ms\n",
1813 o->name, f1->subclass.integer, f1->len);
1814 }
1815 o->toneflag = 0;
1816 } else {
1817 if (o->toneflag) {
1818 ast_frfree(f1);
1819 f1 = NULL;
1820 } else {
1821 o->tonetime = ast_radio_tvnow();
1822 o->toneflag = 1;
1823 }
1824 }
1825 if (f1) {
1826 return f1;
1827 }
1828 }
1829 }
1830
1831 if (o->radio->b.txCtcssReady) {
1832 struct ast_frame wf = {
1833 .frametype = AST_FRAME_TEXT,
1834 .src = __PRETTY_FUNCTION__,
1835 };
1836 char msg[32];
1837
1838 snprintf(msg, sizeof(msg), "cstx=%.26s", o->radio->txctcssfreq);
1839 wf.data.ptr = msg;
1840 wf.datalen = strlen(msg) + 1;
1841 ast_queue_frame(o->owner, &wf);
1842
1843 ast_debug(3, "Channel %s: got b.txCtcssReady %s.\n", o->name,
1844 o->radio->txctcssfreq);
1845 o->radio->b.txCtcssReady = 0;
1846 }
1847 /* report channel rssi */
1848 if (o->sendvoter && o->count_rssi_update && o->rxkeyed) {
1849 if (--o->count_rssi_update <= 0) {
1850 struct ast_frame wf = {
1851 .frametype = AST_FRAME_TEXT,
1852 .src = __PRETTY_FUNCTION__,
1853 };
1854 char msg[32];
1855
1856 snprintf(msg, sizeof(msg), "R %i",
1857 ((32767 - o->radio->rxRssi) * 1000) / 32767);
1858 wf.data.ptr = msg;
1859 wf.datalen = strlen(msg) + 1;
1860 ast_queue_frame(o->owner, &wf);
1861
1862 o->count_rssi_update = 10;
1863 ast_debug(4, "Channel %s: Count_rssi_update %i\n", o->name,
1864 ((32767 - o->radio->rxRssi) * 1000 / 32767));
1865 }
1866 }
1867
1868 return f;
1869}

2023{

2025 const int settingstart = 2;
2026 const int maxtries = 12;
2027

2029 int tolerance = 2750;

2031 unsigned int rms = 0;
2032 double peak_dbfs, rms_dbfs;

2091 setting++;
2092 } else if (meas > (target + tolerance)) {
2093 setting--;

2095 break;
2096 }
2097
2098 if (setting < settingmin) {
2099 setting = settingmin;
2100 } else if (setting > settingmax) {
2101 setting = settingmax;
2102 }
2103 tries++;
2104 }

2110 o->radio->spsRx->amax = o->radio->spsRx->amin = 0;

2112 o->radio->b.tuning = 0;
2113 return;
2114 }

2116
2117 /* Measure RSSI */
2118 o->radio->spsRx->discfactor = tmpdiscfactor;
2119 o->radio->spsRx->discounteru = o->radio->spsRx->discounterl = 0;
2120 o->radio->spsRx->amax = o->radio->spsRx->amin = 0;

2123 return;
2124 }
2125
2126 ast_cli(fd,

2128 tries, ((setting * 1000) + (o->micmax / 2)) / o->micmax, meas, peak_dbfs, rms,
2129 rms_dbfs, measnoise);
2130

2144 "receiver.\n");
2145 ast_cli(fd, "WARNING: Rx input point may be de-emphasized and not "

2147 ast_cli(fd, " usbradioplus.conf setting of "

2161 } else {

2165 "loose.\n",
2166 normRssi, o->rxsquelchadj);
2167 ast_cli(fd,
2168 " Use 'radio tune rxsquelch' to adjust.\n");
2169 }
2170 }
2171 }
2172 }
2173 o->radio->b.tuning = 0;
