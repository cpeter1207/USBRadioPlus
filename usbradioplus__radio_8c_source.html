144/*
145 take source string allocate and copy
146 copy is modified, delimiters are replaced with zeros to mark
147 end of string
148 count set pointers
149 string_parse( char *src, char *dest, char **sub)
150*/
151i16 string_parse(const char *src, char **dest, char ***ptrs)
152{
153 char *p, *pd;
154 char *ptstr[1000];
155 i16 i, slen, numsub;
156
157 TRACEJ(2, "string_parse(%s)\n", src);
158
159 slen = strlen(src);
160 TRACEJ(2, " source len = %i\n", slen);
161
162 pd = *dest;
163 if (pd) {
164 ast_free(pd);
165 *dest = NULL;
166 }
167 pd = ast_calloc(slen + 1, 1);
168 if (!pd) {
169 return -1;
170 }
171 memcpy(pd, src, slen);
172 *dest = pd;
173
174 p = 0;
175 numsub = 0;
176 for (i = 0; i < slen + 1; i++) {
177 TRACEJ(5, " pd[%i] = %c\n", i, pd[i]);
178
179 if (p == 0 && pd[i] != ',' && pd[i] != ' ') {
180 p = &(pd[i]);
181 } else if (pd[i] == ',' || pd[i] == 0) {
182 ptstr[numsub] = p;
183 pd[i] = 0;
184 p = 0;
185 numsub++;
186 }
187 }
188
189 for (i = 0; i < numsub; i++) {
190 TRACEJ(5, " ptstr[%i] = %p %s\n", i, ptstr[i], ptstr[i]);
191 }
192
193 if (*ptrs) {
194 ast_free((void *)*ptrs);
195 *ptrs = NULL;
196 }
197 *ptrs = ast_calloc(numsub, sizeof(char *));
198 if (!*ptrs) {
199 ast_free(*dest);
200 *dest = NULL;
201 return -1;
202 }
203 for (i = 0; i < numsub; i++) {
204 (*ptrs)[i] = ptstr[i];
205 TRACEJ(5, " %i = %s\n", i, (*ptrs)[i]);
206 }
207 TRACEJ(5, "string_parse()=%i\n\n", numsub);
208
209 return numsub;
210}

212/*
213 the parent program defines
214 pRxCodeSrc and pTxCodeSrc string pointers to the list of codes
215 pTxCodeDefault the default Tx Code.
216
217*/
218i16 urp_radio_parse_codes(urp_radio_state *pChan)
219{
220 i16 i, ii, hit, ti;
221 char *p;
222 float f, maxctcsstxfreq;
223
224 urp_radio_stage *pSps;
225 i16 maxctcssindex;
226
227 TRACEF(1, "urp_radio_parse_codes(%i)\n", 0);
228 TRACEF(1, "pChan->pRxCodeSrc %s \n", pChan->pRxCodeSrc);
229 TRACEF(1, "pChan->pTxCodeSrc %s \n", pChan->pTxCodeSrc);
230 TRACEF(1, "pChan->pTxCodeDefault %s \n", pChan->pTxCodeDefault);
231
232 maxctcssindex = CTCSS_NULL;
233 maxctcsstxfreq = CTCSS_NULL;
234 pChan->txctcssdefault_index = CTCSS_NULL;
235 pChan->txctcssdefault_value = CTCSS_NULL;
236
237 pChan->b.ctcssRxEnable = pChan->b.ctcssTxEnable = 0;
238 pChan->b.dcsRxEnable = pChan->b.dcsTxEnable = 0;
239 pChan->b.lmrRxEnable = pChan->b.lmrTxEnable = 0;
240 pChan->b.mdcRxEnable = pChan->b.mdcTxEnable = 0;
241 pChan->b.dstRxEnable = pChan->b.dstTxEnable = 0;
242 pChan->b.p25RxEnable = pChan->b.p25TxEnable = 0;
243
244 TRACEF(1, "urp_radio_parse_codes(%i) 05\n", 0);
245
246 pChan->numrxcodes =
247 string_parse(pChan->pRxCodeSrc, &(pChan->pRxCodeStr), &(pChan->pRxCode));
248 if (pChan->numrxcodes < 0) {
249 return 1;
250 }
251 pChan->numtxcodes =
252 string_parse(pChan->pTxCodeSrc, &(pChan->pTxCodeStr), &(pChan->pTxCode));
253 if (pChan->numtxcodes < 0) {
254 return 1;
255 }
256
257 if (pChan->numrxcodes != pChan->numtxcodes) {
258 ast_log(LOG_ERROR, "numrxcodes != numtxcodes \n");
259 }
260 pChan->rxCtcss->enabled = 0;
261 pChan->rxCtcss->gain = 1 * M_Q8;
262 pChan->rxCtcss->limit = 8192;
263 pChan->rxCtcss->input = pChan->pRxLsdLimit;
264 pChan->rxCtcss->decode = CTCSS_NULL;
265
266 pChan->rxCtcss->testIndex = 3;
267
268 pChan->rxctcssfreq[0] = 0; /* decode now CTCSS_RXONLY */
269
270 for (i = 0; i < CTCSS_NUM_CODES; i++) {
271 pChan->rxctcss[i] = 0;
272 pChan->txctcss[i] = 0;
273 pChan->rxCtcssMap[i] = CTCSS_NULL;
274 }
275
276 TRACEF(1, "urp_radio_parse_codes(%i) 10\n", 0);
277
278 /* Do Receive Codes String */
279 for (i = 0; i < pChan->numrxcodes; i++) {
280 p = pChan->pStr = pChan->pRxCode[i];
281
282 {
283 i16 rx_index, tx_index;
284 float frequency;
285
286 sscanf(p, N_FMT(frequency), &frequency);
287 rx_index = urp_ctcss_frequency_index(frequency);
288 if (rx_index == CTCSS_NULL) {
289 ast_log(LOG_ERROR,
290 "Invalid RX CTCSS code detected and ignored. %i %s\n", i,
291 pChan->pRxCode[i]);
292
293 } else if (rx_index > maxctcssindex) {
294 maxctcssindex = rx_index;
295 }
296
297 if (i < pChan->numtxcodes) { /* more rx codes than tx codes */
298 sscanf(pChan->pTxCode[i], N_FMT(frequency), &frequency);
299 tx_index = urp_ctcss_frequency_index(frequency);
300 if (tx_index == CTCSS_NULL) {
301 if (frequency != 0.0) {
302 frequency = -1.0; /* tone freq not valid */
303 ast_log(LOG_ERROR,
304 "Invalid TX CTCSS code detected and "
305 "ignored. %i %s\n",
306 i, pChan->pTxCode[i]);
307 }
308 } else if (frequency > maxctcsstxfreq) {
309 maxctcsstxfreq = frequency;
310 }
311 } else {
312 tx_index = CTCSS_NULL;
313 frequency = -1.0; /* tone freq not provided */
314 ast_log(LOG_ERROR, "Invalid CTCSS configuration. Number of rx "
315 "codes > number of tx codes\n");
316 }
317
318 if (rx_index > CTCSS_NULL && tx_index > CTCSS_NULL) {
319 pChan->b.ctcssRxEnable = 1;
320 pChan->b.ctcssTxEnable = 1;
321 pChan->rxCtcssMap[rx_index] = tx_index;
322 pChan->numrxctcssfreqs++;
323 TRACEF(1, "pChan->rxctcss[%i]=%s pChan->rxCtcssMap[%i]=%i\n", i,
324 pChan->rxctcss[i], rx_index, tx_index);
325 } else if (rx_index > CTCSS_NULL && frequency == 0) {
326 pChan->b.ctcssRxEnable = 1;
327 pChan->rxCtcssMap[rx_index] = CTCSS_RXONLY;
328 pChan->numrxctcssfreqs++;
329 TRACEF(1,
330 "pChan->rxctcss[%i]=%s pChan->rxCtcssMap[%i]=%i RXONLY\n",
331 i, pChan->rxctcss[i], rx_index, tx_index);
332 } else {
333 i16 clear_index;
334
335 pChan->numrxctcssfreqs = 0;
336 ast_log(LOG_ERROR,
337 "Invalid CTCSS configuration. CTCSS has been disabled\n");
338 for (clear_index = 0; clear_index < CTCSS_NUM_CODES;
339 clear_index++) {
340 pChan->rxCtcssMap[clear_index] = CTCSS_NULL;
341 }
342 }
343 }
344 }
345
346 TRACEF(1, "urp_radio_parse_codes() CTCSS Init Struct %i %i\n", pChan->b.ctcssRxEnable,
347 pChan->b.ctcssTxEnable);
348 if (pChan->b.ctcssRxEnable) {
349 pChan->rxHpfEnable = 1;
350 pChan->spsRxLsdNrz->enabled = pChan->rxCenterSlicerEnable = 1;
351 pChan->rxCtcssDecodeEnable = 1;
352 pChan->rxCtcss->enabled = 1;
353 } else {
354 pChan->rxHpfEnable = 1;
355 pChan->spsRxLsdNrz->enabled = pChan->rxCenterSlicerEnable = 0;
356 pChan->rxCtcssDecodeEnable = 0;
357 pChan->rxCtcss->enabled = 0;
358 }
359
360 TRACEF(1, "urp_radio_parse_codes() CTCSS Init Decoders \n");
361 for (i = 0; i < CTCSS_NUM_CODES; i++) {
362 urp_ctcss_tone_detector *ptdet;
363 ptdet = &(pChan->rxCtcss->tdet[i]);
364 ptdet->counterFactor = coef_ctcss_div[i];
365 ptdet->state = 1;
366 ptdet->setpt = (M_Q15 * 0.041); /* 0.069 */
367 ptdet->hyst = (M_Q15 * 0.0130);
368 ptdet->binFactor = (M_Q15 * 0.135); /* was 0.140 */
369 ptdet->fudgeFactor = 8;
370 }
371
372 /* DEFAULT TX CODE */
373 TRACEF(1, "urp_radio_parse_codes() Default Tx Code %s \n", pChan->pTxCodeDefault);
374 pChan->txcodedefaultsmode = SMODE_NULL;
375 p = pChan->pStr = pChan->pTxCodeDefault;
376
377 {
378 sscanf(p, N_FMT(f), &f);
379 ti = urp_ctcss_frequency_index(f);
380 if (ti == CTCSS_NULL) {
381 ast_log(LOG_ERROR,
382 "Invalid default TX CTCSS code detected and ignored. %s\n",
383 pChan->pTxCodeDefault);
384 } else if (f > maxctcsstxfreq) {
385 maxctcsstxfreq = f;
386 }
387
388 if (ti > CTCSS_NULL) {
389 pChan->b.ctcssTxEnable = 1;
390 pChan->txctcssdefault_index = ti;
391 pChan->txctcssdefault_value = f;
392 pChan->txCtcssFreq10 = f * 10;
393 pChan->txcodedefaultsmode = SMODE_CTCSS;
394 TRACEF(1, "urp_radio_parse_codes() Tx Default CTCSS = %s %i %f\n", p, ti,
395 f);
396 }
397 }
398
399 /* Native CTCSS uses the same legacy filter selection for level matching. */
400 TRACEF(1, "urp_radio_parse_codes() Filter Config \n");
401 if (maxctcsstxfreq > 203.5) {
402 pChan->txCtcssFilter250 = 1;
403 TRACEF(1, "urp_radio_parse_codes() Tx Filter Freq High\n");
404 } else {
405 pChan->txCtcssFilter250 = 0;
406 TRACEF(1, "urp_radio_parse_codes() Tx Filter Freq Low\n");
407 }
408
409 /* CTCSS Rx Decoder Low Pass Filter */
410 hit = 0;
411 ii = urp_ctcss_frequency_index(203.5);
412 for (i = ii; i < CTCSS_NUM_CODES; i++) {
413 if (pChan->rxCtcssMap[i] > CTCSS_NULL) {
414 hit = 1;
415 }
416 }
417
418 pSps = pChan->spsRxLsd;
419 ast_free(pSps->x);
420 pSps->x = NULL;
421
422 if (hit) {
423 pSps->ncoef = taps_fir_lpf_250_9_66;
424 pSps->size_coef = 2;
425 pSps->coef = (void *)coef_fir_lpf_250_9_66;
426 pSps->nx = taps_fir_lpf_250_9_66;
427 pSps->size_x = 2;
428 pSps->x = ast_calloc(pSps->nx, pSps->size_x);
429 if (pSps->x == NULL) {
430 return 1;
431 }
432 pSps->calcAdjust = gain_fir_lpf_250_9_66;
433 TRACEF(1, "urp_radio_parse_codes() Rx Filter Freq High\n");
434 } else {
435 pSps->ncoef = taps_fir_lpf_215_9_88;
436 pSps->size_coef = 2;
437 pSps->coef = (void *)coef_fir_lpf_215_9_88;
438 pSps->nx = taps_fir_lpf_215_9_88;
439 pSps->size_x = 2;
440 pSps->x = ast_calloc(pSps->nx, pSps->size_x);
441 if (pSps->x == NULL) {
442 return 1;
443 }
444 pSps->calcAdjust = gain_fir_lpf_215_9_88;
445 TRACEF(1, "urp_radio_parse_codes() Rx Filter Freq Low\n");
446 }
447
448 if (pChan->b.ctcssRxEnable) {
449 pChan->rxCenterSlicerEnable = 1;
450 pSps->enabled = 1;
451 } else {
452 pChan->rxCenterSlicerEnable = 0;
453 pSps->enabled = 0;
454 }
455
456#if URP_RADIO_DEBUG == 1
457 TRACEF(2, "urp_radio_parse_codes() ctcssRxEnable = %i \n", pChan->b.ctcssRxEnable);
458 TRACEF(2, " ctcssTxEnable = %i \n", pChan->b.ctcssTxEnable);
459 TRACEF(2, " dcsRxEnable = %i \n", pChan->b.dcsRxEnable);
460 TRACEF(2, " lmrRxEnable = %i \n", pChan->b.lmrRxEnable);
461 TRACEF(2, " txcodedefaultsmode = %i \n", pChan->txcodedefaultsmode);
462 for (i = 0; i < CTCSS_NUM_CODES; i++) {
463 TRACEF(2, "rxCtcssMap[%i] = %i \n", i, pChan->rxCtcssMap[i]);
464 }
465#endif
466
467 TRACEF(1, "urp_radio_parse_codes(%i) end\n", 0);
468
469 return 0;
470}

487/*
488 urp_radio_receive_frontend
489 Takes a block of data and low pass filters it.
490 Determines the amplitude of high frequency noise for carrier detect.
491 Decimates input data to change the rate.
492*/
493i16 urp_radio_receive_frontend(urp_radio_stage *mySps)
494{
495
496#define DCgainBpfNoise 65536
497
498 i16 samples, nx, iOutput, *output, *noutput;
499 const i16 *input;
500 i16 *x;
501 i16 decimator, decimate, doNoise, fever, fev1;
502 i32 i, naccum, outputGain, calcAdjust;
503 i64 y, npwr;
504
505 TRACEJ(5, "urp_radio_receive_frontend()\n");
506
507 if (!mySps->enabled) {
508 return 1;
509 }
510
511 decimator = mySps->decimator;
512 decimate = mySps->decimate;
513
514 input = mySps->source;
515 output = mySps->sink;
516 noutput = mySps->parentChan->pRxNoise;
517 fever = mySps->parentChan->fever;
518
519 nx = mySps->nx;
520
521 calcAdjust = mySps->calcAdjust;
522 outputGain = mySps->outputGain;
523
524 samples = mySps->nSamples * decimate;
525 x = mySps->x;
526 iOutput = 0;
527 npwr = 0;
528
529 if (mySps->parentChan->rxCdType != CD_XPMR_VOX) {
530 doNoise = 1;
531 } else {
532 doNoise = 0;
533 }
534
535 if (fever) {
536 fev1 = (nx - 1) * 2;
537 } else {
538 fev1 = nx - 1;
539 }
540
541 for (i = 0; i < samples; i++) {
542 i16 n;
543
544 /* shift the old samples */
545 memmove(x + 1, x, fev1);
546 x[0] = input[i * 2];
547
548#if URP_RADIO_TRACE_FRONTEND == 1
549 y = 0;
550 for (n = 0; n < nx; n++) {
551 y += fir_rxlpf[mySps->parentChan->rxlpf].coefs[n] * x[n];
552 }
553
554 y = ((y / calcAdjust) * outputGain) / M_Q8;
555 input[i * 2] = y; /* debug output LowPass at 48KS/s */
556#endif
557
558 if (doNoise) {
559 /* calculate noise filter output */
560 naccum = 0;
561 if (mySps->parentChan->rxNoiseFilType == 0) {
562 for (n = 0; n < taps_fir_bpf_noise_1; n++) {
563 naccum += coef_fir_bpf_noise_1[n] * x[n];
564 }
565 naccum /= DCgainBpfNoise;
566 } else {
567 for (n = 0; n < taps_fir_bpf_noise_2; n++) {
568 naccum += coef_fir_bpf_noise_2[n] * x[n];
569 }
570 naccum /= gain_fir_bpf_noise_2;
571 }
572#if URP_RADIO_TRACE_FRONTEND == 1
573 input[i * 2 + 1] = naccum; /* output noise filter results */
574#endif
575 npwr += naccum * naccum;
576 /* The calibration meter remains sqrt(sum(960 samples))/16. Its
577 * equivalent sample-power scale is 960/256, independent of where
578 * the USB block boundary falls relative to carrier loss. */
579 mySps->compOut = urp_micor_squelch_update(
580 &mySps->micor_squelch, mySps->compOut,
581 (double)naccum * naccum * 3.75, (uint32_t)mySps->setpt,
582 (uint32_t)mySps->hyst);
583 mySps->parentChan->rxCarrierGate[i] = !mySps->compOut;
584 }
585
586 --decimator;
587
588 if (decimator <= 0) {
589 decimator = decimate;
590
591 y = 0;
592 for (n = 0; n < nx; n++) {
593 y += fir_rxlpf[mySps->parentChan->rxlpf].coefs[n] * x[n];
594 }
595
596 y = ((y / calcAdjust) * outputGain) / M_Q8;
597
598#if URP_RADIO_TRACE_OVFLW == 1
599 if (y > 32767) {
600 y = 32767;
601 ast_log(LOG_ERROR, "urp_radio_receive_frontend() OVRFLW \n");
602 } else if (y < -32767) {
603 y = -32767;
604 ast_log(LOG_ERROR, "urp_radio_receive_frontend() UNDFLW \n");
605 }
606#else
607 if (y > 32767) {
608 y = 32767;
609 } else if (y < -32767) {
610 y = -32767;
611 }
612#endif
613 output[iOutput++] = y; /* Rx Baseband decimated */
614
615 } /* if decimator */
616 }
617
618 if (doNoise) {
619 npwr = sqrt(npwr) / 16;
620
621#if URP_RADIO_DEBUG == 1
622 if (mySps->parentChan->tracetype) {
623 for (i = 0; i < mySps->nSamples; i++) {
624 noutput[i] = npwr;
625 }
626 }
627#endif
628
629 ((urp_radio_state *)(mySps->parentChan))->rxRssi = mySps->apeak = npwr;
630 }
631
632 return 0;
633}

637*/
638i16 urp_radio_fir(urp_radio_stage *mySps)
639{
640 i32 nsamples, inputGain, outputGain, calcAdjust;

642 i16 *output;
643 i16 *x;

645 i32 i, ii;

647 i16 amax, amin, apeak = 0, discounteru = 0, discounterl = 0, discfactor;
648 i16 decimator, decimate, interpolate;
649 i16 numChanOut, selChanOut, mixOut, monoOut;
650
651 TRACEJ(5, "urp_radio_fir() %i %i\n", mySps->index, mySps->enabled);
652
653 if (!mySps->enabled) {
654 return 1;
655 }
656
657 inputGain = mySps->inputGain;
658 calcAdjust = mySps->calcAdjust;
659 outputGain = mySps->outputGain;
660
661 input = mySps->source;
662 output = mySps->sink;
663 x = mySps->x;
664 nx = mySps->nx;
665 coef = mySps->coef;
666
667 decimator = mySps->decimator;
668 decimate = mySps->decimate;
669 interpolate = mySps->interpolate;
670
671 setpt = mySps->setpt;

674 numChanOut = mySps->numChanOut;
675 selChanOut = mySps->selChanOut;
676 mixOut = mySps->mixOut;
677 monoOut = mySps->monoOut;
678
679 amax = mySps->amax;
680 amin = mySps->amin;
681 discounteru = mySps->discounteru;
682 discounterl = mySps->discounterl;

684 discfactor = mySps->discfactor;
685 hyst = mySps->hyst;
686 nsamples = mySps->nSamples;

695 output[(i * numChanOut) + selChanOut] = 0;
696 }
697 }
698 return 0;
699 }
700
701 ii = 0;
702 for (i = 0; i < nsamples; i++) {
703 int ix;
704
705 int64_t y = 0;
706
707 if (decimate < 0) {
708 decimator = decimate;
709 }
710
711 for (ix = 0; ix < interpolate; ix++) {
712 i16 n;
713 y = 0;
714
715 for (n = nx - 1; n > 0; n--) {
716 x[n] = x[n - 1];
717 }
718 x[0] = (input[i] * inputGain) / M_Q8;
719
720 for (n = 0; n < nx; n++) {
721 y += coef[n] * x[n];
722 }
723
724 y = ((y / calcAdjust) * outputGain) / M_Q8;
725

728 } else if (y < -32767) {
729 y = -32767;

736 output[(ii * numChanOut) + selChanOut] += y;

738 } else {

741 } else {

746 }

755 } else if (--discounteru <= 0) {

762 discounterl = discfactor;
763 } else if (--discounterl <= 0) {
764 discounterl = discfactor;
765 amin = (i32)((amin * 32700) / 32768);
766 }
767

771 compOut = 1;
772 } else if (compOut && (apeak < (setpt - hyst))) {
773 compOut = 0;
774 }
775 }
776 }
777
778 mySps->decimator = decimator;
779
780 mySps->amax = amax;
781 mySps->amin = amin;
782 mySps->apeak = apeak;
783 mySps->discounteru = discounteru;
784 mySps->discounterl = discounterl;
785
786 mySps->compOut = compOut;
787
788 return 0;
789}

1075/*
1076 Continuous Tone Coded Squelch (CTCSS) Detector
1077*/
1078i16 urp_ctcss_decode(urp_radio_state *pChan)
1079{
1080 i16 i, points2do, thit, relax;
1081 const i16 *pInput;
1082 i16 tnum, tmp, indexNow, diffpeak;
1083 i16 tv0, tv1, tv2, tv3, indexDebug;
1084 i16 points = 0;
1085 i16 indexWas = 0;
1086
1087 TRACEF(5, "urp_ctcss_decode(%p) %i %i %i %i\n", pChan, pChan->rxCtcss->enabled, 0,
1088 pChan->rxCtcss->testIndex, pChan->rxCtcss->decode);
1089
1090 if (!pChan->rxCtcss->enabled) {
1091 return 1;
1092 }
1093
1094 relax = pChan->rxCtcss->relax;
1095 pInput = pChan->rxCtcss->input;
1096
1097 thit = -1;
1098
1099 for (tnum = 0; tnum < CTCSS_NUM_CODES; tnum++) {
1100 i32 accum, peak;
1101 urp_ctcss_tone_detector *ptdet;
1102 i16 fudgeFactor;
1103 i16 binFactor;
1104
1105 TRACEF(6, " urp_ctcss_decode() tnum=%i %i\n", tnum, pChan->rxCtcssMap[tnum]);
1106
1107 if ((pChan->rxCtcssMap[tnum] == CTCSS_NULL) ||
1108 (pChan->rxCtcss->decode > CTCSS_NULL && (tnum != pChan->rxCtcss->decode))) {
1109 continue;
1110 }
1111
1112 TRACEF(6, " urp_ctcss_decode() tnum=%i\n", tnum);
1113
1114 ptdet = &(pChan->rxCtcss->tdet[tnum]);
1115 indexDebug = 0;
1116 points = points2do = pChan->nSamplesRx;
1117 fudgeFactor = ptdet->fudgeFactor;
1118 binFactor = ptdet->binFactor;
1119
1120 while (ptdet->counter < (points2do * CTCSS_SCOUNT_MUL)) {
1121 tmp = (ptdet->counter / CTCSS_SCOUNT_MUL) + 1;
1122 ptdet->counter -= (tmp * CTCSS_SCOUNT_MUL);
1123 points2do -= tmp;
1124 indexNow = points - points2do;
1125
1126 ptdet->counter += ptdet->counterFactor;
1127
1128 accum = pInput[indexNow - 1]; /* duuuude's major bug fix! */
1129
1130 ptdet->z[ptdet->zIndex] +=
1131 (((accum - ptdet->z[ptdet->zIndex]) * binFactor) / M_Q15);
1132
1133 peak = abs(ptdet->z[0] - ptdet->z[2]) + abs(ptdet->z[1] - ptdet->z[3]);
1134
1135 if (ptdet->peak < peak) {
1136 ptdet->peak += (((peak - ptdet->peak) * binFactor) / M_Q15);
1137 } else {
1138 ptdet->peak = peak;
1139 }
1140
1141 {
1142 static const i16 a0 = 13723;
1143 static const i16 a1 = -13723;
1144 i32 temp0, temp1;
1145 i16 x0;
1146
1147 /* differentiate */
1148 x0 = ptdet->zd;
1149 temp0 = x0 * a1;
1150 ptdet->zd = ptdet->peak;
1151 temp1 = ptdet->peak * a0;
1152 diffpeak = (temp0 + temp1) / 1024;
1153 }
1154
1155 if (diffpeak < (-0.03 * M_Q15)) {
1156 ptdet->dvd -= 4;
1157 } else if (ptdet->dvd < 0) {
1158 ptdet->dvd++;
1159 }
1160
1161 if ((ptdet->dvd < -12) && diffpeak > (-0.02 * M_Q15)) {
1162 ptdet->dvu += 2;
1163 } else if (ptdet->dvu) {
1164 ptdet->dvu--;
1165 }
1166
1167 tmp = ptdet->setpt;
1168 if (pChan->rxCtcss->decode == tnum) {
1169 if (relax) {
1170 tmp = (tmp * 55) / 100;
1171 } else {
1172 tmp = (tmp * 80) / 100;
1173 }
1174 }
1175
1176 if (ptdet->peak > tmp) {
1177 if (ptdet->decode < (fudgeFactor * 32)) {
1178 ptdet->decode++;
1179 }
1180 } else if (pChan->rxCtcss->decode == tnum) {
1181 if (ptdet->peak > ptdet->hyst) {
1182 ptdet->decode--;
1183 } else if (relax) {
1184 ptdet->decode--;
1185 } else {
1186 ptdet->decode -= 4;
1187 }
1188 } else {
1189 ptdet->decode = 0;
1190 }
1191
1192 if ((pChan->rxCtcss->decode == tnum) && !relax &&
1193 (ptdet->dvu > (0.00075 * M_Q15))) {
1194 ptdet->decode = 0;
1195 ptdet->z[0] = ptdet->z[1] = ptdet->z[2] = ptdet->z[3] = ptdet->dvu =
1196 0;
1197 TRACEF(4,
1198 "urp_ctcss_decode() turnoff detected by dvdt for tnum = "
1199 "%i.\n",
1200 tnum);
1201 }
1202
1203 if (ptdet->decode < 0 || !pChan->rxCarrierDetect) {
1204 ptdet->decode = 0;
1205 }
1206
1207 if (ptdet->decode >= fudgeFactor) {
1208 thit = tnum;
1209 if (pChan->rxCtcss->decode != tnum) {
1210 ptdet->zd = ptdet->dvu = ptdet->dvd = 0;
1211 }
1212 }
1213
1214#if URP_RADIO_DEBUG == 1
1215 if (thit >= 0 && thit == tnum) {
1216 TRACEF(6, " urp_ctcss_decode() %i %i %i %i \n", tnum, ptdet->peak,
1217 ptdet->setpt, ptdet->hyst);
1218 }
1219
1220 tv0 = ptdet->peak;
1221 tv1 = ptdet->decode;
1222 tv2 = tmp;
1223 tv3 = ptdet->dvu * 32;
1224
1225 if (indexDebug == 0) {
1226 ptdet->lasttv0 = ptdet->pDebug0[points - 1];
1227 ptdet->lasttv1 = ptdet->pDebug1[points - 1];
1228 ptdet->lasttv2 = ptdet->pDebug2[points - 1];
1229 ptdet->lasttv3 = ptdet->pDebug3[points - 1];
1230 }
1231
1232 while (indexDebug < indexNow) {
1233 ptdet->pDebug0[indexDebug] = ptdet->lasttv0;
1234 ptdet->pDebug1[indexDebug] = ptdet->lasttv1;
1235 ptdet->pDebug2[indexDebug] = ptdet->lasttv2;
1236 ptdet->pDebug3[indexDebug] = ptdet->lasttv3;
1237 indexDebug++;
1238 }
1239 ptdet->lasttv0 = tv0;
1240 ptdet->lasttv1 = tv1;
1241 ptdet->lasttv2 = tv2;
1242 ptdet->lasttv3 = tv3;
1243#endif
1244 indexWas = indexNow;
1245 ptdet->zIndex = (ptdet->zIndex + 1) % 4;
1246 }
1247 ptdet->counter -= (points2do * CTCSS_SCOUNT_MUL);
1248
1249#if URP_RADIO_DEBUG == 1
1250 for (i = indexWas; i < points; i++) {
1251 ptdet->pDebug0[i] = ptdet->lasttv0;
1252 ptdet->pDebug1[i] = ptdet->lasttv1;
1253 ptdet->pDebug2[i] = ptdet->lasttv2;
1254 ptdet->pDebug3[i] = ptdet->lasttv3;
1255 }
1256#endif
1257 }
1258
1259 if (pChan->rxCtcss->BlankingTimer > 0) {
1260 pChan->rxCtcss->BlankingTimer -= points;
1261 }
1262
1263 if (pChan->rxCtcss->BlankingTimer < 0) {
1264 pChan->rxCtcss->BlankingTimer = 0;
1265 }
1266
1267 if (thit > CTCSS_NULL && pChan->rxCtcss->decode <= CTCSS_NULL &&
1268 !pChan->rxCtcss->BlankingTimer) {
1269 pChan->rxCtcss->decode = thit;
1270 sprintf(pChan->rxctcssfreq, "%.1f", freq_ctcss[thit]);
1271 TRACEC(1, "ctcss decode %i %.1f\n", thit, freq_ctcss[thit]);
1272 } else if (thit <= CTCSS_NULL && pChan->rxCtcss->decode > CTCSS_NULL) {
1273 pChan->rxCtcss->BlankingTimer = SAMPLE_RATE_NETWORK / 5;
1274 pChan->rxCtcss->decode = CTCSS_NULL;
1275 strcpy(pChan->rxctcssfreq, "0");
1276 TRACEC(1, "ctcss decode NULL\n");
1277 for (tnum = 0; tnum < CTCSS_NUM_CODES; tnum++) {
1278 urp_ctcss_tone_detector *ptdet = NULL;
1279 ptdet = &(pChan->rxCtcss->tdet[tnum]);
1280 ptdet->decode = 0;
1281 ptdet->z[0] = ptdet->z[1] = ptdet->z[2] = ptdet->z[3] = 0;
1282 }
1283 }
1284 return 0;
1285}
