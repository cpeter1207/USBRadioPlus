43static uint32_t golay_remainder(uint32_t word)
44{
45 int bit;
46 for (bit = 22; bit >= 11; --bit)
47 if (word & (1U << bit))
48 word ^= DCS_GENERATOR << (bit - 11);
49 return word & 0x7ffU;
50}

56static uint32_t dcs_word(int code)
57{
58 uint32_t data = ((uint32_t)code & 0x1ffU) | 0x800U;
59 uint32_t remainder = data;
60 int bit;
61
62 for (bit = 0; bit < 12; ++bit) {
63 remainder <<= 1;
64 if (remainder & 0x1000U)
65 remainder ^= DCS_WIRE_GENERATOR;
66 }
67 return data | ((remainder & 0x0ffeU) << 11);
68}

88static void dcs_reset_turnoff_detector(struct urp_dcs_state *state, unsigned int sample_rate)
89{
90 uint32_t window = (sample_rate * DCS_TURNOFF_WINDOW_MS) / 1000U;

121static int dcs_process_turnoff_sample(struct urp_dcs_state *state, int32_t sample)
122{
123 double current = (double)sample +
124 state->receive_turnoff_coefficient * state->receive_turnoff_one -
125 state->receive_turnoff_two;
126
127 state->receive_turnoff_two = state->receive_turnoff_one;
128 state->receive_turnoff_one = current;
129 state->receive_turnoff_energy += (double)sample * sample;
130 ++state->receive_turnoff_window_samples;
131 if (state->receive_turnoff_window_samples < state->receive_turnoff_window_length)
132 return state->receive_turnoff_active;
133 {
134 double samples = state->receive_turnoff_window_samples;
135 double power = state->receive_turnoff_one * state->receive_turnoff_one +
136 state->receive_turnoff_two * state->receive_turnoff_two -
137 state->receive_turnoff_coefficient * state->receive_turnoff_one *
138 state->receive_turnoff_two;
139 double minimum_energy = samples * DCS_TURNOFF_MINIMUM_RMS * DCS_TURNOFF_MINIMUM_RMS;
140 int tone = state->receive_turnoff_energy >= minimum_energy &&
141 power >= state->receive_turnoff_energy * samples *
142 DCS_TURNOFF_MINIMUM_COHERENCE;
143
144 if (tone) {
145 state->receive_turnoff_consecutive_samples +=
146 state->receive_turnoff_window_samples;
147 if (state->receive_turnoff_consecutive_samples >=
148 state->receive_turnoff_minimum_samples)
149 state->receive_turnoff_active = 1;
150 } else {
151 state->receive_turnoff_consecutive_samples = 0;
152 state->receive_turnoff_active = 0;
153 }
154 }
155 state->receive_turnoff_one = 0.0;
156 state->receive_turnoff_two = 0.0;
157 state->receive_turnoff_energy = 0.0;
158 state->receive_turnoff_window_samples = 0;
159 return state->receive_turnoff_active;
160}

187int urp_dcs_parse_code(const char *text, int *code, int *inverted)
188{
189 int value;
190 if (!text || strlen(text) != 4 || !code || !inverted)
191 return -1;
192 if (text[0] < '0' || text[0] > '7' || text[1] < '0' || text[1] > '7' || text[2] < '0' ||
193 text[2] > '7')
194 return -1;
195 /* Three validated octal digits always represent a supported 000..777 code. */
196 value = ((text[0] - '0') << 6) | ((text[1] - '0') << 3) | (text[2] - '0');
197 if (toupper((unsigned char)text[3]) == 'N')
198 *inverted = 0;
199 else if (toupper((unsigned char)text[3]) == 'I')
200 *inverted = 1;
201 else
202 return -1;
203 *code = (int)value;
204 return 0;
205}

214void urp_dcs_init(struct urp_dcs_state *state)
215{
216 int first, second, third;
217 if (!state)
218 return;
219 memset(state, 0, sizeof(*state));
220 for (first = 0; first < 23; ++first) {
221 install_error(state, 1U << first);
222 for (second = first + 1; second < 23; ++second) {
223 install_error(state, (1U << first) | (1U << second));
224 for (third = second + 1; third < 23; ++third)
225 install_error(state,
226 (1U << first) | (1U << second) | (1U << third));
227 }
228 }
229 install_error(state, 0);
230}

264static int decode_word(struct urp_dcs_state *state, uint32_t received)
265{
266 uint32_t error = state->syndrome[golay_remainder(received)];
267 uint32_t corrected = received ^ error;
268 uint32_t data = corrected & 0xfffU;
269 if ((data & 0xe00U) != 0x800U)
270 return 0;
271 return (int)(data & 0x1ffU) == state->receive_code;
272}

321static void dcs_process_received_symbol(struct urp_dcs_state *state,
322 struct urp_dcs_receive_phase *receiver, int bit)
323{
324 int matched;
325
326 ++receiver->symbols_since_match;
327 receiver->word = ((receiver->word >> 1) | ((uint32_t)bit << 22)) & DCS_WORD_MASK;
328 matched = decode_word(state, receiver->word);
329 if (matched) {
330 if (receiver->match_count && receiver->symbols_since_match == 23U) {
331 if (receiver->match_count < 2U)
332 ++receiver->match_count;
333 } else {
334 receiver->match_count = 1;
335 }
336 receiver->symbols_since_match = 0;
337 if (receiver->match_count >= 2U) {
338 receiver->qualified = 1;
339 receiver->hold = 23U;
340 }
341 return;
342 }
343 if (receiver->qualified) {
344 /* Qualification always starts a nonzero one-word hold interval. */
345 --receiver->hold;
346 if (!receiver->hold) {
347 receiver->qualified = 0;
348 receiver->match_count = 0;
349 }
350 }

372int urp_dcs_process(struct urp_dcs_state *state, const int16_t *samples, size_t count,
373 size_t stride, unsigned int sample_rate)
374{
375 size_t index;
376 uint32_t clock_limit;
377 if (!state || !samples || !stride || !sample_rate || !state->enabled_receive)
378 return 0;
379 if (state->receive_sample_rate != sample_rate)
380 dcs_reset_receive_phase_bank(state, sample_rate);
381 clock_limit = sample_rate * DCS_CLOCK_SCALE;
382 for (index = 0; index < count; ++index) {
383 int32_t sample = samples[index * stride];
384 int32_t centered;
385 int tail_was_active;
386 int tail_active;
387 unsigned int phase;
388
389 centered = dcs_remove_dc(state, sample);
390 tail_was_active = state->receive_turnoff_active;
391 tail_active = dcs_process_turnoff_sample(state, centered);
392 if (tail_active && !tail_was_active)
393 dcs_clear_receive_qualification(state);
394 for (phase = 0; phase < URP_DCS_RX_PHASE_COUNT; ++phase) {
395 struct urp_dcs_receive_phase *receiver = &state->receive_phase[phase];
396 int bit;
397
398 receiver->sample_accumulator += centered;
399 receiver->bit_accumulator += DCS_CLOCK_INCREMENT;
400 if (receiver->bit_accumulator < clock_limit)
401 continue;
402 receiver->bit_accumulator -= clock_limit;
403 bit = receiver->sample_accumulator >= 0;
404 if (state->receive_inverted)
405 bit = !bit;
406 receiver->sample_accumulator = 0;
407 /* A qualified 134.4 Hz tail replaces data; ignore its symbol
408 * decisions until a non-tail window arrives, rather than allowing
409 * a stale or accidental Golay word to re-open the receiver. */
410 if (!tail_active)
411 dcs_process_received_symbol(state, receiver, bit);
412 }
413 }
414 state->valid = dcs_phase_bank_valid(state);
415 return state->valid;
416}

418void urp_dcs_generate(struct urp_dcs_state *state, double *output, size_t count,
419 unsigned int sample_rate, double peak, int enabled, int turnoff)
420{
421 size_t index;
422 if (!output || !count)
423 return;
424 if (!state || !enabled || peak <= 0.0 || !sample_rate) {
425 memset(output, 0, count * sizeof(*output));
426 return;
427 }
428 if (turnoff) {
429 double step = 2.0 * DCS_PI * URP_DCS_TURNOFF_FREQUENCY_HZ / sample_rate;
430 for (index = 0; index < count; ++index) {
431 output[index] = peak * sin(state->tail_phase);
432 state->tail_phase += step;
433 if (state->tail_phase >= 2.0 * DCS_PI)
434 state->tail_phase -= 2.0 * DCS_PI;
435 }
436 return;
437 }
438 state->transmit_word = dcs_word(state->transmit_code);
439 for (index = 0; index < count; ++index) {
440 int bit = (state->transmit_word >> state->transmit_phase) & 1U;
441 if (state->transmit_inverted)
442 bit = !bit;
443 output[index] = bit ? peak : -peak;
444 state->transmit_bit_accumulator += DCS_CLOCK_INCREMENT;
445 if (state->transmit_bit_accumulator >= sample_rate * DCS_CLOCK_SCALE) {
446 state->transmit_bit_accumulator -= sample_rate * DCS_CLOCK_SCALE;
447 state->transmit_phase = (state->transmit_phase + 1) % 23U;
448 }
449 }
