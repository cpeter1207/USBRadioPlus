60void urp_sample_queue_discard(struct urp_sample_queue *queue)
61{
62 unsigned int write = atomic_load_explicit(&queue->write, memory_order_acquire);
63
64 /* Only the consumer advances read. Keeping write monotonic means a

90int urp_sample_queue_push_sample(struct urp_sample_queue *queue, short sample)
91{
92 unsigned int write = atomic_load_explicit(&queue->write, memory_order_relaxed);
93 unsigned int read = atomic_load_explicit(&queue->read, memory_order_acquire);
94
95 if (!queue->samples || !queue->capacity)
96 return 0;
97 if (write - read >= queue->capacity)
98 return 0;
99 queue->samples[write % queue->capacity] = sample;
100 atomic_store_explicit(&queue->write, write + 1U, memory_order_release);
101 urp_sample_queue_note_high_water(queue, write + 1U - read);
102 return 1;
103}

105int urp_sample_queue_pop_sample(struct urp_sample_queue *queue, short *sample)
106{
107 unsigned int read = atomic_load_explicit(&queue->read, memory_order_relaxed);
108 unsigned int write = atomic_load_explicit(&queue->write, memory_order_acquire);
109
110 if (!queue->samples || !queue->capacity)
111 return 0;
112 if (read == write)
113 return 0;
114 *sample = queue->samples[read % queue->capacity];
115 atomic_store_explicit(&queue->read, read + 1U, memory_order_release);
116 return 1;
117}

168unsigned int urp_pcm_peak(const short *samples, size_t count)
169{
170 unsigned int peak = 0;
171 size_t i;
172 for (i = 0; i < count; ++i) {
173 unsigned int value =
174 samples[i] == INT16_MIN ? 32768U : (unsigned int)abs(samples[i]);
175 if (value > peak)
176 peak = value;
177 }

266size_t urp_parrot_play(struct urp_parrot_state *state, double *output, size_t count)
267{
268 size_t remaining;
269
270 if (!state->playing)
271 return 0;
272 remaining = state->count - state->play;
273 if (count > remaining)
274 count = remaining;
275 memcpy(output, state->audio + state->play, count * sizeof(*output));
276 state->play += count;
277 if (state->play >= state->count)
278 state->playing = 0;
279 return count;
280}

297void urp_prepare_receive_block(const short *stereo, short *pcm, double *working, size_t count,
298 short *delay, size_t delay_samples, unsigned int *delay_index,
299 struct urp_receive_block_stats *stats)
300{
301 size_t i;
302
303 stats->peak = 0;
304 stats->rail_samples = 0;
305 if (delay_samples && *delay_index >= delay_samples)
306 *delay_index = 0;
307 for (i = 0; i < count; ++i) {
308 unsigned int magnitude;
309 short sample = stereo[i * 2];
310
311 magnitude = sample == INT16_MIN ? 32768U : (unsigned int)abs(sample);
312 if (magnitude > stats->peak)
313 stats->peak = magnitude;
314 if (sample == INT16_MAX || sample == INT16_MIN)
315 stats->rail_samples++;
316 if (delay_samples) {
317 short delayed = delay[*delay_index];
318 delay[*delay_index] = sample;
319 sample = delayed;
320 if (++*delay_index == delay_samples)
321 *delay_index = 0;
322 }
323 pcm[i] = sample;
324 working[i] = sample;
325 }
326}

328unsigned long urp_render_transmit_block(const double *program, const double *ctcss,
329 const double *dcs, size_t count,
330 enum urp_tx_output_mode output_a,
331 enum urp_tx_output_mode output_b, double ctcss_peak_a,
332 double ctcss_bias_a, double ctcss_peak_b,
333 double ctcss_bias_b, short *stereo, short *meter_stereo)
334{
335 unsigned long rail_samples = 0;
336 size_t i;
337
338 for (i = 0; i < count; ++i) {
339 short output;
340
341 if (program[i] > INT16_MAX || program[i] < INT16_MIN)
342 rail_samples++;
343 output = (short)lrint(fmax(INT16_MIN, fmin(INT16_MAX, program[i])));
344 if (meter_stereo) {
345 meter_stereo[i * 2] = output;
346 meter_stereo[i * 2 + 1] = output;
347 }
348 if (urp_tx_output_has_program(output_a))
349 stereo[i * 2] = urp_saturating_add(stereo[i * 2], output);
350 if (urp_tx_output_has_program(output_b))
351 stereo[i * 2 + 1] = urp_saturating_add(stereo[i * 2 + 1], output);
352 if (output_a == URP_TX_OUTPUT_TONE || output_a == URP_TX_OUTPUT_COMPOSITE) {
353 short tone = (short)lrint(fmax(
354 INT16_MIN,
355 fmin(INT16_MAX, ctcss[i] * ctcss_peak_a + ctcss_bias_a + dcs[i])));
356 stereo[i * 2] = urp_saturating_add(stereo[i * 2], tone);
357 }
358 if (output_b == URP_TX_OUTPUT_TONE || output_b == URP_TX_OUTPUT_COMPOSITE) {
359 short tone = (short)lrint(fmax(
360 INT16_MIN,
361 fmin(INT16_MAX, ctcss[i] * ctcss_peak_b + ctcss_bias_b + dcs[i])));
362 stereo[i * 2 + 1] = urp_saturating_add(stereo[i * 2 + 1], tone);
363 }
364 }
365 return rail_samples;
366}
