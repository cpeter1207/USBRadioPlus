426static void test_sink_and_fifo_failures(void)
427{
428 struct txagc_config cleanup = configuration(1);
429 struct txagc_avfilter state;
430 double samples[960] = {0};
431
432 for (size_t index = 0; index < 7; ++index) {
433 /* The helper needs the corresponding sink from its own configured state. */
434 reset_failures();
435 txagc_avfilter_init(&state);
436 assert(!txagc_avfilter_process(&state, &cleanup, samples, 960, 48000));
437 AVFilterContext *own_sinks[] = {
438 state.meter_sink,
439 state.cleanup_pre_sink,
440 state.cleanup_pre_5_8_sink,
441 state.cleanup_pre_8_plus_sink,
442 state.cleanup_post_5_8_sink,
443 state.cleanup_post_8_plus_sink,
444 state.sink,
445 };
446 fail_buffer_sink = own_sinks[index];
447 assert(txagc_avfilter_process(&state, &cleanup, samples, 960, 48000) < 0);
448 txagc_avfilter_destroy(&state);
449 }
450 expect_runtime_failure(NULL, 1);
451 expect_runtime_failure(NULL, 2);
452
453 reset_failures();
454 txagc_avfilter_init(&state);
455 cleanup = configuration(0);
456 assert(!txagc_avfilter_process(&state, &cleanup, samples, 960, 48000));
457 av_audio_fifo_reset(state.fifo);
458 state.output_started = 1;
459 eagain_buffer_sink = state.sink;
460 assert(!txagc_avfilter_process(&state, &cleanup, samples, 960, 48000));
461 assert(state.runtime_underrun_samples == 960);
462 /* A buffering stage can also return nothing before its first output. The
463 * causal AGC no longer produces this case incidentally, so inject it. */
464 state.output_started = 0;
465 assert(!txagc_avfilter_process(&state, &cleanup, samples, 960, 48000));
466 assert(state.startup_fill_samples == 960);
467 assert(!state.output_started);
468 txagc_avfilter_destroy(&state);
469
470 reset_failures();
471 txagc_avfilter_init(&state);
472 assert(!txagc_avfilter_process(&state, &cleanup, samples, 960, 48000));
473 eof_buffer_sink = state.meter_sink;
474 assert(!txagc_avfilter_process(&state, &cleanup, samples, 960, 48000));
475 eof_buffer_sink = state.sink;
476 assert(!txagc_avfilter_process(&state, &cleanup, samples, 960, 48000));
477 txagc_avfilter_destroy(&state);
478}
