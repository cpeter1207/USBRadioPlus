807static void test_frontend_sample_gate(void)
808{
809 urp_radio_state template = {.pRxCodeSrc = "0",
810 .pTxCodeSrc = "0",
811 .pTxCodeDefault = "0",
812 .rxCdType = CD_XPMR_NOISE};
813 int16_t input[SAMPLES_PER_BLOCK * 6 * 2] = {0};
814 urp_radio_state *whole = urp_radio_create(&template, SAMPLES_PER_BLOCK);
815 urp_radio_state *split = urp_radio_create(&template, SAMPLES_PER_BLOCK);
816 assert(whole && split && whole->rxCarrierGate && split->rxCarrierGate);
817 whole->spsRx->setpt = split->spsRx->setpt = 7000;
818 whole->spsRx->hyst = split->spsRx->hyst = 500;
819 split->spsRx->nSamples = SAMPLES_PER_BLOCK / 2;
820
821 for (unsigned int block = 0; block < 97; ++block) {
822 double split_power = 0.0;
823 /* Train the idle reference, receive a fully quieted carrier, then
824 * remove it halfway through a block. No detector state is reset. */
825 for (size_t i = 0; i < SAMPLES_PER_BLOCK * 6; ++i) {
826 int noise = block < 64 || (block == 96 && i >= SAMPLES_PER_BLOCK * 3);
827 input[2 * i] = noise ? ((i & 2U) ? 12000 : -12000) : 0;
828 }
829 whole->spsRx->source = input;
830 assert(!urp_radio_receive_frontend(whole->spsRx));
831 for (size_t half = 0; half < 2; ++half) {
832 split->spsRx->source = input + half * SAMPLES_PER_BLOCK * 6;
833 assert(!urp_radio_receive_frontend(split->spsRx));
834 assert(!memcmp(whole->rxCarrierGate + half * SAMPLES_PER_BLOCK * 3,
835 split->rxCarrierGate, SAMPLES_PER_BLOCK * 3));
836 split_power += (double)split->rxRssi * split->rxRssi;
837 }
838 assert(whole->spsRx->compOut == split->spsRx->compOut);
839 /* Integer RSSI truncation loses less than two codes when the same
840 * block is metered in two halves: sqrt(sum(sample^2))/16. */
841 assert(fabs(whole->rxRssi - sqrt(split_power)) < 2.0);
842 }
843 assert(whole->rxCarrierGate[0]);
844 assert(!whole->rxCarrierGate[SAMPLES_PER_BLOCK * 6 - 1]);
845 assert(whole->spsRx->compOut);
846
847 /* VOX keeps its separate detector and never consumes the DSP mask. */
848 whole->rxCdType = CD_XPMR_VOX;
849 memset(whole->rxCarrierGate, 0x5a, SAMPLES_PER_BLOCK * 6);
850 int previous_rssi = whole->rxRssi;
851 assert(!urp_radio_receive_frontend(whole->spsRx));
852 assert(whole->rxRssi == previous_rssi);
853 for (size_t i = 0; i < SAMPLES_PER_BLOCK * 6; ++i)
854 assert(whole->rxCarrierGate[i] == 0x5a);
855 assert(!urp_radio_destroy(split));
856 assert(!urp_radio_destroy(whole));
857}

864static void fill_discriminator_noise(int16_t *input, uint32_t *random_state, int divisor)
865{
866 for (size_t sample = 0; sample < SAMPLES_PER_BLOCK * 6; ++sample) {
867 int sum = -1530;
868 /* Twelve uniforms approximate Gaussian noise without unbounded ADC
869 * peaks. The real frontend supplies the selected detector bandpass. */
870 for (unsigned int term = 0; term < 12; ++term) {
871 *random_state ^= *random_state << 13;
872 *random_state ^= *random_state >> 17;
873 *random_state ^= *random_state << 5;
874 sum += (int)(*random_state >> 24);
875 }
876 input[2 * sample] = (int16_t)(sum * 20 / divisor);
877 input[2 * sample + 1] = 0;
878 }
879}

882static void test_frontend_gaussian_noise(void)
883{
884 for (int noise_filter = 0; noise_filter < 2; ++noise_filter) {
885 urp_radio_state template = {.pRxCodeSrc = "0",
886 .pTxCodeSrc = "0",
887 .pTxCodeDefault = "0",
888 .rxCdType = CD_XPMR_NOISE};
889 urp_radio_state *state = urp_radio_create(&template, SAMPLES_PER_BLOCK);
890 uint32_t random_state = 0x9e3779b9U;
891 int16_t input[SAMPLES_PER_BLOCK * 6 * 2];
892 double idle_level = 0.0;
893 assert(state);
894 state->spsRx->source = input;
895 state->rxNoiseFilType = noise_filter;
896 state->spsRx->setpt = 0;
897 state->spsRx->hyst = 0;
898 for (unsigned int block = 0; block < 100; ++block) {
899 fill_discriminator_noise(input, &random_state, 1);
900 assert(!urp_radio_receive_frontend(state->spsRx));
901 idle_level += state->rxRssi / 100.0;
902 }
903 assert(idle_level > 1000.0 && idle_level < 30000.0);
904 state->spsRx->setpt = (int16_t)(idle_level * 0.8);
905 state->spsRx->hyst = (int16_t)(idle_level * 0.12);
906
907 /* Thirty seconds before and after a transmission catch slow hold
908 * accumulation; all comparator and capacitor states remain live. */
909 for (unsigned int phase = 0; phase < 2; ++phase) {
910 for (unsigned int block = 0; block < 1500; ++block) {
911 fill_discriminator_noise(input, &random_state, 1);
912 assert(!urp_radio_receive_frontend(state->spsRx));
913 for (size_t sample = 0; sample < SAMPLES_PER_BLOCK * 6; ++sample)
914 assert(!state->rxCarrierGate[sample]);
915 }
916 if (phase != 0)
917 break;
918 /* A 64:1 voltage reduction is 36 dB of discriminator quieting. */
919 for (unsigned int block = 0; block < 100; ++block) {
920 fill_discriminator_noise(input, &random_state, 64);
921 assert(!urp_radio_receive_frontend(state->spsRx));
922 if (block >= 10) {
923 for (size_t sample = 0; sample < SAMPLES_PER_BLOCK * 6;
924 ++sample)
925 assert(state->rxCarrierGate[sample]);
926 }
927 }
928 fill_discriminator_noise(input, &random_state, 1);
929 const size_t carrier_loss = SAMPLES_PER_BLOCK * 6 / 4;
930 for (size_t sample = 0; sample < carrier_loss; ++sample)
931 input[2 * sample] /= 64;
932 assert(!urp_radio_receive_frontend(state->spsRx));
933 size_t first_closed = carrier_loss;
934 while (first_closed < SAMPLES_PER_BLOCK * 6 &&
935 state->rxCarrierGate[first_closed])
936 ++first_closed;
937 assert(first_closed - carrier_loss < 480); /* Less than 10 ms at 48 kHz. */
938 for (size_t sample = first_closed; sample < SAMPLES_PER_BLOCK * 6; ++sample)
939 assert(!state->rxCarrierGate[sample]);
940 printf("noise filter %d: idle %.1f, abrupt carrier loss closes in %.3f "
941 "ms\n",
942 noise_filter, idle_level, (first_closed - carrier_loss) / 48.0);
943 }
944 assert(!urp_radio_destroy(state));
945 }
946}
