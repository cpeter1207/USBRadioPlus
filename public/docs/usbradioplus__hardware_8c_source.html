46static void urp_hardware_spi(struct urp_parallel_bus *bus, uint32_t data)
47{
48 static int initialized;
49 uint32_t bit = 0x00080000U;
50 int i;
51
52 bus->value &= ~(URP_DTX_CLK | URP_DTX_DATA | URP_DTX_ENABLE | URP_DTX_TXPWR | URP_DTX_TX);
53 urp_hardware_write(bus);
54 urp_hardware_delay(initialized ? 4 : 200);
55 initialized = 1;
56
57 for (i = 0; i < URP_PP_REGISTER_BITS; ++i, bit >>= 1) {
58 if (data & bit)
59 bus->value |= URP_DTX_DATA;
60 else
61 bus->value &= ~URP_DTX_DATA;
62 urp_hardware_write(bus);
63 urp_hardware_delay(1);
64 bus->value |= URP_DTX_CLK;
65 urp_hardware_write(bus);
66 urp_hardware_delay(1);
67 bus->value &= ~URP_DTX_CLK;
68 urp_hardware_write(bus);
69 urp_hardware_delay(1);
70 }

72 urp_hardware_write(bus);
73 bus->value |= URP_DTX_ENABLE;
74 urp_hardware_write(bus);
75 urp_hardware_delay(1);

80struct urp_rtx_words urp_hardware_rtx_words(uint32_t rx_freq, uint32_t tx_freq, int transmitting)
81{
82 const uint32_t reference_freq = rx_freq > 200000000U ? 16012500U : 16000000U;
83 const uint32_t step_freq = rx_freq > 200000000U ? 12500U : 5000U;
84 const uint32_t rx_if_freq = rx_freq > 200000000U ? 21400000U : 10700000U;
85 uint32_t synth_freq = transmitting ? tx_freq : rx_freq - rx_if_freq;
86 uint32_t word = (synth_freq / step_freq) << 1;
87 struct urp_rtx_words result;
88
89 result.reference = ((reference_freq / step_freq) << 1) | 1U;
90 result.synthesizer = ((word & 0xffffff80U) << 1) + (word & 0x7fU);
91 return result;
92}

109 int transmitting, int high_power)
110{

113 if (!bus || !bus->write || !rx_freq)

115 words = urp_hardware_rtx_words(rx_freq, tx_freq, transmitting);
116 urp_hardware_spi(bus, words.reference);
