45void *pulserthread(void *arg)
46{
47 (void)arg;
48 struct timeval now, then;
49 register int i, j, k;
50
51#ifdef HAVE_SYS_IO
52 if (haspp == 2) {
53 ioperm(pbase, 2, 1);
54 }
55#endif
56 stoppulser = 0;
57 pp_lastmask = 0;
58 ast_mutex_lock(&pp_lock);
59 ast_radio_ppwrite(haspp, ppfd, pbase, pport, pp_val);
60 ast_mutex_unlock(&pp_lock);
61 then = ast_radio_tvnow();
62
63 while (!stoppulser) {
64 usleep(50000);
65 ast_mutex_lock(&pp_lock);
66 now = ast_radio_tvnow();
67 j = ast_tvdiff_ms(now, then);
68 then = now;
69 pp_lastmask = pp_pulsemask;
70 pp_pulsemask = 0;
71 for (i = 2; i <= 9; i++) {
72 k = pp_pulsetimer[i];
73 if (k) {
74 k -= j;
75 if (k < 0) {
76 k = 0;
77 }
78 pp_pulsetimer[i] = k;
79 }
80 if (k) {
81 pp_pulsemask |= 1 << (i - 2);
82 }
83 }
84 if (pp_pulsemask != pp_lastmask) {
85 pp_val ^= pp_lastmask ^ pp_pulsemask;
86 ast_radio_ppwrite(haspp, ppfd, pbase, pport, pp_val);
87 }
88 ast_mutex_unlock(&pp_lock);
89 }
90 return NULL;
91}

110int hidhdwconfig(struct chan_usbradio_pvt *o)
111{
112 int i;
113
114 /* NOTE: on the CM-108AH, GPIO2 is *not* a REAL GPIO.. it was re-purposed
115 * as a signal called "HOOK" which can only be read from the HID.
116 * Apparently, in a REAL CM-108, GPIO really works as a GPIO
117 */
118
119 if (o->hdwtype == 1) {
120 /* sphusb */
121 o->hid_gpio_ctl = 0x08; /* set GPIO4 to output mode */
122 o->hid_gpio_ctl_loc = 2; /* For CTL of GPIO */
123 o->hid_io_cor = 4; /* GPIO3 is COR */
124 o->hid_io_cor_loc = 1; /* GPIO3 is COR */
125 o->hid_io_ctcss = 2; /* GPIO 2 is External CTCSS */
126 o->hid_io_ctcss_loc = 1; /* is GPIO 2 */
127 o->hid_io_ptt = 8; /* GPIO 4 is PTT */
128 o->hid_gpio_loc = 1; /* For ALL GPIO */
129 o->valid_gpios = 1; /* for GPIO 1 */
130 } else if (o->hdwtype == 0) {
131 /* dudeusb */
132 o->hid_gpio_ctl = 0x04; /* set GPIO 3 to output mode */
133 o->hid_gpio_ctl_loc = 2; /* For CTL of GPIO */
134 o->hid_io_cor = 2; /* VOLD DN is COR */
135 o->hid_io_cor_loc = 0; /* VOL DN COR */
136 o->hid_io_ctcss = 1; /* VOL UP External CTCSS */
137 o->hid_io_ctcss_loc = 0; /* VOL UP External CTCSS */
138 o->hid_io_ptt = 4; /* GPIO 3 is PTT */
139 o->hid_gpio_loc = 1; /* For ALL GPIO */
140 o->valid_gpios = 0xfb; /* for GPIO 1,2,4,5,6,7,8 (5,6,7,8 for CM-119 only) */
141 } else if (o->hdwtype == 2) {
142 /* NHRC (N1KDO) (dudeusb w/o user GPIO) */
143 o->hid_gpio_ctl = 0x04; /* set GPIO 3 to output mode */
144 o->hid_gpio_ctl_loc = 2; /* For CTL of GPIO */
145 o->hid_io_cor = 2; /* VOLD DN is COR */
146 o->hid_io_cor_loc = 0; /* VOL DN COR */
147 o->hid_io_ctcss = 1; /* VOL UP is External CTCSS */
148 o->hid_io_ctcss_loc = 0; /* VOL UP CTCSS */
149 o->hid_io_ptt = 4; /* GPIO 3 is PTT */
150 o->hid_gpio_loc = 1; /* For ALL GPIO */
151 o->valid_gpios = 0; /* for GPIO 1,2,4 */
152 } else if (o->hdwtype == 3) {
153 /* custom version */
154 o->hid_gpio_ctl = 0x0c; /* set GPIO 3 & 4 to output mode */
155 o->hid_gpio_ctl_loc = 2; /* For CTL of GPIO */
156 o->hid_io_cor = 2; /* VOLD DN is COR */
157 o->hid_io_cor_loc = 0; /* VOL DN COR */
158 o->hid_io_ctcss = 2; /* GPIO 2 is External CTCSS */
159 o->hid_io_ctcss_loc = 1; /* is GPIO 2 */
160 o->hid_io_ptt = 4; /* GPIO 3 is PTT */
161 o->hid_gpio_loc = 1; /* For ALL GPIO */
162 o->valid_gpios = 1; /* for GPIO 1 */
163 }
164 /* validate clipledgpio setting (Clip LED GPIO#) */
165 if (o->clipledgpio) {
166 if (o->clipledgpio >= GPIO_PINCOUNT ||
167 !(o->valid_gpios & (1 << (o->clipledgpio - 1)))) {
168 ast_log(LOG_ERROR, "Channel %s: clipledgpio = GPIO%d not supported\n",
169 o->name, o->clipledgpio);
170 o->clipledgpio = 0;
171 } else {
172 o->hid_gpio_ctl |= 1 << (o->clipledgpio -
173 1); /* confirm Clip LED GPIO set to output mode */
174 }
175 }
176 o->hid_gpio_val = 0;
177 for (i = 0; i < GPIO_PINCOUNT; i++) {
178 /* skip if this one not specified */
179 if (!o->gpios[i]) {
180 continue;
181 }
182 /* skip if not out */
183 if (strncasecmp(o->gpios[i], "out", 3)) {
184 continue;
185 }
186 /* skip if PTT */
187 if ((1 << i) & o->hid_io_ptt) {
188 ast_log(LOG_ERROR,
189 "Channel %s: You can't specify gpio%d, since its the PTT.\n",
190 o->name, i + 1);
191 continue;
192 }
193 /* skip if not a valid GPIO */
194 if (!(o->valid_gpios & (1 << i))) {
195 ast_log(LOG_ERROR,
196 "Channel %s: You can't specify gpio%d, it is not valid in this "
197 "configuration.\n",
198 o->name, i + 1);
199 continue;
200 }
201 o->hid_gpio_ctl |= (1 << i); /* set this one to output, also */
202 /* if default value is 1, set it */
203 if (!strcasecmp(o->gpios[i], "out1")) {
204 o->hid_gpio_val |= (1 << i);
205 }
206 }
207 if (o->invertptt) {
208 o->hid_gpio_val |= o->hid_io_ptt;
209 }
210 return 0;
211}

251void usbradioplus_queue_program(struct chan_usbradio_pvt *o, const short *samples, size_t count)
252{
253 unsigned int seed_frames, target_samples;
254 ast_mutex_lock(&o->plus_link_lock);
255 /* Seed stream startup or underrun recovery with the target reserve.
256 * A resampled stream needs one extra frame for sinc history.
257 * PTT transitions do not reseed the stream. */
258 target_samples = o->plus_native_fifo.target_samples ? o->plus_native_fifo.target_samples
259 : URP_FIFO_TARGET_NORMAL;
260 seed_frames = !o->plus_native_fifo.primed && !o->plus_native_fifo.count &&
261 !urp_program_queue_pending(&o->plus_program_queue)
262 ? (target_samples + URP_NATIVE_SAMPLES - 1U) / URP_NATIVE_SAMPLES -
263 1U + (o->plus_app_rpt_rate != URP_RATE_NATIVE)
264 : 0;
265 if (urp_program_queue_push(&o->plus_program_queue, samples, count, o->plus_app_rpt_samples,
266 seed_frames)) {
267 o->plus_link_queue_overflows++;
268 }
269 ast_mutex_unlock(&o->plus_link_lock);
270}

281int usbradio_indicate(struct ast_channel *c, int cond_in, const void *data, size_t datalen)
282{
283 struct chan_usbradio_pvt *o = ast_channel_tech_pvt(c);
284 enum ast_control_frame_type cond = cond_in;
285
286 switch (cond) {
287 case AST_CONTROL_BUSY:
288 case AST_CONTROL_CONGESTION:
289 case AST_CONTROL_RINGING:
290 break;
291 case AST_CONTROL_VIDUPDATE:
292 break;
293 case AST_CONTROL_HOLD:
294 ast_verbose("Channel %s: Console has been placed on hold.\n", o->name);
295 ast_moh_start(c, data, "default");
296 break;
297 case AST_CONTROL_UNHOLD:
298 ast_verbose("Channel %s: Console has been retrieved from hold.\n", o->name);
299 ast_moh_stop(c);
300 break;
301 case AST_CONTROL_PROCEEDING:
302 ast_verbose("Channel %s: Call Proceeding.\n", o->name);
303 ast_moh_stop(c);
304 break;
305 case AST_CONTROL_PROGRESS:
306 ast_verbose("Channel %s: Call Progress.\n", o->name);
307 ast_moh_stop(c);
308 break;
309 case AST_CONTROL_RADIO_KEY:
310 o->txkeyed = 1;
311 kickptt(o);
312 ast_debug(1, "Channel %s: ACRK code=%s TX ON.\n", o->name, (char *)data);
313 if (datalen && ((char *)(data))[0] != '0') {
314 o->forcetxcode = 1;
315 memset(o->set_txctcssfreq, 0,
316 sizeof(o->set_txctcssfreq)); /* Possibly unnecessary, if this is used
317 as a string? */
318 ast_copy_string(o->set_txctcssfreq, data, sizeof(o->set_txctcssfreq));
319 radio_config(o);
320 }
321 break;
322 case AST_CONTROL_RADIO_UNKEY:
323 o->txkeyed = 0;
324 kickptt(o);
325 ast_debug(1, "Channel %s: ACRUK TX OFF.\n", o->name);
326 if (o->forcetxcode) {
327 o->forcetxcode = 0;
328 o->radio->pTxCodeDefault = o->txctcssdefault;
329 ast_debug(1, "Channel %s: Forced Tx Squelch Code cleared.\n", o->name);
330 }
331 break;
332 default:
333 ast_log(LOG_WARNING, "Channel %s: Don't know how to display condition %d.\n",
334 o->name, cond);
335 return -1;
336 }
337
338 return 0;
339}

454 }
455 o->radio->txPttIn = 0;
456 o->txtestkey = 0;

1145void tune_rxvoice(int fd, struct chan_usbradio_pvt *o, int intflag)
1146{
1147 const int target = 7200; /* peak */
1148 const int tolerance = 360; /* peak to peak */
1149 const float settingmin = 0.1;
1150 const float settingmax = 5;
1151 const float settingstart = 1;
1152 const int maxtries = 12;
1153
1154 float setting;
1155
1156 int tries = 0, meas;
1157
1158 ast_cli(fd, "INFO: RX VOICE ADJUST START.\n");
1159 ast_cli(fd, "target=%i tolerance=%i \n", target, tolerance);
1160
1161 o->radio->b.tuning = 1;
1162 if (!o->radio->spsMeasure) {
1163 ast_cli(fd, "ERROR: NO MEASURE BLOCK.\n");
1164 o->radio->b.tuning = 0;
1165 return;
1166 }
1167
1168 if (!o->radio->spsMeasure->source || !o->radio->prxVoiceAdjust) {
1169 ast_cli(fd, "ERROR: NO SOURCE OR MEASURE SETTING.\n");
1170 o->radio->b.tuning = 0;
1171 return;
1172 }
1173
1174 o->radio->spsMeasure->source = o->radio->spsRxOut->sink;
1175 o->radio->spsMeasure->enabled = 1;
1176 o->radio->spsMeasure->discfactor = 1000;
1177
1178 setting = settingstart;
1179
1180 while (tries < maxtries) {
1181 *(o->radio->prxVoiceAdjust) = setting * M_Q8;
1182 if (ast_radio_wait_or_poll(fd, 10, intflag)) {
1183 o->radio->b.tuning = 0;
1184 return;
1185 }
1186 o->radio->spsMeasure->amax = o->radio->spsMeasure->amin = 0;
1187 if (ast_radio_wait_or_poll(fd, 1000, intflag)) {
1188 o->radio->b.tuning = 0;
1189 return;
1190 }
1191 meas = o->radio->spsMeasure->apeak;
1192 ast_cli(fd, "tries=%i, setting=%f, meas=%i\n", tries, setting, meas);
1193
1194 if (meas < (target - tolerance) || meas > (target + tolerance) || tries < 3) {
1195 setting = setting * target / meas;
1196 } else if (tries > 4 && meas > (target - tolerance) &&
1197 meas < (target + tolerance)) {
1198 break;
1199 }
1200 if (setting < settingmin) {
1201 setting = settingmin;
1202 } else if (setting > settingmax) {
1203 setting = settingmax;
1204 }
1205
1206 tries++;
1207 }
1208
1209 o->radio->spsMeasure->enabled = 0;
1210
1211 ast_cli(fd, "DONE tries=%i, setting=%f, meas=%f\n", tries, setting, (float)meas);
1212 if (meas < (target - tolerance) || meas > (target + tolerance)) {
1213 ast_cli(fd, "ERROR: RX VOICE GAIN ADJUST FAILED.\n");
1214 } else {
1215 ast_cli(fd, "INFO: RX VOICE GAIN ADJUST SUCCESS.\n");
1216 usbradioplus_processing_set_local_input_gain(
1217 o->name, 20.0 * log10(fmax(0.000001, 2.0 * setting)));
1218 }
1219 o->radio->b.tuning = 0;
1220}

1254 }
1255 meas = o->radio->spsMeasure->apeak;
1256 ast_cli(fd, "tries=%i, setting=%f, meas=%i\n", tries, setting, meas);
1257
1258 if (meas < (target - tolerance) || meas > (target + tolerance) || tries < 3) {
1259 setting = setting * target / meas;
1260 } else if (tries > 4 && meas > (target - tolerance) &&
1261 meas < (target + tolerance)) {
1262 break;
1263 }
1264 if (setting < settingmin) {
1265 setting = settingmin;
1266 } else if (setting > settingmax) {
1267 setting = settingmax;
1268 }
1269
1270 tries++;
1271 }
1272 o->radio->spsMeasure->enabled = 0;
1273 ast_cli(fd, "DONE tries=%i, setting=%f, meas=%.2f\n", tries, setting, (float)meas);
1274 if (meas < (target - tolerance) || meas > (target + tolerance)) {
1275 ast_cli(fd, "ERROR: RX CTCSS GAIN ADJUST FAILED.\n");
1276 } else {
1277 ast_cli(fd, "INFO: RX CTCSS GAIN ADJUST SUCCESS.\n");
1278 o->rxctcssadj = setting;
1279 }
1280
1281 if (o->rxcdtype == CD_XPMR_NOISE) {
1282 int normRssi;
1283
1284 if (ast_radio_wait_or_poll(fd, 200, intflag)) {
1285 o->radio->b.tuning = 0;
1286 return;
1287 }
1288
1289 normRssi = ((32767 - o->radio->rxRssi) * AUDIO_ADJUSTMENT / 32767);
1290
1291 if (o->rxsquelchadj > normRssi) {
1292 ast_cli(fd,
1293 "WARNING: RSSI=%i SQUELCH=%i and is too tight. Use 'radio tune "
1294 "rxsquelch'.\n",
1295 normRssi, o->rxsquelchadj);
1296 } else {
1297 ast_cli(fd, "INFO: RX RSSI=%i\n", normRssi);
1298 }
1299 }
1300 o->radio->b.tuning = 0;
1301}
