92void urp_ctcss_legacy_scaled_levels(double frequency, int filter_250, int tone_gain_q8,
93 int output_gain_q8, double *amplitude, double *bias)
94{
95 size_t index = closest_frequency(frequency);
96 long maximum = (long)(filter_250 ? peak_250[index] : peak_215[index]);
97 long difference = filter_250 ? bias_250[index] : bias_215[index];
98 long positive = difference > 0 ? maximum : maximum + difference;
99 long negative = difference < 0 ? maximum : maximum - difference;
100
101 /* Preserve XPMR's fixed-point truncation at both gain stages. */
102 positive = positive * tone_gain_q8 / 256;
103 positive = positive * output_gain_q8 / 256;
104 negative = negative * tone_gain_q8 / 256;
105 negative = negative * output_gain_q8 / 256;
106 *amplitude = (double)(positive + negative) / 2.0;
107 *bias = (double)(positive - negative) / 2.0;
108}

119static void generate(struct urp_ctcss_generator *generator, double *output, size_t count,
120 double frequency, double peak, int enabled, double phase_shift_degrees)
121{
122 size_t i;
123 double step, sine, cosine, sine_step, cosine_step;
124
125 if (!enabled || frequency <= 0.0 || peak <= 0.0) {
126 memset(output, 0, count * sizeof(*output));
127 return;
128 }
129 if (phase_shift_degrees) {
130 generator->phase += 2.0 * URP_PI * (double)phase_shift_degrees / 360.0;
131 generator->phase = fmod(generator->phase, 2.0 * URP_PI);
132 }
133 step = 2.0 * URP_PI * frequency / URP_CTCSS_RATE;
134 sine = sin(generator->phase);
135 cosine = cos(generator->phase);
136 sine_step = sin(step);
137 cosine_step = cos(step);
138 for (i = 0; i < count; ++i) {
139 double next_sine, next_cosine;
140 output[i] = peak * sine;
141 next_sine = sine * cosine_step + cosine * sine_step;
142 next_cosine = cosine * cosine_step - sine * sine_step;
143 sine = next_sine;
144 cosine = next_cosine;
145 }
146 generator->phase = fmod(generator->phase + step * (double)count, 2.0 * URP_PI);
147}
