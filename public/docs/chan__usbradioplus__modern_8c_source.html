421URP_CHANNEL_LOCAL void usbradio_adjust_txmix_for_mono(struct chan_usbradio_pvt *o)
422{
423 enum radio_tx_mix a = o->txmixa;
424 enum radio_tx_mix b = o->txmixb;
425
426 if (o->pa.output_channels != 1) {
427 return;
428 }
429
430 if (a == TX_OUT_OFF && b == TX_OUT_OFF) {
431 return;
432 }
433
434 if (a == TX_OUT_OFF) {
435 /* B holds the configured source; duplicate it to both channels. */
436 a = b;
437 } else if (b == TX_OUT_OFF) {
438 b = a;
439 } else if (a != b) {
440 /* Voice+tone on A/B is composite once PortAudio averages L/R. */
441 if ((a == TX_OUT_VOICE && b == TX_OUT_LSD) ||
442 (a == TX_OUT_LSD && b == TX_OUT_VOICE)) {
443 ast_log(LOG_WARNING,
444 "Channel %s: mono TX device; voice+tone A/B averaged into "
445 "composite (both sources kept)\n",
446 o->name);
447 o->txmixa = TX_OUT_COMPOSITE;
448 o->txmixb = TX_OUT_COMPOSITE;
449 if (o->radio) {
450 o->radio->txMixA = TX_OUT_COMPOSITE;
451 o->radio->txMixB = TX_OUT_COMPOSITE;
452 }
453 return;
454 }
455
456 ast_log(LOG_WARNING,
457 "Channel %s: unsupported txmixa/txmixb pair on mono TX; keeping A, "
458 "disabling B\n",
459 o->name);
460 b = a;
461 }
462
463 if (a != o->txmixa || b != o->txmixb) {
464 ast_log(LOG_WARNING,
465 "Channel %s: mono TX device; forcing equal A/B mix so PortAudio downmix "
466 "stays full level\n",
467 o->name);
468 }
469
470 o->txmixa = a;
471 o->txmixb = b;
472
473 if (o->radio) {
474 o->radio->txMixA = a;
475 o->radio->txMixB = b;
476 }
477}

519URP_CHANNEL_LOCAL int usbradio_swap_hid_wait(struct chan_usbradio_pvt *o)
520{
521 int swapping;
522
523 ast_mutex_lock(&o->swap_lock);
524 swapping = o->swap_state == DEVICE_SWAP_QUIESCING;
525 while (swapping && !o->swap_audio_ready && !o->stophid) {
526 ast_mutex_unlock(&o->swap_lock);
527 usleep(10000);
528 ast_mutex_lock(&o->swap_lock);
529 swapping = o->swap_state == DEVICE_SWAP_QUIESCING;
530 }
531 if (swapping && o->swap_audio_ready) {
532 o->swap_state = DEVICE_SWAP_READY;
533 } else {
534 swapping = 0;
535 }
536 while (o->swap_state == DEVICE_SWAP_READY && !o->stophid) {
537 ast_mutex_unlock(&o->swap_lock);
538 usleep(10000);
539 ast_mutex_lock(&o->swap_lock);
540 }
541 ast_mutex_unlock(&o->swap_lock);
542 return swapping;
543}

639URP_CHANNEL_LOCAL int init_audio_device(struct chan_usbradio_pvt *o)
640{
641 struct ast_radio_device *radio_device;
642 struct ast_radio_device_request request = {
643 .devstr = o->devstr,
644 .serial = o->serial,
645 .owner = o->name,
646 .required_caps = AST_RADIO_CAP_CM108_HID,
647 .minimum_input_channels = 1,
648 .minimum_output_channels = 1,
649 };
650 enum ast_radio_device_result result;
651 int automatic;
652
653 automatic = ast_strlen_zero(request.serial) && ast_strlen_zero(request.devstr);
654 result = ast_radio_device_acquire(&request, &radio_device);
655 if (result != AST_RADIO_DEVICE_READY) {
656 if (o->device_error != result) {
657 usbradio_log_fault(o, 0, "Channel %s: %s\n", o->name,
658 ast_radio_device_result_str(result));
659 }
660 o->device_error = result;
661 return -1;
662 }
663
664 /* Required receive and transmit mixer limits */
665 if (ast_radio_device_mixer_max(radio_device, radio_device->mixer_rx_paths,
666 AST_RADIO_MIXER_CAPTURE_VOLUME) <= 0 ||
667 ast_radio_device_mixer_max(radio_device, radio_device->mixer_tx_paths,
668 AST_RADIO_MIXER_PLAYBACK_VOLUME) <= 0) {
669 usbradio_log_fault(
670 o, 0, "Channel %s: Cannot use USB radio device %s without mixer limits\n",
671 o->name, radio_device->devstr);
672 ast_radio_device_release(radio_device);
673 o->device_error = AST_RADIO_DEVICE_ERROR;
674 return -1;
675 }
676
677 /* Acquired device publication without configured selector changes */
678 ast_mutex_lock(&o->device_lock);
679 o->radio_device = radio_device;
680 ast_mutex_unlock(&o->device_lock);
681 o->device_error = AST_RADIO_DEVICE_READY;
682
683 if (automatic) {
684 ast_log(LOG_NOTICE, "Channel %s: Automatically assigned USB device %s\n", o->name,
685 radio_device->devstr);
686 }
687 return 0;
688}

694URP_CHANNEL_LOCAL int usbradio_start_audio(struct chan_usbradio_pvt *o)
695{
696 char devstr[sizeof(o->devstr)];
697 PaError res;
698
699 if (o->pa.active) {
700 return 0;
701 }
702 o->plus_portaudio_playout_hold_callbacks = 0U;
703
704 /* Exact PortAudio endpoints from the active device lease */
705 ast_mutex_lock(&o->device_lock);
706 if (!o->radio_device) {
707 ast_mutex_unlock(&o->device_lock);
708 return -1;
709 }
710 ast_copy_string(devstr, o->radio_device->devstr, sizeof(devstr));
711 o->pa.input_channels = o->radio_device->pa_input_channels >= 2 ? 2 : 1;
712 res = ast_radio_pa_open_device(&o->pa, o->radio_device);
713 ast_mutex_unlock(&o->device_lock);
714 if (res != paNoError) {
715 ast_log(LOG_WARNING, "Channel %s: Unable to open PortAudio stream for %s (%s)\n",
716 o->name, devstr, Pa_GetErrorText(res));
717 return -1;
718 }
719 {
720 const PaStreamInfo *info = o->pa.stream ? Pa_GetStreamInfo(o->pa.stream) : NULL;
721 double callbacks = 1.0;
722
723 /* PortAudio exposes its best queue-delay estimate only while the stream
724 * is open. Cache a callback count here; the hardware loop stays lock-free. */
725 if (info && isfinite(info->outputLatency) && info->outputLatency >= 0.0)
726 callbacks += ceil(info->outputLatency * (double)URP_RATE_NATIVE /
727 (double)URP_NATIVE_SAMPLES);
728 o->plus_portaudio_playout_hold_callbacks =
729 callbacks > (double)UINT_MAX ? UINT_MAX : (unsigned int)callbacks;
730 }
731
732 usbradio_adjust_txmix_for_mono(o);
733
734 res = ast_radio_pa_start(&o->pa);
735 if (res != paNoError) {
736 ast_log(LOG_WARNING, "Channel %s: Unable to start PortAudio stream for %s (%s)\n",
737 o->name, devstr, Pa_GetErrorText(res));
738 ast_radio_pa_stop(&o->pa);
739 o->plus_portaudio_playout_hold_callbacks = 0U;
740 return -1;
741 }
742
743 return 0;
744}

750static uint8_t hidthread_parallel_ptt_mask(const struct chan_usbradio_pvt *o)
751{
752 uint8_t mask = 0;
753 int pin;
754
755 if (!haspp)
756 return 0;
757 for (pin = 2; pin <= 9; ++pin) {
758 if (o->pps[pin] && !strncasecmp(o->pps[pin], "ptt", 3))
759 mask |= (uint8_t)(1U << (pin - 2));
760 }
761 return mask;
762}

773URP_CHANNEL_LOCAL void hidthread_apply_ptt(struct chan_usbradio_pvt *o,
774 struct libusb_device_handle *usb_handle,
775 unsigned char *buf, unsigned char *bufsave, int asserted,
776 const struct usbradioplus_radio_program_request *request,
777 int reprogram)
778{
779 const uint8_t parallel_mask = hidthread_parallel_ptt_mask(o);
780
781 if (haspp) {
782 ast_mutex_lock(&pp_lock);
783 if (reprogram) {
784 struct urp_parallel_bus bus = {
785 .value = (uint8_t)pp_val,
786 .write = usbradioplus_parallel_program_write,
787 };
788
789 urp_hardware_program_radio(&bus, request->rx_frequency,
790 request->tx_frequency, asserted,
791 request->high_power);
792 pp_val = (int8_t)bus.value;
793 } else if (!asserted) {
794 struct urp_parallel_bus bus = {
795 .value = (uint8_t)pp_val,
796 .write = usbradioplus_parallel_program_write,
797 };
798
799 /* A concurrent control-plane publication must never postpone an unkey. */
800 urp_hardware_clear_transmit(&bus);
801 pp_val = (int8_t)bus.value;
802 }
803 urp_apply_ptt_outputs(asserted, o->invertptt, parallel_mask, o->hid_io_ptt,
804 &o->hid_gpio_val, &pp_val);
805 if (parallel_mask)
806 ast_radio_ppwrite(haspp, ppfd, pbase, pport, pp_val);
807 ast_mutex_unlock(&pp_lock);
808 } else {
809 /* HID PTT remains available on interfaces without a parallel port. */
810 urp_apply_ptt_outputs(asserted, o->invertptt, 0, o->hid_io_ptt, &o->hid_gpio_val,
811 &pp_val);
812 }
813 buf[o->hid_gpio_loc] = o->hid_gpio_val ^ o->hid_gpio_pulsemask;
814 buf[o->hid_gpio_ctl_loc] = o->hid_gpio_ctl;
815 memcpy(bufsave, buf, 4);
816 ast_radio_hid_set_outputs(usb_handle, buf);
817 o->lasttx = !!asserted;
818 /* Publish only after both physical output paths have received their state. */
819 atomic_store_explicit(&o->plus_hardware_ptt_applied, !!asserted, memory_order_release);
820}

827static void hidthread_failsafe_unkey(struct chan_usbradio_pvt *o,
828 struct libusb_device_handle *usb_handle, unsigned char *buf)
829{
830 const uint8_t parallel_mask = hidthread_parallel_ptt_mask(o);
831
832 if (haspp) {
833 struct urp_parallel_bus bus;
834
835 ast_mutex_lock(&pp_lock);
836 bus.value = (uint8_t)pp_val;
837 bus.write = usbradioplus_parallel_program_write;
838 bus.opaque = NULL;
839 /* Clear the RTX PTT bit immediately; a serial reprogram cycle must never
840 * be a prerequisite for transmitter release. */
841 urp_hardware_clear_transmit(&bus);
842 pp_val = (int8_t)bus.value;
843 urp_apply_ptt_outputs(0, o->invertptt, parallel_mask, o->hid_io_ptt,
844 &o->hid_gpio_val, &pp_val);
845 ast_radio_ppwrite(haspp, ppfd, pbase, pport, pp_val);
846 ast_mutex_unlock(&pp_lock);
847 }
848 if (usb_handle) {
849 ast_mutex_lock(&o->usblock);
850 o->hid_gpio_val &= ~o->hid_io_ptt;
851 if (o->invertptt)

854 buf[o->hid_gpio_ctl_loc] = o->hid_gpio_ctl;

856 ast_mutex_unlock(&o->usblock);
857 }
858 o->lasttx = 0;
859 atomic_store_explicit(&o->plus_hardware_ptt_applied, 0, memory_order_release);
860 atomic_store_explicit(&o->plus_hardware_online, 0, memory_order_release);
861}

1307 };

1310 /* skip if not specified */

1312 continue;
1313 }
1314 /* skip if not input */
1315 if (strcasecmp(o->pps[i], "in")) {
1316 continue;
1317 }
1318 /* if bit has changed, or never reported */
1319 if ((!o->had_pp_in) ||
1320 ((o->last_pp_in & (1 << ppinshift[i])) !=
1321 (j & (1 << ppinshift[i])))) {

1323 (j & (1 << ppinshift[i])) ? 1 : 0);
1324 fr.data.ptr = buf1;
1325 fr.datalen = strlen(buf1);
1326 ast_queue_frame(o->owner, &fr);
1327 }
1328 }
1329 o->had_pp_in = 1;
1330 o->last_pp_in = j;
1331 }
1332 for (i = 10; i <= 15; i++) {
1333 if ((o->pps[i]) && (!strcasecmp(o->pps[i], "cor"))) {
1334 if (k & (1 << ppinshift[i]))
1335 hardware_inputs |=
1336 URP_HARDWARE_INPUT_PARALLEL_CARRIER;
1337 } else if ((o->pps[i]) &&
1338 (!strcasecmp(o->pps[i], "ctcss"))) {
1339 if (k & (1 << ppinshift[i]))
1340 hardware_inputs |=
1341 URP_HARDWARE_INPUT_PARALLEL_CTCSS;
1342 }
1343 }
1344 }
1345 usbradioplus_publish_hardware_inputs(o, hardware_inputs);
1346 j = ast_tvdiff_ms(ast_radio_tvnow(), then);
1347 /* make output inversion mask (for pulseage) */
1348 o->hid_gpio_lastmask = o->hid_gpio_pulsemask;
1349 o->hid_gpio_pulsemask = 0;
1350 for (i = 0; i < GPIO_PINCOUNT; i++) {
1351 k = o->hid_gpio_pulsetimer[i];
1352 if (k) {
1353 k -= j;
1354 if (k < 0) {
1355 k = 0;
1356 }
1357 o->hid_gpio_pulsetimer[i] = k;
1358 }
1359 if (k) {
1360 o->hid_gpio_pulsemask |= 1 << i;
1361 }
1362 }
1363 if (atomic_exchange_explicit(&o->plus_clip_led_request, 0,
1364 memory_order_acq_rel) &&
1365 o->clipledgpio && !o->hid_gpio_pulsetimer[o->clipledgpio - 1]) {
1366 /* The callback only posts this request; timing and GPIO state
1367 * remain entirely owned by this hardware worker. */
1368 o->hid_gpio_pulsetimer[o->clipledgpio - 1] = CLIP_LED_HOLD_TIME_MS;
1369 }
1370 if (o->hid_gpio_pulsemask ||
1371 o->hid_gpio_lastmask) { /* if anything inverted (temporarily) */
1372 buf[o->hid_gpio_loc] = o->hid_gpio_val ^ o->hid_gpio_pulsemask;
1373 buf[o->hid_gpio_ctl_loc] = o->hid_gpio_ctl;
1374 ast_radio_hid_set_outputs(usb_handle, buf);
1375 }
1376 if (o->gpio_set) {
1377 o->gpio_set = 0;
1378 buf[o->hid_gpio_loc] = o->hid_gpio_val ^ o->hid_gpio_pulsemask;
1379 buf[o->hid_gpio_ctl_loc] = o->hid_gpio_ctl;
1380 ast_radio_hid_set_outputs(usb_handle, buf);
1381 }
1382 {
1383 const int desired_ptt = atomic_load_explicit(
1384 &o->plus_hardware_ptt_request, memory_order_acquire);
1385 const int have_program = usbradioplus_read_radio_program_request(
1386 o, &program_request);
1387 const int program_changed =
1388 have_program &&
1389 program_request.generation != program_generation;
1390 const int ptt_changed = o->lasttx != !!desired_ptt;
1391 /* A missing in-progress program snapshot must never defer PTT
1392 * release. */
1393 const int reprogram =
1394 have_program && (program_changed || ptt_changed);
1395
1396 if (program_changed || ptt_changed) {
1397 hidthread_apply_ptt(
1398 o, usb_handle, buf, bufsave, desired_ptt,
1399 have_program ? &program_request : NULL, reprogram);
1400 if (have_program)
1401 program_generation = program_request.generation;
1402 }
1403 }
1404 ast_radio_time(&o->lasthidtime);
1405 ast_mutex_unlock(&o->usblock);
1406 }
1407 /* Device loss, a stop request, and an audio-worker fault all leave this
1408 * loop through one physical-unkey path. */
1409 hidthread_failsafe_unkey(o, usb_handle, buf);
1410 }
1411 /* clean up before exiting the thread */
1412 hidthread_failsafe_unkey(o, usb_handle, buf);
1413 if (usb_handle) {
1414 /* Hangup joins this thread; release HID so the next call can reopen it. */
1415 libusb_close(usb_handle);
1416 usb_handle = NULL;
1417 }
1418
1419 usbradio_release_device(o);
1420
1421 return NULL;
1422}

1430static int soundcard_write_admitted_frame(struct chan_usbradio_pvt *o, short *data, int *submitted)
1431{
1432 PaError res;
1433 int audio_bearing;
1434
1435 if (submitted)
1436 *submitted = 0;
1437
1438 if (!o->pa.active) {
1439 if (usbradio_start_audio(o) < 0) {
1440 return 0;
1441 }
1442 }
1443
1444 /*
1445 * Always write something so the USB TX buffer does not intentionally
1446 * underrun (same idea as simpleusb #1161). Radio PTT is HID-gated;
1447 * when unkeyed, feed silence instead of the rendered TX buffer.
1448 */
1449 /* The audio worker publishes this before releasing its parser-state read
1450 * lease. Keeping the output decision separate lets a reload emit silence
1451 * without dereferencing a radio object being reparsed by the control plane. */
1452 audio_bearing = atomic_load_explicit(&o->plus_radio_tx_active, memory_order_acquire) &&
1453 usbradioplus_pcm_has_audio(data, AST_RADIO_PA_48K_STEREO_SAMPLES);
1454 if (!audio_bearing) {
1455 data = silence_buf;
1456 }
1457
1458 /*
1459 * ast_radio_pa_write() already primes one silence frame on
1460 * paOutputUnderflowed (#593 / #598). Treat that as success here.
1461 */
1462 res = ast_radio_pa_write(&o->pa, data, AST_RADIO_PA_FRAMES_PER_BUFFER);
1463 if (res < 0 && res != paOutputUnderflowed) {
1464 usbradio_log_fault(
1465 o, 0, "Channel %s: PortAudio write failed (%s); restarting audio stream\n",
1466 o->name, Pa_GetErrorText(res));
1467 /* Force HID reinit so recovery log clears the latch. */
1468 o->hasusb = 0;
1469 ast_radio_pa_stop(&o->pa);
1470 return 0;
1471 }
1472 if (res == paNoError) {
1473 usbradioplus_tx_playout_hold_note_output(
1474 o, 1, audio_bearing,
1475 audio_bearing ? o->plus_portaudio_playout_hold_callbacks : 0U);
1476 if (submitted)
1477 *submitted = 1;
1478 }
1479
1480 return AST_RADIO_PA_FRAMES_PER_BUFFER * AST_RADIO_PA_OUTPUT_CHANNELS * (int)sizeof(short);
1481}

1515URP_CHANNEL_LOCAL int usbradio_text(struct ast_channel *c, const char *text)
1516{
1517 struct chan_usbradio_pvt *o = ast_channel_tech_pvt(c);
1518 char cmd[16], pwr;
1519 int cnt, i, j;
1520 double tx, rx;
1521
1522#define STR_SZ 15 /* Size of text strings */
1523 char rxs[STR_SZ + 1], txs[STR_SZ + 1], txpl[STR_SZ + 1], rxpl[STR_SZ + 1];
1524 if (!o) {
1525 return -1;
1526 }
1527
1528#ifdef HAVE_SYS_IO
1529 if (haspp == 2) {
1530 ioperm(pbase, 2, 1);
1531 }
1532#endif
1533
1534 cmd[0] = rxs[0] = txs[0] = rxpl[0] = txpl[0] = pwr = '\0';
1535
1536 /* print received messages */
1537 ast_debug(3, "Channel %s: Console Received usbradio text %s >>\n", o->name, text);
1538
1539 cnt = sscanf(text, "%15s " S_FMT(STR_SZ) S_FMT(STR_SZ) S_FMT(STR_SZ) S_FMT(STR_SZ) "%c",
1540 cmd, rxs, txs, rxpl, txpl, &pwr);
1541
1542 /* set channel on parallel port */
1543 if (strcmp(cmd, "SETCHAN") == 0) {
1544 u8 chan;
1545 chan = strtod(rxs, NULL);
1546 usbradioplus_set_channel(chan);
1547 ast_debug(3, "Channel %s: SETCHAN cmd: %s chan: %i\n", o->name, text, chan);
1548 return 0;
1549 }
1550
1551 /* set receive CTCSS */
1552 if (strcmp(cmd, "RXCTCSS") == 0) {
1553 u8 x;
1554 x = strtod(rxs, NULL);
1555 o->rxctcssoverride = !x;
1556 ast_debug(3, "Channel %s: RXCTCSS cmd: %s\n", o->name, text);
1557 return 0;
1558 }
1559
1560 /* set transmit CTCSS (app_rpt itxctcss → TXCTCSS 0/1) */
1561 if (strcmp(cmd, "TXCTCSS") == 0) {
1562 if (cnt < 2 || (strcmp(rxs, "0") && strcmp(rxs, "1"))) {
1563 ast_log(LOG_WARNING, "Channel %s: Invalid TXCTCSS command: %s\n", o->name,
1564 text);
1565 return 0;
1566 }
1567 if (o->radio) {
1568 o->radio->b.txCtcssOff = rxs[0] != '1';
1569 }
1570 ast_debug(3, "Channel %s: TXCTCSS cmd: %s\n", o->name, text);
1571 return 0;
1572 }
1573
1574 /* GPIO command */
1575 if (!strncmp(text, "GPIO", 4)) {
1576 cnt = sscanf(text, "%15s " N_FMT(d) " " N_FMT(d), cmd, &i, &j);
1577 if (cnt < 3) {
1578 return 0;
1579 }
1580 if ((i < 1) || (i > GPIO_PINCOUNT)) {
1581 return 0;
1582 }
1583 i--;
1584 /* skip if not valid */
1585 if (!(o->valid_gpios & (1 << i))) {
1586 return 0;
1587 }
1588 ast_mutex_lock(&o->usblock);
1589 if (j > 1) { /* if to request pulse-age */
1590 o->hid_gpio_pulsetimer[i] = j - 1;
1591 } else {
1592 /* clear pulsetimer, if in the middle of running */
1593 o->hid_gpio_pulsetimer[i] = 0;
1594 o->hid_gpio_val &= ~(1 << i);
1595 if (j) {
1596 o->hid_gpio_val |= 1 << i;
1597 }
1598 o->gpio_set = 1;
1599 }
1600 ast_mutex_unlock(&o->usblock);
1601 kickptt(o);
1602 return 0;
1603 }
1604
1605 /* Parallel port command */
1606 if (!strncmp(text, "PP", 2)) {
1607 cnt = sscanf(text, "%15s " N_FMT(d) " " N_FMT(d), cmd, &i, &j);
1608 if (cnt < 3) {
1609 return 0;
1610 }
1611 if ((i < 2) || (i > 9)) {
1612 return 0;
1613 }
1614 ast_mutex_lock(&pp_lock);
1615 if (j > 1) { /* if to request pulse-age */
1616 pp_pulsetimer[i] = j - 1;
1617 } else {
1618 /* clear pulsetimer, if in the middle of running */
1619 pp_pulsetimer[i] = 0;
1620 pp_val &= ~(1 << (i - 2));
1621 if (j) {
1622 pp_val |= 1 << (i - 2);
1623 }
1624 ast_radio_ppwrite(haspp, ppfd, pbase, pport, pp_val);
1625 }
1626 ast_mutex_unlock(&pp_lock);
1627 return 0;
1628 }
1629
1630 if (cnt < 6) {
1631 ast_log(LOG_ERROR, "Channel %s: Cannot parse usbradio text: %s\n", o->name, text);
1632 return 0;
1633 } else {
1634 ast_debug(3, "Channel %s: << %s %s %s %s %s %c >> \n", o->name, cmd, rxs, txs, rxpl,
1635 txpl, pwr);
1636 }
1637
1638 /* set frequency command */
1639 if (strcmp(cmd, "SETFREQ") == 0) {
1640 ast_debug(3, "Channel %s: SETFREQ cmd: %s\n", o->name, text);
1641 tx = strtod(txs, NULL);
1642 rx = strtod(rxs, NULL);
1643 o->set_txfreq = round(tx * (double)1000000);
1644 o->set_rxfreq = round(rx * (double)1000000);
1645 o->set_txpower = (pwr == 'H');
1646 ast_copy_string(o->set_rxctcssfreqs, rxpl, sizeof(o->set_rxctcssfreqs));
1647 ast_copy_string(o->set_txctcssfreqs, txpl, sizeof(o->set_txctcssfreqs));
1648
1649 o->remoted = 1;
1650 radio_config(o);
1651 return 0;
1652 }
1653 ast_log(LOG_ERROR, "Channel %s: Cannot parse usbradio cmd: %s\n", o->name, text);
1654 return 0;
1655}

1658{
1659 (void)dest;
1660 (void)timeout;
1661 struct chan_usbradio_pvt *o = ast_channel_tech_pvt(c);
1662
1663 if (o->hidthread == AST_PTHREADT_NULL) {

1681 o->hidthread = AST_PTHREADT_NULL;
1682 return -1;
1683 }
1684 }
1685 ast_setstate(c, AST_STATE_UP);
1686 return 0;
1687}

1759URP_CHANNEL_LOCAL void *usbradio_audio_thread(void *arg)
1760{
1761 PaError pa_res;
1762 int cd, sd;
1763 int tx_write_ready;
1764 long frames_available;
1765 struct chan_usbradio_pvt *o = arg;
1766 struct ast_frame *f = &o->read_f, *f1;
1767 time_t now;
1768
1769 ast_debug(5, "Audio thread is starting\n");
1770 ast_radio_time(&o->lastaudiotime);
1771
1772 while (!o->stopaudiothread) {
1773 ast_radio_time(&o->lastaudiotime);
1774
1775 if (!o->hasusb) {
1776 usbradio_swap_audio_stopped(o);
1777 usleep(DEVICE_RETRY);
1778 continue;
1779 }
1780
1781 if (!o->pa.active && usbradio_start_audio(o) < 0) {
1782 usbradio_log_fault(o, 0, "Channel %s: Failed to start audio stream\n",
1783 o->name);
1784 o->hasusb = 0;
1785 usleep(DEVICE_RETRY);
1786 continue;
1787 }
1788
1789 /* Prime one silence frame so a late first wake-up does not underrun. */
1790 ast_radio_pa_write(&o->pa, silence_buf, AST_RADIO_PA_FRAMES_PER_BUFFER);
1791 o->audio_thread_ready = 1;
1792
1793 while (!o->stopaudiothread && o->hasusb) {
1794 if (o->lasthidtime) {
1795 ast_radio_time(&now);
1796 if ((now - o->lasthidtime) > 3) {
1797 usbradio_log_fault(o, 0,
1798 "Channel %s: HID process has died or is "
1799 "not responding.\n",
1800 o->name);
1801 o->hasusb = 0;
1802 stream_cleanup(o);
1803 break;
1804 }
1805 }
1806
1807 ast_radio_time(&o->lastaudiotime);
1808 memset(f, 0, sizeof(struct ast_frame));
1809 f->frametype = AST_FRAME_NULL;
1810 f->src = __PRETTY_FUNCTION__;
1811
1812 if (!o->hasusb) {
1813 if (o->rxkeyed) {
1814 struct ast_frame wf = {
1815 .frametype = AST_FRAME_CONTROL,
1816 .subclass.integer = AST_CONTROL_RADIO_UNKEY,
1817 .src = __PRETTY_FUNCTION__,
1818 };
1819
1820 o->lastrx = 0;
1821 o->rxkeyed = 0;
1822 if (o->owner) {
1823 ast_queue_frame(o->owner, &wf);
1824 }
1825 if (o->duplex3) {
1826 usbradio_set_sidetone_switch(o, 0);
1827 }
1828 }
1829 stream_cleanup(o);
1830 break;
1831 }
1832
1833 /* One PortAudio input block for loop pacing */
1834 pa_res = usbradio_read_pa_stereo(o);
1835 if (pa_res != paNoError) {
1836 if (pa_res == paTimedOut || pa_res == paInputOverflowed) {
1837 /* RX silence when no input block is available */
1838 memset(o->usbradio_read_buf + AST_FRIENDLY_OFFSET, 0,
1839 sizeof(o->usbradio_read_buf) - AST_FRIENDLY_OFFSET);
1840 } else {
1841 usbradio_log_fault(o, 0,
1842 "Channel %s: PortAudio read failed "
1843 "(%s); restarting audio stream\n",
1844 o->name, Pa_GetErrorText(pa_res));
1845 o->hasusb = 0;
1846 stream_cleanup(o);
1847 break;
1848 }
1849 }
1850
1851 /* If we have stopped echoing, clear the echo queue */
1852 if (o->plus_advanced || !o->echomode) {
1853 usbradioplus_echo_clear(o);
1854 }
1855
1856 /* PortAudio output capacity check */
1857 frames_available = ast_radio_pa_write_available(&o->pa);
1858 if (frames_available < 0) {
1859 ast_debug(2, "Channel %s: Pa_GetStreamWriteAvailable error %s\n",
1860 o->name, Pa_GetErrorText(frames_available));
1861 o->usb_faulted = 1;
1862 o->hasusb = 0;
1863 stream_cleanup(o);
1864 break;
1865 }
1866 tx_write_ready = frames_available >= AST_RADIO_PA_FRAMES_PER_BUFFER;
1867
1868 /* Legacy app_rpt echo remains a distinct source queue. */
1869 if (!o->plus_advanced && tx_write_ready && o->echomode &&
1870 !usbradioplus_native_echo(o) && (!o->rxkeyed)) {
1871 (void)usbradioplus_echo_start(o);
1872 }
1873
1874 o->readerrs = 0;
1875 o->readpos = sizeof(o->usbradio_read_buf);
1876
1877 if (ast_radio_check_audio(
1878 (short *)(o->usbradio_read_buf + AST_FRIENDLY_OFFSET),
1879 &o->rxaudiostats, AST_RADIO_PA_48K_STEREO_SAMPLES, 0)) {
1880 usbradioplus_request_clip_led(o);
1881 }
1882
1883 /* CTCSS/DCS code maps own parser-allocated state. A processing reload
1884 * never waits in this hardware-paced worker: one quiet block is safer
1885 * than dereferencing a map while the control plane replaces it. */
1886 if (!usbradioplus_radio_access_acquire(o)) {
1887 memset(o->usbradio_read_buf_8k + AST_FRIENDLY_OFFSET, 0,
1888 o->plus_app_rpt_samples * sizeof(short));
1889 memset(o->usbradio_write_buf, 0, sizeof(o->usbradio_write_buf));
1890 atomic_store_explicit(&o->plus_radio_tx_active, 0,
1891 memory_order_release);
1892 if (tx_write_ready &&
1893 !soundcard_writeframe(o, o->usbradio_write_buf)) {
1894 stream_cleanup(o);
1895 break;
1896 }
1897 o->readpos = AST_FRIENDLY_OFFSET;
1898 continue;
1899 }
1900 /* HID sampling and physical PTT completion are independent workers.
1901 * Import their atomic snapshots before the audio-owned signaling tick. */
1902 usbradioplus_audio_load_hardware_state(o);
1903 /* Only app_rpt and tuning own PTT. */
1904 if (atomic_load_explicit(&o->txkeyed, memory_order_acquire) ||
1905 atomic_load_explicit(&o->txtestkey, memory_order_acquire)) {
1906 if (!o->radio->txPttIn) {
1907 o->radio->txPttIn = 1;
1908 }
1909 } else if (o->radio->txPttIn) {
1910 o->radio->txPttIn = 0;
1911 }
1912 usbradioplus_prepare_squelch_audio(o);
1913 /* A full PortAudio queue freezes TX state. Restoring raw PTT before a
1914 * non-advancing tick could release audio already queued for playout. */
1915 if (tx_write_ready)
1916 usbradioplus_tx_playout_hold_prepare(o);
1917 urp_radio_process_timed(
1918 o->radio, o->plus_squelch_native,
1919 (i16 *)(o->usbradio_read_buf_8k + AST_FRIENDLY_OFFSET),

2339int usb_device_swap(int fd, const char *other)
2340{
2341 struct chan_usbradio_pvt *p, *o;
2342 int attempts;
2343 int result;
2344
2345 if (!other) {
2346 return -1;
2347 }
2348 o = find_desc(usbradio_active);
2349 if (o == NULL) {
2350 return -1;
2351 }
2352 p = find_desc(other);
2353 if (p == NULL) {
2354 ast_cli(fd, "USB Device %s not found\n", other);
2355 return -1;
2356 }
2357 if (p == o) {
2358 ast_cli(fd, "You can't swap active device with itself!!\n");
2359 return -1;
2360 }
2361
2362 /* Serialize the complete device swap transaction */
2363 ast_mutex_lock(&device_swap_lock);
2364 if (!o->hasusb || !p->hasusb) {
2365 ast_cli(fd, "Both channels must have assigned USB devices before they can be "
2366 "swapped.\n");
2367 ast_mutex_unlock(&device_swap_lock);
2368 return -1;
2369 }
2370
2371 /* Stop both device handles while retaining their shared leases */
2372 usbradio_swap_begin(o);
2373 usbradio_swap_begin(p);
2374 o->hasusb = 0;
2375 p->hasusb = 0;
2376 kickptt(o);
2377 kickptt(p);
2378
2379 /* Wait for the HID and PortAudio handles to close */
2380 for (attempts = 0; attempts < 500; attempts++) {
2381 if (usbradio_swap_ready(o) && usbradio_swap_ready(p)) {
2382 break;
2383 }
2384 usleep(10000);
2385 }
2386 if (attempts == 500) {
2387 usbradio_swap_finish(o);
2388 usbradio_swap_finish(p);
2389 ast_cli(fd, "Timed out waiting for both USB devices to stop.\n");
2390 ast_mutex_unlock(&device_swap_lock);
2391 return -1;
2392 }
2393
2394 /* Exchange the parked leases and use their identities for future requests */
2395 ast_mutex_lock(&o->device_lock);
2396 ast_mutex_lock(&p->device_lock);
2397 result = ast_radio_device_swap(&o->radio_device, &p->radio_device);
2398 if (!result) {
2399 ast_copy_string(o->devstr, o->radio_device->devstr, sizeof(o->devstr));
2400 ast_copy_string(o->serial, S_OR(o->radio_device->serial, ""), sizeof(o->serial));
2401 ast_copy_string(p->devstr, p->radio_device->devstr, sizeof(p->devstr));
2402 ast_copy_string(p->serial, S_OR(p->radio_device->serial, ""), sizeof(p->serial));
2403 }
2404 ast_mutex_unlock(&p->device_lock);
2405 ast_mutex_unlock(&o->device_lock);
2406 usbradio_swap_finish(o);
2407 usbradio_swap_finish(p);
2408
2409 if (result) {
2410 ast_cli(fd, "Unable to exchange USB device leases.\n");
2411 ast_mutex_unlock(&device_swap_lock);
2412 return -1;
2413 }
2414 ast_cli(fd, "USB Devices successfully swapped.\n");
2415 ast_mutex_unlock(&device_swap_lock);
2416 return 0;
2417}
