82AVFILTER_PRIVATE int graph_append(char *graph, size_t size, const char *format, ...)
83{
84 va_list ap;
85 size_t used = strlen(graph);
86 int result;
87
88 if (used >= size) {
89 return AVERROR(ENOSPC);
90 }
91 va_start(ap, format);
92 result = vsnprintf(graph + used, size - used, format, ap);
93 va_end(ap);
94 if (result < 0 || (size_t)result >= size - used) {
95 return AVERROR(ENOSPC);
96 }
97 return 0;

167AVFILTER_PRIVATE void update_cleanup_astats(struct txagc_avfilter *state, const AVFrame *frame,
168 enum cleanup_meter meter)
169{
170 double peak;
171 double rms;
172 double *current;
173 double *maximum;
174
175 if (astats_value(frame, "lavfi.astats.Overall.RMS_level", &rms)) {
176 return;
177 }
178 if (meter == CLEANUP_PRE_FULL) {
179 if (astats_value(frame, "lavfi.astats.Overall.Peak_level", &peak)) {
180 return;
181 }
182 state->cleanup_pre_peak_dbfs = peak;
183 state->cleanup_pre_rms_dbfs = rms;
184 if (peak > state->cleanup_pre_max_peak_dbfs) {
185 state->cleanup_pre_max_peak_dbfs = peak;
186 }
187 if (rms > state->cleanup_pre_max_rms_dbfs) {
188 state->cleanup_pre_max_rms_dbfs = rms;
189 }
190 return;
191 }
192 switch (meter) {
193 case CLEANUP_PRE_5_8:
194 current = &state->cleanup_pre_5_8_rms_dbfs;
195 maximum = &state->cleanup_pre_5_8_max_rms_dbfs;
196 break;
197 case CLEANUP_PRE_8_PLUS:
198 current = &state->cleanup_pre_8_plus_rms_dbfs;
199 maximum = &state->cleanup_pre_8_plus_max_rms_dbfs;
200 break;
201 case CLEANUP_POST_5_8:
202 current = &state->cleanup_post_5_8_rms_dbfs;
203 maximum = &state->cleanup_post_5_8_max_rms_dbfs;
204 break;
205 default:
206 current = &state->cleanup_post_8_plus_rms_dbfs;
207 maximum = &state->cleanup_post_8_plus_max_rms_dbfs;
208 break;
209 }
210 *current = rms;
211 if (rms > *maximum) {
212 *maximum = rms;
213 }

282AVFILTER_PRIVATE int add_emphasis(char *graph, size_t size, const char *input, const char *output,
283 int production, double corner_hz, double reference_hz,
284 unsigned int sample_rate)
285{
286 double pole = exp(-2.0 * M_PI * corner_hz / sample_rate);
287 double omega = 2.0 * M_PI * reference_hz / sample_rate;
288 double inverse_at_reference =
289 sqrt(1.0 + pole * pole - 2.0 * pole * cos(omega)) / (1.0 - pole);
290 if (production) {
291 double scale = 1.0 / inverse_at_reference;
292 return graph_append(graph, size,
293 "[%s]biquad=b0=%.17g:b1=%.17g:b2=0:a0=1:a1=0:a2=0:"
294 "precision=f64[%s];",
295 input, scale / (1.0 - pole), -pole * scale / (1.0 - pole),
296 output);
297 }
298 return graph_append(graph, size,
299 "[%s]biquad=b0=%.17g:b1=0:b2=0:a0=1:a1=%.17g:a2=0:"
300 "precision=f64[%s];",
301 input, inverse_at_reference * (1.0 - pole), -pole, output);
302}

326static int add_three_band_dynamics(char *graph, size_t size, const char *input, const char *output,
327 const char *prefix, double low_crossover, double high_crossover,
328 const struct dynamics_band bands[3], const char *detection)
329{
330 static const char *const names[] = {"lo", "mid", "hi"};
331 int result;
332
333 /* Complementary crossover outputs retain unity summed response. Each band's
334 * detector follows its own program audio; no speech sidechain trims a band. */
335 result = graph_append(graph, size,
336 "[%s]acrossover=split=%.9g %.9g:order=4th[%slo][%smid][%shi];", input,
337 low_crossover, high_crossover, prefix, prefix, prefix);
338 if (result < 0)
339 return result;
340 for (unsigned int index = 0; index < 3; ++index) {
341 const struct dynamics_band *band = &bands[index];
342 char makeup[80] = "";
343 /* FFmpeg's compressor makeup cannot attenuate. A post-compression
344 * gain supports the configured negative gains without changing detection. */
345 if (band->makeup_gain_db != 0.0)
346 snprintf(makeup, sizeof(makeup), ",volume=%.12g:precision=double",
347 db_to_linear(band->makeup_gain_db));
348 result = graph_append(graph, size,
349 "[%s%s]acompressor=mode=downward:threshold=%.12g:ratio=%.9g:"
350 "knee=%.12g:attack=%.9g:release=%.9g:detection=%s%s[%s%sc];",
351 prefix, names[index], db_to_linear(band->threshold_dbfs),
352 clamp(band->ratio, 1.0, 20.0), db_to_linear(band->knee_db),
353 band->attack_ms, band->release_ms, detection, makeup, prefix,
354 names[index]);
355 if (result < 0)
356 return result;
357 }
358 return graph_append(graph, size, "[%sloc][%smidc][%shic]amix=inputs=3:normalize=0[%s];",
359 prefix, prefix, prefix, output);
360}

372AVFILTER_PRIVATE int add_dynamic_stage(char *graph, size_t size, const struct txagc_config *cfg,
373 enum txagc_stage kind, const char *current, const char *next,
374 unsigned int serial)
375{
376 char options[1024];
377 char prefix[32];
378
379 snprintf(prefix, sizeof(prefix), "d%u", serial);
380 switch (kind) {
381 case TXAGC_STAGE_DEESSER: {
382 double octave_ratio;
383 double q;
384 if (!cfg->deesser_enabled)
385 return 0;
386 octave_ratio = pow(2.0, cfg->deesser_width_octaves);
387 q = sqrt(octave_ratio) / (octave_ratio - 1.0);
388#if LIBAVFILTER_VERSION_MAJOR >= 9
389 return graph_append(
390 graph, size,
391 "[%s]adynamicequalizer=threshold=%.12g:dfrequency=%.9g:"
392 "dqfactor=%.12g:tfrequency=%.9g:tqfactor=%.12g:"
393 "attack=%.9g:release=%.9g:ratio=%.9g:range=%.12g:"
394 "mode=cutabove:dftype=bandpass:tftype=bell:precision=double[%s];",
395 current, db_to_linear(cfg->deesser_threshold_dbfs),
396 cfg->deesser_frequency_hz, q, cfg->deesser_frequency_hz, q,
397 cfg->deesser_attack_ms, cfg->deesser_release_ms, cfg->deesser_ratio,
398 cfg->deesser_max_reduction_db / 2.0, next);
399#else
400 double ratio = cfg->deesser_ratio;
401 double span = -cfg->deesser_threshold_dbfs;
402 double detector_reduction = cfg->deesser_max_reduction_db / 2.0;
403 /* FFmpeg 5.1 does not apply range to cuts. Cap its ratio so a
404 * full-scale detector input cannot exceed the requested reduction.
405 * Its bell response applies the detector gain twice at band center. */
406 if (span > detector_reduction) {
407 double limited_ratio = span / (span - detector_reduction);
408 if (ratio > limited_ratio)
409 ratio = limited_ratio;
410 }
411 return graph_append(graph, size,
412 "[%s]adynamicequalizer=threshold=%.12g:dfrequency=%.9g:"
413 "dqfactor=%.12g:tfrequency=%.9g:tqfactor=%.12g:"
414 "attack=%.9g:release=%.9g:ratio=%.9g:range=%.12g:"
415 "mode=cut:tftype=bell[%s];",
416 current, db_to_linear(cfg->deesser_threshold_dbfs),
417 cfg->deesser_frequency_hz, q, cfg->deesser_frequency_hz, q,
418 cfg->deesser_attack_ms, cfg->deesser_release_ms, ratio, 0.0,
419 next);
420#endif
421 }
422 case TXAGC_STAGE_EQUALIZER:
423 if (!cfg->equalizer_enabled)
424 return 0;
425 return graph_append(graph, size,
426 "[%s]bass=g=%.9g:f=%.9g:t=s:w=%.9g:r=f64,"
427 "equalizer=f=%.9g:t=o:w=%.9g:g=%.9g:r=f64,"
428 "treble=g=%.9g:f=%.9g:t=s:w=%.9g:r=f64[%s];",
429 current, cfg->equalizer_low_gain_db,
430 cfg->equalizer_low_frequency_hz, cfg->equalizer_low_slope,
431 cfg->equalizer_mid_frequency_hz,
432 cfg->equalizer_mid_width_octaves, cfg->equalizer_mid_gain_db,
433 cfg->equalizer_high_gain_db, cfg->equalizer_high_frequency_hz,
434 cfg->equalizer_high_slope, next);
435 case TXAGC_STAGE_AGC:
436 if (!cfg->agc_enabled)
437 return 0;
438 /* Pack program and detector into explicit channels for the LADSPA
439 * two-input/one-output effect. Only its detector sees the filters.
440 * FFmpeg 5 needs the effect's single output labeled explicitly as mono. */
441 snprintf(options, sizeof(options),
442 "inputs=2:channel_layout=stereo:map=0.0-FL|1.0-FR,"
443 "ladspa=file='%s':plugin=usbradioplus_agc:controls="
444 "c0=%.12g|c1=%.12g|c2=%.12g|c3=%.12g|c4=%.12g|c5=%.12g|"
445 "c6=%.12g|c7=%.12g|c8=%.12g|c9=%.12g,"
446 "aformat=channel_layouts=mono",
447 URP_AGC_PLUGIN_PATH, cfg->target_dbfs, cfg->agc_rms_averaging_ms,
448 cfg->agc_gain_increase_db_per_second, cfg->agc_gain_decrease_db_per_second,
449 cfg->max_gain_db, cfg->max_attenuation_db,
450 cfg->agc_activity_threshold_dbfs, cfg->agc_activity_hysteresis_db,
451 cfg->agc_hold_ms, cfg->agc_deadband_db);
452 return add_sidechain_stage(graph, size, current, next, prefix, "join",
453 cfg->sidechain_highpass_hz, cfg->sidechain_lowpass_hz,
454 options);
455 case TXAGC_STAGE_EXPANDER:
456 if (!cfg->expander_enabled)
457 return 0;
458 snprintf(options, sizeof(options),
459 "threshold=%.12g:ratio=%.9g:range=%.12g:attack=%.9g:"
460 "release=%.9g:knee=2.828427:detection=rms",
461 db_to_linear(cfg->expander_threshold_dbfs),
462 clamp(cfg->expander_ratio, 1.0, 20.0),
463 db_to_linear(-fabs(cfg->expander_max_attenuation_db)),
464 cfg->expander_attack_ms, cfg->expander_release_ms);
465 return add_sidechain_stage(graph, size, current, next, prefix, "sidechaingate",
466 cfg->expander_sidechain_highpass_hz,
467 cfg->expander_sidechain_lowpass_hz, options);
468 case TXAGC_STAGE_COMPRESSOR: {
469 const struct dynamics_band bands[] = {
470 {cfg->compressor_low_threshold_dbfs, cfg->compressor_low_ratio,
471 cfg->compressor_low_makeup_gain_db, cfg->compressor_low_knee_db,
472 cfg->compressor_low_attack_ms, cfg->compressor_low_release_ms},
473 {cfg->compressor_mid_threshold_dbfs, cfg->compressor_mid_ratio,
474 cfg->compressor_mid_makeup_gain_db, cfg->compressor_mid_knee_db,
475 cfg->compressor_mid_attack_ms, cfg->compressor_mid_release_ms},
476 {cfg->compressor_high_threshold_dbfs, cfg->compressor_high_ratio,
477 cfg->compressor_high_makeup_gain_db, cfg->compressor_high_knee_db,
478 cfg->compressor_high_attack_ms, cfg->compressor_high_release_ms},
479 };
480 if (!cfg->compressor_enabled)
481 return 0;
482 if (cfg->compressor_bands == 3)
483 return add_three_band_dynamics(graph, size, current, next, prefix,
484 cfg->compressor_low_crossover_hz,
485 cfg->compressor_high_crossover_hz, bands,
486 "rms");
487 snprintf(options, sizeof(options),
488 "mode=downward:threshold=%.12g:ratio=%.9g:attack=%.9g:"
489 "release=%.9g:knee=2.828427:detection=rms,volume=%.12g:precision=double",
490 db_to_linear(cfg->compressor_threshold_dbfs),
491 clamp(cfg->compressor_ratio, 1.0, 20.0), cfg->compressor_attack_ms,
492 cfg->compressor_release_ms, db_to_linear(cfg->compressor_makeup_gain_db));
493 return add_sidechain_stage(graph, size, current, next, prefix, "sidechaincompress",
494 cfg->compressor_sidechain_highpass_hz,
495 cfg->compressor_sidechain_lowpass_hz, options);
496 }
497 case TXAGC_STAGE_LIMITER: {
498 const struct dynamics_band bands[] = {
499 {cfg->low_limiter_threshold_dbfs, cfg->low_limiter_ratio, 0.0,
500 cfg->low_limiter_knee_db, cfg->low_limiter_attack_ms,
501 cfg->low_limiter_release_ms},
502 {cfg->mid_limiter_threshold_dbfs, cfg->mid_limiter_ratio, 0.0,
503 cfg->mid_limiter_knee_db, cfg->mid_limiter_attack_ms,
504 cfg->mid_limiter_release_ms},
505 {cfg->high_limiter_threshold_dbfs, cfg->high_limiter_ratio, 0.0,
506 cfg->high_limiter_knee_db, cfg->high_limiter_attack_ms,
507 cfg->high_limiter_release_ms},
508 };
509 if (!cfg->limiter_enabled)
510 return 0;
511 if (cfg->limiter_bands == 3)
512 return add_three_band_dynamics(
513 graph, size, current, next, prefix, cfg->limiter_low_crossover_hz,
514 cfg->limiter_high_crossover_hz, bands, "peak");
515 return graph_append(graph, size,
516 "[%s]acompressor=mode=downward:threshold=%.12g:ratio=%.9g:"
517 "knee=%.12g:attack=%.9g:release=%.9g:detection=peak[%s];",
518 current, db_to_linear(cfg->limiter_threshold_dbfs),
519 clamp(cfg->limiter_ratio, 1.0, 20.0),
520 db_to_linear(cfg->limiter_knee_db), cfg->limiter_attack_ms,
521 cfg->limiter_release_ms, next);
522 }
523 }
524 return AVERROR(EINVAL);
525}

534AVFILTER_PRIVATE int build_description(char *graph, size_t size, const struct txagc_config *cfg,
535 unsigned int sample_rate)
536{
537 char current[NAME_SIZE] = "s0";
538 const char *graph_input = "in";
539 char next[NAME_SIZE];
540 unsigned int stage = 1;
541 unsigned int order_index;
542
543 graph[0] = '\0';
544 /* A link can run at a lower sample rate than the native radio chain. Reject
545 * impossible band definitions instead of silently moving their crossovers. */
546 if (cfg->compressor_enabled && cfg->compressor_bands == 3 &&
547 cfg->compressor_high_crossover_hz >= sample_rate * 0.5) {
548 av_log(NULL, AV_LOG_ERROR,
549 "USBRadioPlus: compressor crossovers %.9g/%.9g Hz must be below "
550 "Nyquist %.9g Hz at %u Hz sample rate\n",
551 cfg->compressor_low_crossover_hz, cfg->compressor_high_crossover_hz,
552 sample_rate * 0.5, sample_rate);
553 return AVERROR(EINVAL);
554 }
555 if (cfg->limiter_enabled && cfg->limiter_bands == 3 &&
556 cfg->limiter_high_crossover_hz >= sample_rate * 0.5) {
557 av_log(NULL, AV_LOG_ERROR,
558 "USBRadioPlus: limiter crossovers %.9g/%.9g Hz must be below "
559 "Nyquist %.9g Hz at %u Hz sample rate\n",
560 cfg->limiter_low_crossover_hz, cfg->limiter_high_crossover_hz,
561 sample_rate * 0.5, sample_rate);
562 return AVERROR(EINVAL);
563 }
564 if (cfg->deemphasis_enabled) {
565 if (add_emphasis(graph, size, "in", "deemphasized", 0, cfg->emphasis_corner_hz,
566 cfg->emphasis_reference_hz, sample_rate) < 0) {
567 return AVERROR(ENOSPC);
568 }
569 graph_input = "deemphasized";
570 }
571 if (cfg->receive_bandpass_enabled) {
572 if (add_brickwall_bandpass(graph, size, graph_input, "rxbandpass", "rxbp",
573 cfg->receive_bandpass_highpass_hz,
574 cfg->receive_bandpass_lowpass_hz) < 0) {
575 return AVERROR(ENOSPC);
576 }
577 graph_input = "rxbandpass";
578 }
579 if (cfg->ctcss_filter_mode == TXAGC_CTCSS_FILTER_NOTCH) {
580 const char *cursor = cfg->ctcss_notch_frequencies;
581 char input_name[NAME_SIZE];
582 char output_name[NAME_SIZE];
583 unsigned int notch = 0;
584
585 snprintf(input_name, sizeof(input_name), "%s", graph_input);
586 while (*cursor) {
587 char *end;
588 double frequency = strtod(cursor, &end);
589 unsigned int section;
590 if (end == cursor) {
591 ++cursor;
592 continue;
593 }
594 cursor = end;
595 if (frequency < 50.0 || frequency > 300.0)
596 continue;
597 /* Higher order supplies 50 dB rejection at the TIA-603
598 * frequency-tolerance edges without widening into speech. */
599 for (section = 0; section < CTCSS_NOTCH_SECTIONS; ++section) {
600 snprintf(output_name, sizeof(output_name), "ctn%u_%u", notch,
601 section);
602 if (graph_append(graph, size,
603 "[%s]bandreject=f=%.9g:t=h:w=%.9g:r=f64[%s];",
604 input_name, frequency, cfg->ctcss_notch_width_hz,
605 output_name) < 0)
606 return AVERROR(ENOSPC);
607 snprintf(input_name, sizeof(input_name), "%s", output_name);
608 }
609 ++notch;
610 }
611 if (notch) {
612 snprintf(next, sizeof(next), "%s", input_name);
613 graph_input = next;
614 }
615 } else if (cfg->ctcss_filter_mode == TXAGC_CTCSS_FILTER_HIGHPASS &&
616 cfg->ctcss_highpass_hz > 0.0) {
617 if (graph_append(graph, size,
618 "[%s]acrossover=split=%.9g:order=20th[ctlow][cthigh];"
619 "[ctlow]anullsink;",
620 graph_input, cfg->ctcss_highpass_hz) < 0)
621 return AVERROR(ENOSPC);
622 graph_input = "cthigh";
623 }
624 if (graph_append(graph, size,
625 "[%s]asplit=2[programin][metertap];"
626 "[metertap]astats=metadata=1:reset=1:measure_perchannel=none:measure_"
627 "overall=Peak_level+RMS_level[in_meter];"
628 "[programin]volume=%.12g[%s];",
629 graph_input, db_to_linear(cfg->input_gain_db), current) < 0) {
630 return AVERROR(ENOSPC);
631 }
632
633 for (order_index = 0; order_index < cfg->stage_count; ++order_index) {
634 int enabled =
635 (cfg->stage_order[order_index] == TXAGC_STAGE_AGC && cfg->agc_enabled) ||
636 (cfg->stage_order[order_index] == TXAGC_STAGE_EXPANDER &&
637 cfg->expander_enabled) ||
638 (cfg->stage_order[order_index] == TXAGC_STAGE_COMPRESSOR &&
639 cfg->compressor_enabled) ||
640 (cfg->stage_order[order_index] == TXAGC_STAGE_LIMITER &&
641 cfg->limiter_enabled) ||
642 (cfg->stage_order[order_index] == TXAGC_STAGE_EQUALIZER &&

791 unsigned int sample_rate)

796 AVFilterInOut *outputs = NULL;
797 AVFilterInOut *meter_input = NULL;
798 AVFilterInOut *meter_tail = NULL;
799 AVFilterInOut *cleanup_input;
800 AVFilterContext **cleanup_sink_slots[] = {
801 &state->cleanup_pre_sink, &state->cleanup_pre_5_8_sink,
802 &state->cleanup_pre_8_plus_sink, &state->cleanup_post_5_8_sink,
803 &state->cleanup_post_8_plus_sink,
804 };

806 "cleanup_pre_sink", "cleanup_pre_5_8_sink", "cleanup_pre_8_plus_sink",
807 "cleanup_post_5_8_sink", "cleanup_post_8_plus_sink",
808 };
809 const char *cleanup_pad_names[] = {
810 "cleanup_pre_meter", "cleanup_pre_5_8_meter", "cleanup_pre_8_plus_meter",
811 "cleanup_post_5_8_meter", "cleanup_post_8_plus_meter",
812 };
813 char args[256];
814 char description[GRAPH_SIZE];
815 int result;
816
817 free_graph(state);
818 result = build_description(description, sizeof(description), config, sample_rate);
819 if (result < 0) {

821 }
822 state->graph = avfilter_graph_alloc();

824 return AVERROR(ENOMEM);
825 }
826 source_filter = avfilter_get_by_name("abuffer");
827 sink_filter = avfilter_get_by_name("abuffersink");
828 snprintf(args, sizeof(args),
829 "time_base=1/%u:sample_rate=%u:sample_fmt=dbl:channel_layout=mono", sample_rate,
830 sample_rate);

832 state->graph);
833 if (result < 0) {
834 goto fail;
835 }
836 if (config->post_limiter_lowpass_enabled) {
837 for (size_t index = 0; index < 5; ++index) {
838 AVFilterContext **sink_slot = cleanup_sink_slots[index];
839 result = avfilter_graph_create_filter(sink_slot, sink_filter,
840 cleanup_filter_names[index], NULL,
841 NULL, state->graph);
842 if (result < 0) {
843 goto fail;
844 }
845 set_double_sample_format(*sink_slot);
846 }
847 }
848 result = avfilter_graph_create_filter(&state->sink, sink_filter, "sink", NULL, NULL,
849 state->graph);
850 if (result < 0) {
851 goto fail;
852 }
853 result = avfilter_graph_create_filter(&state->meter_sink, sink_filter, "meter_sink", NULL,
854 NULL, state->graph);
855 if (result < 0) {
856 goto fail;
857 }
858 set_double_sample_format(state->sink);
859 set_double_sample_format(state->meter_sink);
860
861 outputs = avfilter_inout_alloc();

864 if (!outputs || !inputs || !meter_input) {
865 result = AVERROR(ENOMEM);
866 goto fail;
867 }
868 outputs->name = av_strdup("in");
869 outputs->filter_ctx = state->source;
870 outputs->pad_idx = 0;
871 inputs->name = av_strdup("out");
872 inputs->filter_ctx = state->sink;
873 inputs->pad_idx = 0;
874 inputs->next = meter_input;
875 meter_tail = meter_input;
876 meter_input->name = av_strdup("in_meter");
877 meter_input->filter_ctx = state->meter_sink;
878 meter_input->pad_idx = 0;

880 for (size_t index = 0; index < 5; ++index) {
881 cleanup_input = avfilter_inout_alloc();
882 if (!cleanup_input) {
883 result = AVERROR(ENOMEM);
884 goto fail;
885 }
886 meter_tail->next = cleanup_input;
887 meter_tail = cleanup_input;
888 cleanup_input->name = av_strdup(cleanup_pad_names[index]);
889 cleanup_input->filter_ctx = *cleanup_sink_slots[index];
890 cleanup_input->pad_idx = 0;
891 }
892 }
893 result = avfilter_graph_parse_ptr(state->graph, description, &inputs, &outputs, NULL);
894 if (result < 0) {
895 goto fail;
896 }
897 result = avfilter_graph_config(state->graph, NULL);
898 if (result < 0) {
899 goto fail;
900 }
901 state->fifo = av_audio_fifo_alloc(AV_SAMPLE_FMT_DBL, 1, (int)(sample_rate / 2));
902 if (!state->fifo) {
903 result = AVERROR(ENOMEM);
904 goto fail;
905 }
906 memcpy(&state->config, config, sizeof(*config));
907 state->sample_rate = sample_rate;
908 state->configured = 1;
909 state->failed = 0;
910 state->latency_samples = 0;
911 state->buffered_samples = 0;
912 state->output_started = 0;
913 state->startup_fill_samples = 0;
914 avfilter_inout_free(&inputs);
915 avfilter_inout_free(&outputs);
916 return 0;
917
918fail:
919 avfilter_inout_free(&inputs);
920 avfilter_inout_free(&outputs);
921 free_graph(state);
922 state->failed = result;
923 return result;
924}

971int txagc_avfilter_process(struct txagc_avfilter *state, const struct txagc_config *config,
972 double *samples, size_t count, unsigned int sample_rate)
973{
974 AVFrame *input = NULL;
975 AVFrame *output = NULL;
976 int available;
977 int result;
978 int written;
979 double *output_pointer;
980 size_t copied = 0;
981
982 /* Configurations are zero-initialized before fields are assigned, making
983 * their padding deterministic and this real-time comparison well-defined. */
984 if (!state->configured || state->sample_rate != sample_rate ||
985 memcmp(&state->config, config, sizeof(*config)) != 0) { /* NOLINT */
986 result = configure(state, config, sample_rate);
987 if (result < 0) {
988 return result;
989 }
990 }
991 input = av_frame_alloc();
992 output = av_frame_alloc();
993 if (!input || !output) {
994 result = AVERROR(ENOMEM);
995 goto done;
996 }
997 input->format = AV_SAMPLE_FMT_DBL;
998 input->sample_rate = (int)sample_rate;
999 input->nb_samples = (int)count;
1000 av_channel_layout_default(&input->ch_layout, 1);
1001 result = av_frame_get_buffer(input, 0);
1002 if (result < 0) {
1003 goto done;
1004 }
1005 /* Asterisk/usbradio uses floating point with 16-bit PCM units while
1006 * libavfilter's DBL format uses the normalized -1.0 .. +1.0 convention. */
1007 for (size_t index = 0; index < count; ++index) {
1008 ((double *)input->data[0])[index] = samples[index] / 32768.0;
1009 }
1010 result = av_buffersrc_add_frame_flags(state->source, input, AV_BUFFERSRC_FLAG_KEEP_REF);
1011 if (result < 0) {
1012 goto done;
1013 }
1014 state->input_samples += count;
1015 while ((result = av_buffersink_get_frame(state->meter_sink, output)) >= 0) {
1016 update_astats(state, output, 1);
1017 av_frame_unref(output);
1018 }
1019 if (result != AVERROR(EAGAIN) && result != AVERROR_EOF) {
1020 goto done;
1021 }
1022 if (config->post_limiter_lowpass_enabled) {
1023 result = drain_cleanup_meter(state, state->cleanup_pre_sink, output,
1024 CLEANUP_PRE_FULL);
1025 if (result < 0)
1026 goto done;
1027 result = drain_cleanup_meter(state, state->cleanup_pre_5_8_sink, output,
1028 CLEANUP_PRE_5_8);
1029 if (result < 0)
1030 goto done;
1031 result = drain_cleanup_meter(state, state->cleanup_pre_8_plus_sink, output,
1032 CLEANUP_PRE_8_PLUS);
1033 if (result < 0)
1034 goto done;
1035 result = drain_cleanup_meter(state, state->cleanup_post_5_8_sink, output,
1036 CLEANUP_POST_5_8);
1037 if (result < 0)
1038 goto done;
1039 result = drain_cleanup_meter(state, state->cleanup_post_8_plus_sink, output,
1040 CLEANUP_POST_8_PLUS);
1041 if (result < 0)
1042 goto done;
1043 }
1044
1045 while ((result = av_buffersink_get_frame(state->sink, output)) >= 0) {
1046 void *fifo_data[1];
1047 update_astats(state, output, 0);
1048 if (av_audio_fifo_realloc(state->fifo, av_audio_fifo_size(state->fifo) +
1049 output->nb_samples) < 0) {
1050 result = AVERROR(ENOMEM);
1051 goto done;
1052 }
1053 output_pointer = (double *)output->data[0];
1054 fifo_data[0] = output_pointer;
1055 written = av_audio_fifo_write(state->fifo, fifo_data, output->nb_samples);
1056 if (written != output->nb_samples) {
1057 result = AVERROR(EIO);
1058 goto done;
1059 }
1060 av_frame_unref(output);
1061 }
1062 if (result != AVERROR(EAGAIN) && result != AVERROR_EOF) {
1063 goto done;
1064 }
1065 available = av_audio_fifo_size(state->fifo);
1066 copied = available < (int)count ? (size_t)available : count;
1067 if (!state->output_started) {
1068 size_t fill = count - copied;
1069 memset(samples, 0, fill * sizeof(*samples));
1070 state->startup_fill_samples += fill;
1071 state->underrun_samples += fill;
1072 if (copied) {
1073 output_pointer = samples + fill;
1074 av_audio_fifo_read(state->fifo, (void **)&output_pointer, (int)copied);
1075 for (size_t index = fill; index < count; ++index) {
1076 samples[index] *= 32768.0;
1077 }
1078 state->latency_samples = (unsigned int)state->startup_fill_samples;
1079 state->output_started = 1;
1080 }
1081 } else {
1082 if (copied) {
1083 output_pointer = samples;
1084 av_audio_fifo_read(state->fifo, (void **)&output_pointer, (int)copied);
1085 for (size_t index = 0; index < copied; ++index) {
1086 samples[index] *= 32768.0;
1087 }
1088 }
1089 if (copied < count) {
1090 memset(samples + copied, 0, (count - copied) * sizeof(*samples));
1091 state->underrun_samples += count - copied;
1092 state->runtime_underrun_samples += count - copied;
1093 }
1094 }
1095 state->buffered_samples = (unsigned int)av_audio_fifo_size(state->fifo);
1096 state->output_samples += copied;
1097 result = 0;
1098
1099done:
1100 av_frame_free(&input);
1101 av_frame_free(&output);
1102 return result;
1103}
