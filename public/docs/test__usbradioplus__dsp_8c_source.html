133static void simulate_clock_drift(double producer_ppm)
134{
135 struct urp_clock_recovery clock = {0};
136 double app_phase =
137 (URP_FIFO_TARGET_NORMAL + URP_NATIVE_SAMPLES - 1U) / URP_NATIVE_SAMPLES - 1U;
138 double native_fifo = 0.0;
139 size_t target = URP_FIFO_TARGET_NORMAL;
140 unsigned int app_frames = 0, underflows = 0;
141 int tick;
142 for (tick = 0; tick < 60000; ++tick) { /* Twenty minutes at 20 ms. */
143 double correction;
144 app_phase += 1.0 + producer_ppm / 1000000.0;
145 while (app_phase >= 1.0) {
146 app_frames++;
147 app_phase -= 1.0;
148 }
149 correction = urp_clock_recovery_update(
150 &clock, (size_t)native_fifo + app_frames * URP_NATIVE_SAMPLES,
151 target + URP_FIFO_TARGET_STEP);
152 while (native_fifo < target && app_frames) {
153 native_fifo += 960.0 * (1.0 + correction);
154 app_frames--;
155 }
156 if (native_fifo >= 960.0)
157 native_fifo -= 960.0;
158 else {
159 underflows++;
160 target = target < URP_FIFO_TARGET_MAX - URP_FIFO_TARGET_STEP
161 ? target + URP_FIFO_TARGET_STEP
162 : URP_FIFO_TARGET_MAX;
163 urp_clock_recovery_reset(&clock);
164 }
165 }
166 printf("clock drift %+.0f ppm: underruns %u, app frames %u, native %.1f, correction %.6f\n",
167 producer_ppm, underflows, app_frames, native_fifo, clock.correction);
168 assert(underflows <= 1);
169 assert(app_frames < URP_PROGRAM_QUEUE_FRAMES);
170 assert(native_fifo < URP_NATIVE_FIFO_SAMPLES);
171}

184static void simulate_src_clock_drift(double producer_ppm, size_t target_samples)
185{
186 struct urp_clock_recovery clock = {0};
187 struct urp_src *src = urp_src_create(0, 1);
188 int16_t input[URP_LINK_SAMPLES], output[URP_NATIVE_SAMPLES * 2];
189 double phase = (target_samples + URP_NATIVE_SAMPLES - 1U) / URP_NATIVE_SAMPLES - 1U;
190 size_t native_count = 0;
191 unsigned int app_frames = 0, underflows = 0;
192 int primed = 0, tick;
193 assert(src);
194 tone(input, URP_LINK_SAMPLES, URP_RATE_LINK, 1000.0, 5000.0);
195 for (tick = 0; tick < 6000; ++tick) {
196 phase += 1.0 + producer_ppm / 1000000.0;
197 while (phase >= 1.0) {
198 app_frames++;
199 phase -= 1.0;
200 }
201 while (native_count < target_samples && app_frames) {
202 double correction = urp_clock_recovery_update(
203 &clock, native_count + app_frames * URP_NATIVE_SAMPLES,
204 target_samples + URP_FIFO_TARGET_STEP);
205 size_t used = 0, made = 0;
206 assert(!urp_src_process(src, input, URP_LINK_SAMPLES, output,
207 URP_NATIVE_SAMPLES * 2, 6.0 * (1.0 + correction),
208 &used, &made));
209 assert(used == URP_LINK_SAMPLES);
210 native_count += made;
211 assert(native_count < URP_NATIVE_FIFO_SAMPLES);
212 app_frames--;
213 }
214 if (!primed && native_count >= target_samples)
215 primed = 1;
216 if (primed) {
217 if (native_count >= URP_NATIVE_SAMPLES)
218 native_count -= URP_NATIVE_SAMPLES;
219 else {
220 underflows++;
221 primed = 0;
222 }
223 }
224 }
225 assert(underflows == 0);
226 assert(primed);
227 urp_src_destroy(src);
228}
