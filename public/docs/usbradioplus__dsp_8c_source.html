63double urp_clock_recovery_update(struct urp_clock_recovery *clock, size_t queued_samples,
64 size_t target_samples)
65{
66 double error, desired;
67
68 if (!clock || !target_samples)
69 return 0.0;
70 error = ((double)target_samples - (double)queued_samples) / (double)target_samples;
71 /* Filter scheduler jitter, then use a small integral term to follow a
72 * persistent oscillator mismatch without requiring extra FIFO latency. */
73 clock->filtered_error += (error - clock->filtered_error) * 0.10;
74 clock->integral_error += clock->filtered_error * 0.00002;
75 if (clock->integral_error > URP_CLOCK_MAX_CORRECTION)
76 clock->integral_error = URP_CLOCK_MAX_CORRECTION;
77 if (clock->integral_error < -URP_CLOCK_MAX_CORRECTION)
78 clock->integral_error = -URP_CLOCK_MAX_CORRECTION;
79 desired = clock->filtered_error * 0.006 + clock->integral_error;
80 if (desired > URP_CLOCK_MAX_CORRECTION)
81 desired = URP_CLOCK_MAX_CORRECTION;
82 if (desired < -URP_CLOCK_MAX_CORRECTION)
83 desired = -URP_CLOCK_MAX_CORRECTION;
84 /* Smooth ratio changes so clock correction cannot modulate speech abruptly. */
85 clock->correction += (desired - clock->correction) * 0.10;
86 return clock->correction;
87}

136int urp_src_process(struct urp_src *src, const int16_t *input, size_t input_count, int16_t *output,
137 size_t output_capacity, double ratio, size_t *input_used,
138 size_t *output_generated)
139{
140 SRC_DATA data;
141 size_t i, in_values, out_values;
142 int result;
143 if (!src || !input || !output || ratio <= 0.0)
144 return -1;
145 in_values = input_count * src->channels;
146 out_values = output_capacity * src->channels;
147 if (in_values > src->input_capacity) {
148 float *p = URP_REALLOC(src->input, in_values * sizeof(*p));
149 if (!p)
150 return -1;
151 src->input = p;
152 src->input_capacity = in_values;
153 }
154 if (out_values > src->output_capacity) {
155 float *p = URP_REALLOC(src->output, out_values * sizeof(*p));
156 if (!p)
157 return -1;
158 src->output = p;
159 src->output_capacity = out_values;
160 }
161 src_short_to_float_array(input, src->input, (int)in_values);
162 memset(&data, 0, sizeof(data));
163 data.data_in = src->input;
164 data.data_out = src->output;
165 data.input_frames = (long)input_count;
166 data.output_frames = (long)output_capacity;
167 data.src_ratio = ratio;
168 result = src_process(src->state, &data);
169 if (result)
170 return result;
171 src_float_to_short_array(src->output, output,
172 (int)(data.output_frames_gen * src->channels));
173 if (input_used)
174 *input_used = (size_t)data.input_frames_used;
175 if (output_generated)
176 *output_generated = (size_t)data.output_frames_gen;
177 for (i = data.output_frames_gen * src->channels; i < out_values; ++i)
178 output[i] = 0;
179 return 0;
180}

182int urp_rate_convert(struct urp_src *src, const int16_t *input, size_t input_count,
183 unsigned int input_rate, int16_t *output, size_t output_capacity,
184 unsigned int output_rate, size_t *input_used, size_t *output_generated)
185{
186 size_t copied;
187 if (!input || !output || !input_rate || !output_rate)
188 return -1;
189 if (input_rate != output_rate) {
190 return urp_src_process(src, input, input_count, output, output_capacity,
191 (double)output_rate / input_rate, input_used,
192 output_generated);
193 }
194 /* Matching rates need no converter state, allocation, or filter delay. */
195 copied = input_count < output_capacity ? input_count : output_capacity;
196 memmove(output, input, copied * sizeof(*output));
197 if (copied < output_capacity)
198 memset(output + copied, 0, (output_capacity - copied) * sizeof(*output));
199 if (input_used)
200 *input_used = copied;
201 if (output_generated)
202 *output_generated = copied;
203 return input_count <= output_capacity ? 0 : -1;
204}

244int urp_echo_remove(struct urp_echo_replacer *s, int16_t *mixed, int16_t *matched_native,
245 double minimum_correlation)
246{
247 double best_corr = -1.0, best_scale = 0.0;
248 unsigned int best = 0, n;
249 int found = 0;
250 if (!s || !mixed || !matched_native)
251 return 0;
252 for (n = 0; n < URP_ECHO_HISTORY_FRAMES; ++n) {
253 const struct urp_echo_frame *f = &s->history[n];
254 double xy = 0.0, xx = 0.0, yy = 0.0, corr, scale;
255 size_t i;
256 if (!f->sequence)
257 continue;
258 for (i = 0; i < URP_LINK_SAMPLES; ++i) {
259 double x = f->link[i], y = mixed[i];
260 xx += x * x;
261 yy += y * y;
262 xy += x * y;
263 }
264 if (xx < 1.0 || yy < 1.0)
265 continue;
266 corr = xy / sqrt(xx * yy);
267 scale = xy / xx;
268 if (corr > best_corr && scale > 0.25 && scale < 2.5) {
269 best_corr = corr;
270 best_scale = scale;
271 best = n;
272 found = 1;
273 }
274 }
275 if (!found || best_corr < minimum_correlation) {
276 s->misses++;
277 s->last_correlation = found ? best_corr : 0.0;
278 memset(matched_native, 0, URP_NATIVE_SAMPLES * sizeof(*matched_native));
279 return 0;
280 }
281 {
282 const struct urp_echo_frame *f = &s->history[best];
283 size_t i;
284 for (i = 0; i < URP_LINK_SAMPLES; ++i)
285 mixed[i] = saturate(mixed[i] - f->link[i] * best_scale);
286 for (i = 0; i < URP_NATIVE_SAMPLES; ++i)
287 matched_native[i] = saturate(f->native[i] * best_scale);
288 s->last_delay_frames = (int)(s->sequence - f->sequence);
289 }
290 s->last_scale = best_scale;
291 s->last_correlation = best_corr;
292 s->matches++;
293 return 1;
294}
