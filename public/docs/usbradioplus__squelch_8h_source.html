33static inline int urp_micor_squelch_update(struct urp_micor_squelch *state, int squelched,
34 double sample_power, uint32_t open_level,
35 uint32_t hysteresis)
36{
37 /* Motorola US3628058: separate fast detector, finite charging path,
38 * fixed long-tail charge, and strong-signal defeat. A sudden loss must
39 * cross the charging region before the capacitor can reopen the gate.
40 * Coefficients are 1-exp(-1/(48000*tau)); no per-sample transcendental
41 * functions or frame-boundary decisions are needed.
42 *
43 * Detector tau=5 ms, defeat tau=1 ms, recharge tau=15 ms. The detector
44 * smooths random noise troughs that otherwise repeatedly recharge the
45 * long hold even with no carrier. Long decay is calibrated
46 * to 150 ms from full charge to the 3.8/5.5 comparator fraction. These
47 * are behavioral constants, not inferred IC resistor measurements. */
48 const double detector_alpha = 0.00415799815489004;
49 const double defeat_alpha = 0.02061781866875989;
50 const double charge_alpha = 0.00138792482909171;
51 const double release_alpha = 0.00005135243496113;
52 const double idle_alpha = 0.00003255155352044;
53 const double hold_threshold = 3.8 / 5.5;
54 double open_power = (double)open_level * open_level;
55 double limit = (double)open_level + (squelched ? 0.0 : hysteresis);
56 double limit_power = limit * limit;
57 double strong_power, direct_power;
58
59 /* Settle once, without treating a zero-filled FIR history as a carrier.
60 * Keep this running average independent of caller buffer partitioning. */
61 if (state->settling_samples < URP_MICOR_SETTLE_SAMPLES) {
62 state->settling_samples++;
63 state->noise_power += (sample_power - state->noise_power) / state->settling_samples;
64 state->idle_power = state->noise_power;
65 return 1;
66 }
67 state->noise_power += detector_alpha * (sample_power - state->noise_power);
68 if (squelched && state->noise_power >= open_power)
69 state->idle_power += idle_alpha * (state->noise_power - state->idle_power);
70 strong_power = state->idle_power * 0.01;
71
72 /* Preserve the calibrated opening threshold. Interpolate the direct
73 * comparator between the manual's 2.8 V charging and 5.0 V defeat
74 * points; its 3.8 V position is 5/11 of that interval. This maps the
75 * circuit topology onto our noise-power units, not literal voltages. */
76 direct_power = strong_power < limit_power
77 ? limit_power + (strong_power - limit_power) * (5.0 / 11.0)
78 : limit_power;
79 if (state->noise_power <= strong_power) {
80 state->hold_charge -= defeat_alpha * state->hold_charge;
81 } else if (state->noise_power < limit_power) {
82 state->hold_charge += charge_alpha * (1.0 - state->hold_charge);
83 } else {
84 state->hold_charge -= release_alpha * state->hold_charge;
85 }
86 /* Inaudible residual state must not enter slow subnormal arithmetic
87 * during hours of idle or a fully quiet discriminator. */
88 if (state->hold_charge < 1.0e-12)
89 state->hold_charge = 0.0;
90 if (state->noise_power < 1.0e-12)
91 state->noise_power = 0.0;
92 return !(state->noise_power < direct_power || state->hold_charge > hold_threshold);
93}
