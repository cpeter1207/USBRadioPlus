31def function_definition(source, name):
32 """Extract one C function body for a focused implementation invariant.
33
34 @param source Processing source or source text, as declared.
35 @param name Helper, source file, or symbol name selected by this test.
36 """
37 match = re.search(
38 rf"^(?:static )?[^;\n]*\b{re.escape(name)}\‍([^;]*\‍)\n\{{", source, re.MULTILINE
39 )
40 assert match, name
41 end = source.index("\n}\n", match.start()) + 3
42 return source[match.start() : end]
