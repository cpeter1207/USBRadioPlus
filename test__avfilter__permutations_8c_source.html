121static int run_permutation(const enum txagc_stage *order, unsigned int number,
122 unsigned int block_count, unsigned int sample_rate,
123 double *total_seconds)
124{
125 struct txagc_avfilter state;
126 struct txagc_config cfg;
127 double samples[BLOCK];
128 struct timespec begin, end;
129 unsigned int block, index;
130 unsigned int count = sample_rate / 50;
131
132 configure(&cfg);
133 /* Rotate all four single-/three-band combinations across the complete stage
134 * order set; both sample rates then exercise exactly the same combinations. */
135 cfg.compressor_bands = 1 + 2 * (int)(number % 2);
136 cfg.limiter_bands = 1 + 2 * (int)((number / 2) % 2);
137 memcpy(cfg.stage_order, order, sizeof(cfg.stage_order));
138 txagc_avfilter_init(&state);
139 clock_gettime(CLOCK_MONOTONIC, &begin);
140 for (block = 0; block < block_count; ++block) {
141 for (index = 0; index < count; ++index) {
142 double time = (double)(block * count + index) / sample_rate;
143 samples[index] = 6000.0 * sin(2.0 * M_PI * 900.0 * time);
144 }
145 if (txagc_avfilter_process(&state, &cfg, samples, count, sample_rate) < 0) {
146 fprintf(stderr, "permutation %u failed at block %u\n", number, block);
147 txagc_avfilter_destroy(&state);
148 return -1;
149 }
150 }
151 clock_gettime(CLOCK_MONOTONIC, &end);
152 *total_seconds += end.tv_sec - begin.tv_sec + (end.tv_nsec - begin.tv_nsec) / 1000000000.0;
153 txagc_avfilter_destroy(&state);
154 return 0;
155}

164static int permute(enum txagc_stage *order, unsigned int at, unsigned int *number, double *seconds)
165{
166 unsigned int index;
167 if (at == TXAGC_MAX_DYNAMICS_STAGES) {
168 unsigned int current = (*number)++;
169 if (run_permutation(order, current, BLOCKS_PER_PERMUTATION, 8000, seconds))
170 return -1;
171 return run_permutation(order, current, BLOCKS_PER_PERMUTATION, RATE, seconds);
172 }
173 for (index = at; index < TXAGC_MAX_DYNAMICS_STAGES; ++index) {
174 enum txagc_stage swap = order[at];
175 order[at] = order[index];
176 order[index] = swap;
177 if (permute(order, at + 1, number, seconds))
178 return -1;
179 swap = order[at];
180 order[at] = order[index];
181 order[index] = swap;
182 }
183 return 0;
184}
