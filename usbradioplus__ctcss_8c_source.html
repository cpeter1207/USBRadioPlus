84void urp_ctcss_legacy_scaled_levels(double frequency, int filter_250, int tone_gain_q8,
85 int output_gain_q8, double *amplitude, double *bias)
86{

89 long difference = filter_250 ? bias_250[index] : bias_215[index];
90 long positive = difference > 0 ? maximum : maximum + difference;

102void urp_ctcss_generate(struct urp_ctcss_generator *generator, double *output, size_t count,
103 double frequency, double peak, int enabled, int phase_reverse)
104{
105 size_t i;
106 double step, sine, cosine, sine_step, cosine_step;
107
108 if (!enabled || frequency <= 0.0 || peak <= 0.0) {
109 memset(output, 0, count * sizeof(*output));
110 return;
111 }
112 if (phase_reverse) {
113 /* XPMR truncates 240 degrees to 170 positions in its 256-step table. */
114 generator->phase +=
115 2.0 * URP_PI * URP_LEGACY_PHASE_REVERSE_STEPS / URP_LEGACY_SINE_STEPS;
116 generator->phase = fmod(generator->phase, 2.0 * URP_PI);
117 }
118 step = 2.0 * URP_PI * urp_ctcss_legacy_frequency(frequency) / URP_CTCSS_RATE;
119 sine = sin(generator->phase);
120 cosine = cos(generator->phase);
121 sine_step = sin(step);
122 cosine_step = cos(step);
123 for (i = 0; i < count; ++i) {
124 double next_sine, next_cosine;
125 output[i] = peak * sine;
126 next_sine = sine * cosine_step + cosine * sine_step;
127 next_cosine = cosine * cosine_step - sine * sine_step;
128 sine = next_sine;
129 cosine = next_cosine;
130 }
131 generator->phase = fmod(generator->phase + step * (double)count, 2.0 * URP_PI);
132}
